// ---------------------------------------------------------------------------
// Texto da fatura em PDF (auditoria 4.3, A-09)
//
// O cabeçalho de uma fatura traz nome, CPF, endereço, limite e linha digitável.
// Nada disso ajuda a ler lançamento, e tudo sairia para o OpenRouter.

const VALOR = /\d{1,3}(?:\.\d{3})*,\d{2}/;
// o pdf-cliente monta a linha pela coordenada Y: num PDF de duas colunas, um
// lançamento divide a linha com "Limite total" — nela só o pedaço sai
const pareceLancamento = (l: string) => /(^|\D)\d{2}\/\d{2}(\D|$)/.test(l) && VALOR.test(l);

const BLOCO_NUMERICO = /\d[\d. ]*\d/g;
const CPF = /[\d*•Xx]{3}\.[\d*•Xx]{3}\.[\d*•Xx]{3}-[\d*•Xx]{2}|\b\d{11}\b/g;
// 16 posições em grupos de 4 (dígito ou máscara), ou 13 a 19 dígitos corridos
const CARTAO = /(?:[\dXx*•]{4}[ .-]){3}(\d{4})\b|\b\d{9,15}(\d{4})\b/g;
const CEP = /\b\d{5}-\d{3}\b/g;
const EMAIL = /[\w.+-]+@[\w-]+(?:\.[\w-]+)+/g;
const ENDERECO = /^(rua|r\.|av\.?|avenida|alameda|al\.|travessa|tv\.|rodovia|estrada|praça|praca)\s/i;
const SALDO = /\b(limite|saldo|dispon[ií]vel)\b/i;
// até o primeiro valor; sem valor, até o próximo número (uma data de lançamento)
const SALDO_TRECHO = /\b(?:limite|saldo|dispon[ií]vel)\b[^\d]*?(?:(?:R\$\s*)?\d{1,3}(?:\.\d{3})*,\d{2}|(?=\d)|$)/gi;
// só no começo da linha ou seguido de ':' — "RESTAURANTE BOM NOME" não é rótulo
const NOME = /^(?:nome(?: do (?:titular|cliente))?|titular|cliente|portador)\b|\b(?:nome|titular|cliente|portador)\s*:/i;
const NOME_TRECHO = /\b(?:nome(?: do (?:titular|cliente))?|titular|cliente|portador)\s*:?\s*[A-Za-zÀ-ÿ'. ]*/gi;

/** O trecho de fatura sem dado pessoal, pronto para o prompt de extração. */
export function textoDaFaturaParaIA(trecho: string): string {
  return trecho
    .split('\n')
    .flatMap((bruta) => {
      const l = bruta
        .replace(BLOCO_NUMERICO, (b) => (b.replace(/\D/g, '').length >= 20 ? '[código]' : b))
        .replace(CPF, '[CPF]')
        .replace(CARTAO, (_, a?: string, b?: string) => `•••• ${a ?? b}`);
      // os globais acima têm lastIndex; o teste usa cópias sem a flag g
      const sensivel =
        l.includes('[código]') ||
        new RegExp(CEP.source).test(l) ||
        new RegExp(EMAIL.source).test(l) ||
        SALDO.test(l) ||
        NOME.test(l) ||
        ENDERECO.test(l);
      if (!sensivel) return [l];
      if (!pareceLancamento(l)) return [];
      return [
        l
          .replace(CEP, '[CEP]')
          .replace(EMAIL, '[e-mail]')
          .replace(SALDO_TRECHO, '[omitido] ')
          .replace(NOME_TRECHO, '[omitido] ')
          .replace(/\s+/g, ' ')
          .trim(),
      ];
    })
    .join('\n');
}

/** O prompt de extração (texto em `extracao.ts`) com o trecho já filtrado. */
export function promptDeExtracaoParaIA(trecho: string, comCabecalho: boolean, idx: number, n: number, anoDica: string): string {
  return promptExtracao(textoDaFaturaParaIA(trecho), comCabecalho, idx, n, anoDica);
}
```

/**
 * Limite de chamadas de IA por usuário (auditoria 4.3, A-10). Puro: o teste
 * importa. Quem conta e grava é `reservarChamadaIA`, em ia-uso-repo.ts.
 *
 * Os números protegem a ordem de grandeza do gasto, não um valor exato: uma
 * importação usa até 8 chamadas e um sync de 3 meses perto de 10.
 */

declare const marca: unique symbol;
/** Prova de que a chamada passou pela trava. Só `reservarChamadaIA` fabrica. */
export type ReservaIA = { readonly [marca]: true };

export const POR_10_MIN = 60;
export const POR_DIA = 400;

const DEZ_MIN_S = 10 * 60;
const DIA_S = 24 * 60 * 60;

/** Chamadas nas duas janelas e a mais antiga de cada uma. */
export type JanelaIA = { n10: number; antiga10: Date | null; n24: number; antiga24: Date | null };

/** Segundos até poder chamar de novo, ou null se pode agora. */
export function limiteDeIA(j: JanelaIA, agora: Date): number | null {
  // quando a mais antiga sai da janela, abre uma vaga
  const ate = (antiga: Date, janela: number) => Math.max(1, Math.ceil((antiga.getTime() + janela * 1000 - agora.getTime()) / 1000));
  const esperas = [
    j.n10 >= POR_10_MIN && j.antiga10 ? ate(j.antiga10, DEZ_MIN_S) : 0,
    j.n24 >= POR_DIA && j.antiga24 ? ate(j.antiga24, DIA_S) : 0,
  ];
  const maior = Math.max(...esperas);
  return maior > 0 ? maior : null;
}

export function textoLimiteIA(segundos: number): string {
  const min = Math.max(1, Math.ceil(segundos / 60));
  if (min >= 120) return `Muitas chamadas de IA hoje; tente de novo em ${Math.ceil(min / 60)} horas`;
  return `Muitas chamadas de IA em pouco tempo; tente de novo em ${min} ${min === 1 ? 'minuto' : 'minutos'}`;
}
```

/**
 * De quem é o pedido de login. Puro: o teste importa.
 *
 * O X-Forwarded-For é do cliente até um proxy de confiança reescrevê-lo: lido
 * sem essa garantia, cada tentativa vem de um "IP" novo e a espera nunca chega
 * (auditoria 4.3, A-07). Sem fonte confiável, todo pedido cai num balde só —
 * mais lento para o dono num ataque, mas forjar cabeçalho não ajuda.
 */

export const SEM_IP = 'sem-ip';

type Cabecalhos = { get(nome: string): string | null };

export function ipDaRequisicao(h: Cabecalhos, o: { vercel: boolean; ipHeader?: string }): string {
  // a borda da Vercel escreve o x-real-ip; o que o cliente mandou não passa
  if (o.vercel) return h.get('x-real-ip')?.trim() || SEM_IP;
  // o proxy de confiança acrescenta à direita; o que está antes veio do cliente
  if (o.ipHeader) return h.get(o.ipHeader)?.split(',').map((s) => s.trim()).filter(Boolean).at(-1) || SEM_IP;
  return SEM_IP;
}

// A cripto da senha vive em senha.ts, sem dependência de banco nem de runtime.

// ---------------------------------------------------------------------------
// Espera crescente por IP, e uma vaga por minuto para todos sob ataque
//
// A tentativa é reservada ANTES do scrypt, num comando só: o `on conflict do
// update` trava a linha, e o `where` deixa passar só quem não está em espera.
// Ler o bloqueio e gravar a falha em comandos separados deixava N pedidos em
// paralelo passarem juntos (auditoria 4.3, A-07).

// não é IP: não colide com nenhuma chave de IP nem com 'sem-ip'
const GLOBAL = '*';

/** Segundos que ainda faltam para esta chave poder tentar de novo (0 = pode). */
async function bloqueioRestante(chave: string): Promise<number> {
  const linhas = await db.execute<{ faltam: number }>(sql`
    select greatest(0, ceil(extract(epoch from (bloqueado_ate - now()))))::int as faltam
    from login_attempt
    where ip = ${chave} and bloqueado_ate is not null and bloqueado_ate > now()
  `);
  return linhas.rows[0]?.faltam ?? 0;
}

/**
 * Conta uma tentativa para o IP e já grava a espera da próxima (a mesma escada
 * de `esperaSegundos`). Em espera: não conta e devolve os segundos que faltam.
 */
export async function reservarTentativa(ip: string): Promise<number> {
  const r = await db.execute(sql`
    insert into login_attempt (ip, tentativas, bloqueado_ate, atualizado_em)
    values (${ip}, 1, null, now())
    on conflict (ip) do update set
      tentativas = login_attempt.tentativas + 1,
      bloqueado_ate = case
        when login_attempt.tentativas + 1 > ${LIVRES}::int
          then now() + make_interval(secs => least(power(2, login_attempt.tentativas + 1 - ${LIVRES}::int), ${TETO_S}::int))
        else null
      end,
      atualizado_em = now()
    where login_attempt.bloqueado_ate is null or login_attempt.bloqueado_ate <= now()
    returning tentativas
  `);
  if (r.rows.length) return 0;
  // a espera pode ter acabado entre os dois comandos: ainda assim, esta não passou
  return Math.max(1, await bloqueioRestante(ip));
}

/** Com o teto global ativo, uma tentativa por VAGA_GLOBAL_S para todos; senão, 0. */
export async function reservarVagaGlobal(): Promise<number> {
  const c = await db.execute<{ n: number }>(sql`
    select count(*)::int as n from login_falha where em > now() - interval '1 hour'
  `);
  if ((c.rows[0]?.n ?? 0) < FALHAS_GLOBAIS_HORA) return 0;

  const r = await db.execute(sql`
    insert into login_attempt (ip, tentativas, bloqueado_ate, atualizado_em)
    values (${GLOBAL}, 1, now() + make_interval(secs => ${VAGA_GLOBAL_S}::int), now())
    on conflict (ip) do update set
      tentativas = login_attempt.tentativas + 1,
      bloqueado_ate = now() + make_interval(secs => ${VAGA_GLOBAL_S}::int),
      atualizado_em = now()
    where login_attempt.bloqueado_ate is null or login_attempt.bloqueado_ate <= now()
    returning tentativas
  `);
  if (r.rows.length) return 0;
  return Math.max(1, await bloqueioRestante(GLOBAL));
}

/** Senha errada: entra na conta global; as de mais de um dia já não contam para nada. */

    with nova as (insert into login_falha default values returning id)
    delete from login_falha where em < now() - interval '1 day'

import { conferirSenha, limparFalhas, registrarFalhaGlobal, reservarTentativa, reservarVagaGlobal } from '@/lib/auth';
import { ipDaRequisicao } from '@/lib/ip';
import { obterOuCriarUsuario } from '@/lib/sessao-servidor';
import { COOKIE, DURACAO_S, assinar } from '@/lib/session';
import { env } from '@/env';

// scrypt precisa de node:crypto
export const runtime = 'nodejs';

/**
 * A espera por IP e o registro do usuário vivem no banco. Se o banco não
 * responde, o login não acontece — mas o erro devolvido diz isso, em vez de um
 * 500 genérico que manda você procurar bug no lugar errado.
 */
const bancoIndisponivel = () =>
  NextResponse.json({ erro: 'banco_indisponivel' }, { status: 503, headers: { 'Retry-After': '30' } });

export async function POST(req: NextRequest) {
  // o corpo antes da reserva: pedido sem senha não gasta tentativa
  let senha: unknown;
  try {
    senha = (await req.json())?.senha;
  } catch {
    senha = undefined;
  }
  if (typeof senha !== 'string' || senha.length === 0) {
    return NextResponse.json({ erro: 'senha_obrigatoria' }, { status: 400 });
  }

  const ip = ipDaRequisicao(req.headers, { vercel: env.VERCEL === '1', ipHeader: env.IP_HEADER });
  let faltam: number;
  try {
    faltam = (await reservarVagaGlobal()) || (await reservarTentativa(ip));
  } catch {
    return bancoIndisponivel();
  }
  if (faltam > 0) {
    return NextResponse.json(
      { erro: 'muitas_tentativas', tenteEm: faltam },
      { status: 429, headers: { 'Retry-After': String(faltam) } },
    );
  }

  const ok = await conferirSenha(senha, env.APP_PASSWORD_HASH);
  if (!ok) {
    try {
      await registrarFalhaGlobal();
    } catch {
      // a tentativa falhou de todo jeito; não deixar o registro virar 500
    }
    // mensagem genérica de propósito: não conta o que errou
    return NextResponse.json({ erro: 'credencial_invalida' }, { status: 401 });
  }

  let userId: string;
  try {
    await limparFalhas(ip);
    userId = await obterOuCriarUsuario();
  } catch {
    return bancoIndisponivel();
  }
  const exp = Math.floor(Date.now() / 1000) + DURACAO_S;
  const token = await assinar({ sub: userId, exp }, env.SESSION_SECRET);

  const res = NextResponse.json({ ok: true });
  res.cookies.set(COOKIE, token, {
    httpOnly: true,
    secure: env.NODE_ENV === 'production',
    sameSite: 'lax',
    path: '/',
    maxAge: DURACAO_S,
  });
  return res;
}
