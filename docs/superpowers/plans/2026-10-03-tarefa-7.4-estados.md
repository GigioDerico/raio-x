```ts
/**
 * Falha → texto que diz o que aconteceu e o que fazer (guia, "Estados que
 * toda tela precisa ter"). Puro: o teste importa, os componentes também.
 */
export type Falha = { tipo: 'rede' | 'sessao' | 'uso' | 'servidor'; texto: string };

const TEXTOS = {
  rede: 'Sem conexão com o servidor. Confira a internet e tente de novo.',
  sessao: 'Sua sessão expirou. Entre de novo.',
  servidor: 'O servidor não respondeu. Tente de novo em instantes.',
} as const;

// status em que a rota escreve o `detalhe` para o usuário (ErroDeUso)
const DE_USO = new Set([400, 404, 409, 415]);

export function mensagemDeFalha(e: { status?: number; detalhe?: unknown; erro?: unknown; online?: boolean }): Falha {
  // fetch que nem chegou ao servidor rejeita com TypeError
  if (e.online === false || e.erro instanceof TypeError) return { tipo: 'rede', texto: TEXTOS.rede };
  if (e.status === 401) return { tipo: 'sessao', texto: TEXTOS.sessao };
  if (e.status && DE_USO.has(e.status) && typeof e.detalhe === 'string' && e.detalhe.trim()) {
    return { tipo: 'uso', texto: e.detalhe.trim().slice(0, 200) };
  }
  return { tipo: 'servidor', texto: TEXTOS.servidor };
}

/** Resposta não-ok de uma rota do app, com o status e o `detalhe` que ela mandou. */
export class FalhaHttp extends Error {
  constructor(
    readonly status: number,
    readonly detalhe: unknown,
  ) {
    super(`falha ${status}`);
    this.name = 'FalhaHttp';
  }
}

export async function falhaDe(r: Response): Promise<FalhaHttp> {
  const d = (await r.json().catch(() => null)) as { detalhe?: unknown } | null;
  return new FalhaHttp(r.status, d?.detalhe);
}

/**
 * O que um `catch` pegou → Falha. Erro comum (não HTTP, não rede) é do próprio
 * cliente ("não achei texto neste PDF") e já foi escrito para o usuário.
 */
export function textoDoErro(e: unknown, online = true): Falha {
  if (e instanceof FalhaHttp) return mensagemDeFalha({ status: e.status, detalhe: e.detalhe, online });
  if (!online || e instanceof TypeError) return mensagemDeFalha({ erro: e, online });
  if (e instanceof Error && e.message) return { tipo: 'uso', texto: e.message };
  return mensagemDeFalha({});
}

export function avisoIgnorados(n: number): string | null {
  if (n <= 0) return null;
  return n === 1 ? '1 trecho não pôde ser lido e ficou de fora.' : `${n} trechos não puderam ser lidos e ficaram de fora.`;
}
```

```ts
import { paraQuery, rotuloMes, type Filtro } from './filtro';

/** "Nada em jul/26 nos cartões Itaú e Nubank" — diz qual filtro esvaziou a tela. */
export function fraseDoVazio(o: { mes: string | null; nomesCartoes: string[] }): string {
  const n = o.nomesCartoes;
  const cartoes =
    n.length === 0
      ? ''
      : n.length === 1
        ? ` no cartão ${n[0]}`
        : n.length <= 3
          ? ` nos cartões ${n.slice(0, -1).join(', ')} e ${n[n.length - 1]}`
          : ` em ${n.length} cartões`;
  if (o.mes) return `Nada em ${rotuloMes(o.mes)}${cartoes}`;
  return cartoes ? `Nada${cartoes}` : 'Nada por aqui';
}

/**
 * Query do link do vazio, mantendo o resto do filtro. Nunca vazia: sem nenhuma
 * chave na URL o middleware reaplica o filtro do cookie — justamente o que o
 * link quer trocar. `mes=` é lido como "o mais recente".
 */
export function queryDoVazio(f: Filtro, mudanca: 'todos-cartoes' | 'mes-recente'): string {
  const novo = mudanca === 'todos-cartoes' ? { ...f, cartoes: [] } : { ...f, mes: null };
  return paraQuery(novo) || 'mes=';
}

export type LinkVazio = { href: string; rotulo: string };

export function linksDoVazio(f: Filtro, caminho: string): LinkVazio[] {
  const links: LinkVazio[] = [];
  if (f.cartoes.length) links.push({ href: `${caminho}?${queryDoVazio(f, 'todos-cartoes')}`, rotulo: 'Ver todos os cartões' });
  if (f.mes && f.mes !== 'TODOS') links.push({ href: `${caminho}?${queryDoVazio(f, 'mes-recente')}`, rotulo: 'Voltar ao mês mais recente' });
  return links;
}

/** Os rótulos do filtro (apelido ou banco, nunca o last4), na ordem do filtro. */
export function nomesDosCartoes(ids: string[], opcoes: { id: string; nome: string }[]): string[] {
  const nome = new Map(opcoes.map((c) => [c.id, c.nome]));
  return ids.flatMap((id) => (nome.has(id) ? [nome.get(id)!] : []));
}
```

```tsx
'use client';

import { startTransition, useSyncExternalStore } from 'react';
import { useRouter } from 'next/navigation';

const assinar = (cb: () => void) => {
  window.addEventListener('online', cb);
  window.addEventListener('offline', cb);
  return () => {
    window.removeEventListener('online', cb);
    window.removeEventListener('offline', cb);
  };
};

/**
 * Corpo comum dos error.tsx. O detalhe fica no log do servidor; aqui, o que
 * ajuda a decidir: sem rede é com você, o resto é com o servidor (ou o banco
 * acordando) e tentar de novo costuma resolver.
 */
export default function ErroDaTela({ oQue, reset }: { oQue: string; reset: () => void }) {
  const router = useRouter();
  const online = useSyncExternalStore(assinar, () => navigator.onLine, () => true);

  return (
    <section className="cartao vazio" role="alert">
      <h2>Não consegui carregar {oQue}</h2>
      <p className="suave">
        {online
          ? 'O servidor não respondeu — pode ter sido o banco ou a rede. Tente de novo em instantes.'
          : 'Sem conexão com a internet. Quando ela voltar, tente de novo.'}
      </p>
      <button
        type="button"
        onClick={() =>
          // reset() sozinho só re-renderiza no cliente; o payload com erro fica
          // em cache. router.refresh() busca os dados de novo antes.
          startTransition(() => {
            router.refresh();
            reset();
          })
        }
      >
        Tentar de novo
      </button>
    </section>
  );
}
```
