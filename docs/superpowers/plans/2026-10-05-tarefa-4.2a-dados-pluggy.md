```ts
function mesSeguinte(mes: string): string {
  const [a, m] = mes.split('-').map(Number) as [number, number];
  return m === 12 ? `${a + 1}-01` : `${a}-${String(m + 1).padStart(2, '0')}`;
}

function menosDias(dia: string, n: number): string {
  const d = new Date(`${dia}T00:00:00Z`);
  d.setUTCDate(d.getUTCDate() - n);
  return d.toISOString().slice(0, 10);
}

/** O mesmo dia no mês seguinte, ou o último dia dele (31/jan → 28/fev). */
function mesmoDiaNoMesSeguinte(dia: string): string {
  const mes = mesSeguinte(dia.slice(0, 7));
  const [a, m] = mes.split('-').map(Number) as [number, number];
  const ultimo = new Date(Date.UTC(a, m, 0)).getUTCDate();
  return `${mes}-${String(Math.min(Number(dia.slice(8, 10)), ultimo)).padStart(2, '0')}`;
}

/**
 * A fatura aberta não vem em /bills, e os lançamentos dela chegam sem billId —
 * sem isto cairiam no mês da compra, não no mês em que se paga. Compra depois
 * do fechamento da última fatura conhecida é da seguinte. Sem fechamento, o
 * corte é o vencimento menos 7 dias (o intervalo comum entre os dois).
 */
export function mesDaFaturaAberta(faturas: readonly FaturaMapeada[], dataCompra: string): { mesRef: string; vencimento: string } | null {
  if (!faturas.length) return null;
  const ultima = [...faturas].sort((a, b) => a.vencimento.localeCompare(b.vencimento)).at(-1)!;
  const corte = ultima.fechamento ?? menosDias(ultima.vencimento, 7);
  if (dataCompra <= corte) return null;
  return { mesRef: mesSeguinte(ultima.mesRef), vencimento: mesmoDiaNoMesSeguinte(ultima.vencimento) };
}

/**
 * Duas faturas com vencimento no mesmo mês (31/mar e 1º/mar) disputariam o mesmo
 * mes_ref e alternariam total e vencimento a cada sync: a que vence depois vai
 * para o mês seguinte, em cadeia.
 */
export function mesesDasFaturas(faturas: readonly FaturaMapeada[]): FaturaMapeada[] {
  const ordenadas = [...faturas].sort((a, b) => a.vencimento.localeCompare(b.vencimento));
  const saida: FaturaMapeada[] = [];
  for (const f of ordenadas) {
    const anterior = saida.at(-1);
    saida.push(anterior && f.mesRef <= anterior.mesRef ? { ...f, mesRef: mesSeguinte(anterior.mesRef) } : f);
  }
  return saida;
}
```

export type CategoriaLida = { categoria: string | null; nome: string | null };

/** Resposta da IA → categoria e nome por item, na ordem; o que não vale vira nulo. */
export function lerCategorias(dados: unknown, n: number): CategoriaLida[] {
  const saida: CategoriaLida[] = Array.from({ length: n }, () => ({ categoria: null, nome: null }));
  const lista = (dados as { categorias?: unknown } | null)?.categorias;
  if (!Array.isArray(lista)) return saida;
  for (const c of lista) {
    const i = (c as { i?: unknown })?.i;
    if (typeof i !== 'number' || !Number.isInteger(i) || i < 0 || i >= n) continue;
    const cat = (c as { categoria?: unknown })?.categoria;
    const nome = (c as { nome?: unknown })?.nome;
    saida[i] = {
      categoria: typeof cat === 'string' && NOMES.includes(cat) ? cat : null,
      // texto do modelo, exibido na tela: uma linha, curto
      nome: typeof nome === 'string' && nome.trim() && !/[\r\n]/.test(nome) ? nome.trim().slice(0, 40) : null,
    };
  }
  return saida;
}

```ts
/**
 * Uma vez só: dá nome limpo aos lançamentos do Pluggy gravados antes da 4.2a.
 * Reprocessa dado antigo (custa IA), então só roda com autorização do dono.
 * Não mexe em categoria — só em desc_limpa, onde ainda está vazio.
 */
import { neon } from '@neondatabase/serverless';
import { lerCategorias, promptCategorizar, semDocumentos } from '../src/lib/ia';

const sql = neon(process.env.DATABASE_URL!);
const LOTE = 60;

async function pedir(prompt: string, modelo: string) {
  const r = await fetch('https://openrouter.ai/api/v1/chat/completions', {
    method: 'POST',
    headers: { authorization: `Bearer ${process.env.OPENROUTER_API_KEY}`, 'content-type': 'application/json' },
    body: JSON.stringify({ model: modelo, messages: [{ role: 'user', content: prompt }], response_format: { type: 'json_object' }, temperature: 0, usage: { include: true } }),
  });
  const corpo = (await r.json()) as { choices?: { message?: { content?: string } }[]; usage?: { prompt_tokens?: number; completion_tokens?: number; cost?: number } };
  if (!r.ok) throw new Error(`OpenRouter ${r.status}`);
  return corpo;
}

async function main() {
  if (!process.env.OPENROUTER_API_KEY) throw new Error('OPENROUTER_API_KEY vazia');
  const usuarios = await sql`select distinct user_id from txn where fonte = 'pluggy' and desc_limpa is null and excluido_em is null`;
  for (const { user_id } of usuarios) {
    const [cfg] = await sql`select modelo_ia from user_setting where user_id = ${user_id}`;
    const modelo = (cfg?.modelo_ia as string | undefined) || process.env.OPENROUTER_MODEL || 'anthropic/claude-sonnet-4.5';
    const pendentes = await sql`select id, data::text, valor::text, desc_origem from txn where user_id = ${user_id} and fonte = 'pluggy' and desc_limpa is null and excluido_em is null order by data`;
    let feitos = 0;
    for (let i = 0; i < pendentes.length; i += LOTE) {
      const lote = pendentes.slice(i, i + LOTE);
      const corpo = await pedir(promptCategorizar(lote.map((t) => ({ data: t.data, valor: Number(t.valor), desc: semDocumentos(t.desc_origem) }))), modelo);
      const u = corpo.usage ?? {};
      await sql`insert into ai_uso (user_id, kind, modelo, tokens_in, tokens_out, custo_usd) values (${user_id}, 'categorizar', ${modelo}, ${u.prompt_tokens ?? 0}, ${u.completion_tokens ?? 0}, ${typeof u.cost === 'number' ? u.cost.toFixed(6) : null})`;
      let dados: unknown = null;
      try {
        const texto = corpo.choices?.[0]?.message?.content ?? '';
        dados = JSON.parse(texto.slice(texto.indexOf('{'), texto.lastIndexOf('}') + 1));
      } catch {
        console.error(`lote ${i / LOTE + 1}: resposta não é JSON, pulando`);
        continue;
      }
      const lidos = lerCategorias(dados, lote.length);
      for (let j = 0; j < lote.length; j++) {
        const nome = lidos[j]!.nome;
        if (!nome) continue;
        await sql`update txn set desc_limpa = ${nome} where id = ${lote[j]!.id} and user_id = ${user_id} and desc_limpa is null`;
        feitos++;
      }
    }
    console.log(`usuário ${String(user_id).slice(0, 8)}…: ${feitos} de ${pendentes.length} com nome limpo`);
  }
}

main().catch((e) => {
  console.error(e instanceof Error ? e.message : e);
  process.exitCode = 1;
});
```
