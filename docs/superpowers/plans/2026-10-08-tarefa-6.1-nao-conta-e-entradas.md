# 6.1 — "Não conta" e Entradas: plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** o dono corrige o papel dos lançamentos da conta corrente: em Lançamentos, um alternador **Gastos · Não conta (N)** e a ação "não conta"; numa tela nova, **Entradas**, marca o que é renda. As duas marcações aprendem pela origem.

**Architecture:** a regra pura (que papel cabe em que lançamento, que regra grava, como agrupar as entradas, que aviso mostrar) fica em `src/lib/papel.ts`, testada sem banco. Uma função de repositório, `mudarPapel`, grava a marcação manual com a posse e o tipo de conta no `where`, ensina a regra (`learned_label.papel`) e a leva às linhas automáticas da mesma origem. Uma rota só, `PATCH /api/txn/papel`, serve à linha e ao lote, nas duas telas. O sync passa a aplicar também a regra `fora`.

**Tech Stack:** Next.js 15 (App Router, server components + client components), Drizzle 0.45 + Neon, TypeScript, testes sem framework em `scripts/testes.ts`.

Spec: `docs/superpowers/specs/2026-10-07-fatia-6-conta-corrente-design.md`, seção 2 (Telas e API). **Desvio da spec:** em vez de `PATCH /api/txn/[id]` aceitar `papel`, uma rota própria `PATCH /api/txn/papel` com `ids` (1 a 500) serve à linha e ao lote, para a validação e o aprendizado não existirem em dois lugares. /**
```ts
 * O papel de um lançamento da conta corrente quando o dono o corrige (6.1).
 * Puro, sem `server-only`: o teste importa direto.
 */
import type { Papel } from './pluggy/mapear';

/**
 * Débito é dinheiro que sai: gasto ou não conta. Crédito é dinheiro que entra:
 * a marcar, renda ou não conta. Renda num débito somaria saída como entrada.
 */
export function papelAceito(valor: number, papel: Papel): boolean {
  if (papel === 'fora') return true;
  if (valor > 0) return papel === 'gasto';
  if (valor < 0) return papel === 'entrada' || papel === 'renda';
  return false;
}

/** O que a marcação ensina à origem: renda e "não conta" viram regra; voltar apaga a regra. */
export function regraDoPapel(papel: Papel): 'renda' | 'fora' | null {
  return papel === 'renda' || papel === 'fora' ? papel : null;
}

/**
 * De que papel saem as outras linhas automáticas da mesma origem quando o dono
 * marca uma. Voltar a gasto não mexe nelas: o `fora` automático pode ter vindo
 * do mapeamento (pagamento de fatura, aplicação), não de uma regra.
 */
export function origensDoPapel(papel: Papel): Papel[] {
  if (papel === 'renda') return ['entrada'];
  if (papel === 'entrada') return ['renda'];
  if (papel === 'fora') return ['gasto', 'entrada'];
  return [];
}

export type LinhaEntrada = {
  id: string;
  data: string;
  valor: string;
  nome: string;
  conta: string;
  papel: Papel;
  pessoaFisica: boolean;
};

/** Os três blocos da tela Entradas, cada um do maior valor para o menor. */
export function blocosDeEntradas(linhas: readonly LinhaEntrada[]) {
  const porValor = (a: LinhaEntrada, b: LinhaEntrada) => Math.abs(Number(b.valor)) - Math.abs(Number(a.valor));
  const de = (p: Papel) => linhas.filter((l) => l.papel === p).sort(porValor);
  const renda = de('renda');
  return {
    renda,
    aMarcar: de('entrada'),
    fora: de('fora'),
    // crédito é gravado negativo; a renda aparece positiva
    totalRenda: Math.round(-renda.reduce((s, l) => s + Number(l.valor), 0) * 100) / 100,
  };
}

/**
 * O aviso depois de marcar: diz o que mudou junto e o que acontece com os
 * próximos. Sem aprender (Pix para pessoa física), não promete os próximos.
 */
export function textoDoPapel(papel: Papel, outros: number, aprendeu = true): string {
  if (!aprendeu) {
    if (papel === 'renda') return 'Marcado como renda.';
    if (papel === 'fora') return 'Fora dos totais.';
    if (papel === 'entrada') return 'Desmarcado.';
    return 'Volta a contar como gasto.';
  }
  const junto = outros > 0 ? `, com mais ${outros} da mesma origem` : '';
  if (papel === 'renda') return outros > 0 ? `Marcado como renda${junto}. Os próximos já chegam marcados.` : 'Marcado como renda. Os próximos da mesma origem já chegam marcados.';
  if (papel === 'fora') return `Fora dos totais${junto}. Os próximos também ficam fora.`;
  if (papel === 'entrada') return `Desmarcado${junto}. Os próximos chegam para você marcar.`;
  return 'Volta a contar como gasto.';
}
```

```ts
/**
 * A marcação do dono (6.1): grava `papel` como manual — nenhum sync a desfaz —,
 * ensina a regra da origem e a leva às linhas automáticas da mesma origem.
 * Só conta corrente, e só o papel que cabe no sinal: o resto dos `ids` é pulado.
 * Pix para pessoa física é marcado mas não ensina nada (`podeAprender`).
 */
export async function mudarPapel(userId: string, ids: string[], papel: Papel) {
  if (ids.length === 0) return { alterados: 0, outros: 0, aprendeu: false };
  const linhas = await db
    .update(txn)
    .set({ papel, papelFonte: 'manual' })
    .where(
      and(
        inArray(txn.id, ids),
        eq(txn.userId, userId),
        isNull(txn.excluidoEm),
        inArray(txn.accountId, contasCorrentes(userId)),
        sinalDoPapel(papel),
      ),
    )
    .returning({ id: txn.id, descNorm: txn.descNorm, pessoaFisica: txn.pessoaFisica });

  const chaves = new Set(linhas.filter(podeAprender).map((l) => chaveAprendizado(l.descNorm)).filter(Boolean));
  const regra = regraDoPapel(papel);
  for (const chave of chaves) await aprenderPapel(userId, chave, regra);

  let outros = 0;
  const origens = origensDoPapel(papel);
  if (chaves.size && origens.length) {
    // a chave da regra (3 primeiras palavras) só existe em JS: escolhe as candidatas
    // no banco e confere a chave aqui. Linha de pessoa física não recebe regra de
    // outra origem que caiu na mesma chave genérica ("PIX ENVIADO ...")
    const candidatas = await db
      .select({ id: txn.id, descNorm: txn.descNorm, valor: txn.valor })
      .from(txn)
      .where(
        and(
          eq(txn.userId, userId),
          isNull(txn.excluidoEm),
          eq(txn.papelFonte, 'auto'),
          eq(txn.pessoaFisica, false),
          inArray(txn.papel, origens),
          inArray(txn.accountId, contasCorrentes(userId)),
          notInArray(txn.id, ids),
        ),
      );
    const alvo = candidatas
      .filter((c) => chaves.has(chaveAprendizado(c.descNorm)) && papelAceito(Number(c.valor), papel))
      .map((c) => c.id);
    if (alvo.length) {
      // continuam 'auto': vieram da regra, e desfazer a regra as devolve
      const feitas = await db
        .update(txn)
        .set({ papel })
        .where(and(inArray(txn.id, alvo), eq(txn.userId, userId), eq(txn.papelFonte, 'auto')))
        .returning({ id: txn.id });
      outros = feitas.length;
    }
  }
  return { alterados: linhas.length, outros, aprendeu: chaves.size > 0 };
}
```

```ts
/** O N do alternador "Não conta (N)": débitos fora dos totais no recorte. */
export async function contarNaoConta(userId: string, f: { mes?: string | null; accountIds?: string[] }) {
  const cond = [eq(txn.userId, userId), isNull(txn.excluidoEm), eq(txn.suspeitoDuplicado, false), eq(txn.papel, 'fora'), gt(txn.valor, '0')];
  if (f.mes) cond.push(sql`${mesDaTxn} = ${f.mes}`);
  if (f.accountIds?.length) cond.push(inArray(txn.accountId, f.accountIds));
  const [linha] = await db
    .select({ n: sql<number>`count(*)::int` })
    .from(txn)
    .innerJoin(account, and(eq(account.id, txn.accountId), eq(account.userId, userId)))
    .leftJoin(statement, and(eq(statement.id, txn.statementId), eq(statement.accountId, txn.accountId)))
    .where(and(...cond));
  return linha?.n ?? 0;
}

/** Os créditos da conta corrente no recorte, para a tela Entradas. */
export async function listarEntradas(userId: string, f: { mes?: string | null; accountIds?: string[] }): Promise<LinhaEntrada[]> {
  const cond = [eq(txn.userId, userId), isNull(txn.excluidoEm), eq(txn.suspeitoDuplicado, false), lt(txn.valor, '0'), inArray(txn.papel, ['entrada', 'renda', 'fora'])];
  if (f.mes) cond.push(sql`${mesDaTxn} = ${f.mes}`);
  if (f.accountIds?.length) cond.push(inArray(txn.accountId, f.accountIds));
  const linhas = await db
    .select({
      id: txn.id,
      data: txn.data,
      valor: txn.valor,
      nome: nomeNaTelaSql,
      contaApelido: account.apelido,
      contaLast4: account.last4,
      contaBanco: rotuloDaConta,
      papel: txn.papel,
      pessoaFisica: txn.pessoaFisica,
    })
    .from(txn)
    .innerJoin(account, and(eq(account.id, txn.accountId), eq(account.userId, userId), eq(account.kind, 'conta_corrente')))
    .innerJoin(institution, eq(institution.id, account.institutionId))
    .leftJoin(statement, and(eq(statement.id, txn.statementId), eq(statement.accountId, txn.accountId)))
    .where(and(...cond))
    .orderBy(asc(txn.data), asc(txn.id));
  return linhas.map((l) => ({
    id: l.id,
    data: l.data,
    valor: l.valor,
    nome: l.nome,
    conta: l.contaApelido || `${l.contaBanco}${l.contaLast4 ? ` ••${l.contaLast4}` : ''}`,
    papel: l.papel,
    pessoaFisica: l.pessoaFisica,
  }));
}

export async function temContaCorrente(userId: string): Promise<boolean> {
  const [c] = await db
    .select({ id: account.id })
    .from(account)
    .where(and(eq(account.userId, userId), eq(account.kind, 'conta_corrente'), eq(account.ativo, true)))
    .limit(1);
  return Boolean(c);
}
```

```ts
import type { NextRequest } from 'next/server';
import { ehUuid } from '@/lib/filtro';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';
import { mudarPapel } from '@/lib/txn-repo';
import type { Papel } from '@/lib/pluggy/mapear';

export const runtime = 'nodejs';

const MAX = 500;
const PAPEIS: readonly Papel[] = ['gasto', 'entrada', 'renda', 'fora'];

/**
 * A marcação do dono na linha ou no lote, em Lançamentos e em Entradas. O que
 * não é conta corrente, não é do usuário ou não cabe no sinal é pulado no
 * `where`; se nada coube, a mesma resposta para todos os casos — não diz se o
 * id existe.
 */
export async function PATCH(req: NextRequest) {
  return comUsuario(async (userId) => {
    const bruto = await corpoJson<unknown>(req);
    const corpo = (bruto && typeof bruto === 'object' ? bruto : {}) as { ids?: unknown; papel?: unknown };
    const ids = Array.isArray(corpo.ids)
      ? [...new Set(corpo.ids.filter((x): x is string => typeof x === 'string' && ehUuid(x)))]
      : [];
    if (ids.length === 0) throw new ErroDeUso('ids_vazio', 'nenhum lançamento selecionado');
    if (ids.length > MAX) throw new ErroDeUso('lote_grande', `no máximo ${MAX} de cada vez`);
    const papel = corpo.papel as Papel;
    if (!PAPEIS.includes(papel)) throw new ErroDeUso('papel_invalido', 'marcação desconhecida');

    const r = await mudarPapel(userId, ids, papel);
    if (r.alterados === 0) throw new ErroDeUso('papel_nao_coube', 'Nenhum desses lançamentos aceita essa marcação.');
    return { ok: true, alterados: r.alterados, outros: r.outros, aprendeu: r.aprendeu };
  });
}
```
