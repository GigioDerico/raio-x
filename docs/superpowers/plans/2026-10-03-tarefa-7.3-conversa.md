```ts
/**
 * Regras puras da conversa (7.3). Sem `server-only`: a rota, a página, o
 * componente e o teste importam daqui.
 */
export const LIMITE_PERGUNTA = 500;
export const LIMITE_RESPOSTA = 2000;

// as do app antigo (docs/legado/app-antigo.html:2847)
export const SUGESTOES = [
  'Onde eu mais gastei?',
  'O que dá para cortar?',
  'Quanto vou pagar de parcelas?',
  'Tem cobrança repetida ou suspeita?',
  'O que mudou vs o mês anterior?',
] as const;

export type FalaGuardada = { papel: 'pessoa' | 'ia'; texto: string; dadosHash: string };

export function validarPergunta(v: unknown): string | null {
  if (typeof v !== 'string') return null;
  const t = v.trim();
  return t && t.length <= LIMITE_PERGUNTA ? t : null;
}

/**
 * Onde entra "Os números mudaram depois daqui": logo depois da última resposta
 * dada sobre números diferentes dos atuais. Pergunta não conta — quem fala de
 * números é a resposta.
 */
export function divisoriaEm(falas: readonly FalaGuardada[], dadosHash: string): number | null {
  for (let i = falas.length - 1; i >= 0; i--) {
    const f = falas[i]!;
    if (f.papel === 'ia' && f.dadosHash !== dadosHash) return i + 1;
  }
  return null;
}
```

```ts
export function mensagensConversa(dados: string, diagnostico: string | null, historico: Fala[], pergunta: string): Mensagem[] {
```

```ts
/** Lê um corpo NDJSON até o fim; o decoder em stream não parte caractere acentuado entre pedaços. */
export async function consumirNdjson(corpo: ReadableStream<Uint8Array>, aoEvento: (e: EventoLido) => void): Promise<void> {
  const leitor = corpo.getReader();
  const dec = new TextDecoder();
  let buf = '';
  try {
    for (;;) {
      const { value, done } = await leitor.read();
      if (done) break;
      const lido = lerNdjson(buf + dec.decode(value, { stream: true }));
      buf = lido.resto;
      lido.eventos.forEach(aoEvento);
    }
    lerNdjson(buf + dec.decode() + '\n').eventos.forEach(aoEvento);
  } finally {
    leitor.releaseLock();
  }
}
```

```ts
import 'server-only';
import type { Mensagem } from './ia';
import { linhaNdjson, type EventoDiagnostico } from './ndjson';
import { pedirTextoEmStream, type Uso } from './openrouter';

type Opcoes = {
  entrada: string | Mensagem[];
  modelo?: string;
  /** o `req.signal` da rota */
  sinal: AbortSignal;
  /** prefixo do log */
  rotulo: string;
  /** o que o cliente lê se a IA falhar depois de aberto o stream */
  mensagemErro: string;
  /**
   * Chamado uma vez, depois do [DONE], com o texto inteiro. Registra o custo e
   * grava; se lançar, o cliente recebe `mensagemErro`.
   */
  aoTerminar: (texto: string, modelo: string, uso: Uso) => Promise<void>;
};

/** Resposta NDJSON (`{t}`, `{fim}`, `{erro}`) de um texto da IA em stream. Parou ou deu erro: nada é gravado. */
export function respostaEmStream(o: Opcoes): Response {
  const enc = new TextEncoder();
  // req.signal nem sempre dispara quando o cliente some; o cancel() do stream cobre o resto
  const ac = new AbortController();
  const corpo = new ReadableStream<Uint8Array>({
    async start(controle) {
      const enviar = (e: EventoDiagnostico) => controle.enqueue(enc.encode(linhaNdjson(e)));
      let texto = '';
      try {
        for await (const ev of pedirTextoEmStream(o.entrada, { modelo: o.modelo, sinal: AbortSignal.any([o.sinal, ac.signal]) })) {
          if (ev.tipo === 'texto') {
            texto += ev.t;
            enviar({ t: ev.t });
          } else {
            await o.aoTerminar(texto, ev.modelo, ev.uso);
            enviar({ fim: true });
          }
        }
      } catch (e) {
        // interrompido pelo usuário: não há para quem responder
        if (!o.sinal.aborted && !ac.signal.aborted) {
          console.error(`[${o.rotulo}]`, e);
          try {
            enviar({ erro: o.mensagemErro });
          } catch {
            // o cliente pode ter fechado entre o erro e esta linha
          }
        }
      } finally {
        try {
          controle.close();
        } catch {
          // já fechado pelo cancelamento do cliente
        }
      }
    },
    cancel() {
      ac.abort();
    },
  });
  return new Response(corpo, {
    headers: { 'content-type': 'application/x-ndjson; charset=utf-8', 'cache-control': 'no-store' },
  });
}
```

```ts
import 'server-only';
import { and, desc, eq } from 'drizzle-orm';
import { db } from '@/db';
import { conversaTurno } from '@/db/schema';
import type { FalaGuardada } from './conversa';

/** As últimas `limite` falas do recorte, em ordem de conversa. */
export async function falasDoRecorte(userId: string, escopo: string, limite = 50): Promise<FalaGuardada[]> {
  const linhas = await db
    .select({ papel: conversaTurno.papel, texto: conversaTurno.texto, dadosHash: conversaTurno.dadosHash })
    .from(conversaTurno)
    .where(and(eq(conversaTurno.userId, userId), eq(conversaTurno.escopo, escopo)))
    .orderBy(desc(conversaTurno.criadoEm), desc(conversaTurno.id))
    .limit(limite);
  // o check do banco só aceita os dois papéis: o cast não esconde outro valor
  return linhas.reverse().map((l) => ({ ...l, papel: l.papel as FalaGuardada['papel'] }));
}

/**
 * Pergunta e resposta num insert só: atômico, e o id (identity, na ordem do
 * VALUES) desempata o criado_em igual.
 */
export async function gravarPar(userId: string, escopo: string, dadosHash: string, pergunta: string, resposta: string) {
  await db.insert(conversaTurno).values([
    { userId, escopo, papel: 'pessoa', texto: pergunta, dadosHash },
    { userId, escopo, papel: 'ia', texto: resposta, dadosHash },
  ]);
}

export async function limparConversa(userId: string, escopo: string) {
  await db.delete(conversaTurno).where(and(eq(conversaTurno.userId, userId), eq(conversaTurno.escopo, escopo)));
}
```

```ts
import { registrarUso } from '@/lib/ia-uso-repo';
import { carregarContexto } from '@/lib/painel-repo';
import { planoAtivo } from '@/lib/plano-repo';
import { hashDoPrompt, saidaGuardada } from '@/lib/resumo-repo';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';
import { configuracoes } from '@/lib/sessao-servidor';
import { respostaEmStream } from '@/lib/stream-rota';

export const runtime = 'nodejs';
export const maxDuration = 60;

type Preparo = { userId: string; entrada: Mensagem[]; escopo: string; dadosHash: string; pergunta: string; modelo?: string };

// o mês nulo do filtro quer dizer "o mais recente": a chave usa o mês resolvido
async function recorte(userId: string, filtro: Filtro) {
  const { ctx } = await carregarContexto(userId, filtro);
  const m = mesAtual(ctx);
  return { ctx, m, escopo: chaveRecorte(filtro, m) };
}

export async function POST(req: NextRequest) {
  const filtro = lerFiltro(req.nextUrl.searchParams);
  // objeto, não `let`: o TS não acompanha atribuição feita dentro do callback
  const p: { v?: Preparo } = {};

  const antes = await comUsuario(async (userId) => {
    const corpo = await corpoJson<unknown>(req);
    const pergunta = validarPergunta(corpo && typeof corpo === 'object' ? (corpo as { pergunta?: unknown }).pergunta : undefined);
    if (!pergunta) throw new ErroDeUso('pergunta_invalida', `Escreva uma pergunta de até ${LIMITE_PERGUNTA} caracteres`);

    const { ctx, m, escopo } = await recorte(userId, filtro);
    if (!lancsDoMes(ctx, m).length) throw new ErroDeUso('sem_lancamentos', 'Não há lançamentos neste recorte');
    if (!env.OPENROUTER_API_KEY) throw new ErroDeUso('sem_chave', 'Configure OPENROUTER_API_KEY');

    const plano = await planoAtivo(userId);
    const dados = dadosDiagnostico(ctx, filtro.cartoes.length > 0, plano?.resultado ?? null);
    const dadosHash = hashDoPrompt(dados);
    // só o diagnóstico dos mesmos números: um texto velho faria a IA defender números que mudaram
    const diag = await saidaGuardada(userId, 'diagnostico', escopo, hashDoPrompt(promptDiagnostico(dados)));
    const historico = await falasDoRecorte(userId, escopo, 6);
    const cfg = await configuracoes(userId);
    p.v = {
      userId,
      entrada: mensagensConversa(dados, diag?.atual ? diag.texto : null, historico, pergunta),
      escopo,
      dadosHash,
      pergunta,
      modelo: cfg?.modeloIa,
    };
    return null;
  });
  if (!p.v) return antes;

  const { userId, entrada, escopo, dadosHash, pergunta, modelo } = p.v;
  return respostaEmStream({
    entrada,
    modelo,
    sinal: req.signal,
    rotulo: 'conversa',
    mensagemErro: 'A IA não terminou a resposta; tente de novo',
    aoTerminar: async (texto, modeloUsado, uso) => {
      await registrarUso(userId, 'conversa', modeloUsado, uso);
      const resposta = lerTexto({ texto }, LIMITE_RESPOSTA);
      if (!resposta) throw new Error('resposta vazia da IA');
      await gravarPar(userId, escopo, dadosHash, pergunta, resposta);
    },
  });
}

export async function DELETE(req: NextRequest) {
  const filtro = lerFiltro(req.nextUrl.searchParams);
  return comUsuario(async (userId) => {
    const { escopo } = await recorte(userId, filtro);
    await limparConversa(userId, escopo);
    return { ok: true };
  });
}
```

```tsx
'use client';

import { Fragment, useRef, useState, useTransition } from 'react';
import { useRouter } from 'next/navigation';
import TextoIA from './TextoIA';
import { LIMITE_PERGUNTA, SUGESTOES, type FalaGuardada } from '@/lib/conversa';
import { consumirNdjson } from '@/lib/ndjson';

type Props = { falas: FalaGuardada[]; divisoria: number | null; query: string };

const FALHA = 'não consegui falar com a IA, tente de novo';

function Divisoria() {
  return (
    <li className="conv-divisoria" role="separator">
      Os números mudaram depois daqui.
    </li>
  );
}

export default function Conversa({ falas, divisoria, query }: Props) {
  const router = useRouter();
  const [texto, setTexto] = useState('');
  // a pergunta em andamento e a resposta crescendo; somem no refresh, que traz o par gravado
  const [enviada, setEnviada] = useState<string | null>(null);
  const [resposta, setResposta] = useState('');
  const [respondendo, setRespondendo] = useState(false);
  const [interrompida, setInterrompida] = useState(false);
  const [atualizando, startTransition] = useTransition();
  const [status, setStatus] = useState('');
  const [erro, setErro] = useState('');
  const controle = useRef<AbortController | null>(null);
  const dialogo = useRef<HTMLDialogElement>(null);

  const ocupado = respondendo || atualizando;

  async function perguntar(bruta: string) {
    const pergunta = bruta.trim();
    if (!pergunta || ocupado) return;
    const ac = new AbortController();
    controle.current = ac;
    setEnviada(pergunta);
    setResposta('');
    setTexto('');
    setErro('');
    setInterrompida(false);
    setRespondendo(true);
    setStatus('A IA está respondendo…');
    // objeto: o TS não acompanha o que o callback do stream muda
    const st = { fim: false, falhou: false, texto: '' };
    try {
      const r = await fetch(`/api/conversa?${query}`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ pergunta }),
        signal: ac.signal,
      });
      if (!r.ok || !r.body) {
        const d = await r.json().catch(() => ({}));
        st.falhou = true;
        // 400 traz mensagem escrita para o usuário ("configure a chave", "pergunta longa")
        setErro(r.status === 400 ? d?.detalhe || 'não deu certo' : FALHA);
        return;
      }
      await consumirNdjson(r.body, (e) => {
        if (typeof e.t === 'string') {
          st.texto += e.t;
          setResposta(st.texto);
        } else if (e.fim === true) st.fim = true;
        else if (typeof e.erro === 'string') {
          st.falhou = true;
          setErro(e.erro);
        }
      });
      if (!st.fim && !st.falhou) {
        st.falhou = true;
        setErro('a resposta chegou pela metade; tente de novo');
      }
    } catch {
      if (ac.signal.aborted) setInterrompida(true);
      else {
        st.falhou = true;
        setErro(FALHA);
      }
    } finally {
      setRespondendo(false);
      controle.current = null;
      if (st.fim) {
        setStatus('Resposta pronta.');
        startTransition(() => router.refresh());
      } else {
        // nada foi gravado: a pergunta volta para o campo para tentar de novo
        setTexto(pergunta);
        setStatus(ac.signal.aborted ? 'Interrompido. Nada foi guardado.' : '');
        if (st.falhou) {
          setEnviada(null);
          setResposta('');
        }
      }
    }
  }

  async function limpar() {
    setErro('');
    try {
      const r = await fetch(`/api/conversa?${query}`, { method: 'DELETE' });
      if (!r.ok) {
        setErro('não consegui limpar, tente de novo');
        return;
      }
      dialogo.current?.close();
      setStatus('Conversa limpa.');
      startTransition(() => router.refresh());
    } catch {
      setErro('não consegui limpar, tente de novo');
    }
  }

  const vazia = falas.length === 0 && !enviada;

  return (
    <section className="cartao conversa" aria-labelledby="conv-titulo">
      <div className="conv-topo">
        <h2 id="conv-titulo">Conversa</h2>
        {falas.length > 0 && (
          <button type="button" className="link" onClick={() => dialogo.current?.showModal()} disabled={ocupado}>
            Limpar conversa
          </button>
        )}
      </div>

      {vazia ? (
        <p className="suave">Pergunte o que quiser sobre o recorte que está na tela.</p>
      ) : (
        <ol className="conv-falas">
          {falas.map((f, i) => (
            <Fragment key={i}>
              {divisoria === i && <Divisoria />}
              <li className={`bolha ${f.papel}`}>
                {f.papel === 'pessoa' ? <p>{f.texto}</p> : <TextoIA texto={f.texto} />}
              </li>
            </Fragment>
          ))}
          {divisoria !== null && divisoria === falas.length && <Divisoria />}
          {enviada && (
            <>
              <li className="bolha pessoa">
                <p>{enviada}</p>
              </li>
              <li className="bolha ia">
                {resposta ? <TextoIA texto={resposta} /> : <p className="suave">Pensando…</p>}
                {interrompida && <p className="conv-nota">Interrompida, não guardada.</p>}
              </li>
            </>
          )}
        </ol>
      )}

      {vazia && (
        <div className="conv-sugestoes">
          {SUGESTOES.map((s) => (
            <button key={s} type="button" className="conv-sugestao" onClick={() => perguntar(s)} disabled={ocupado}>
              {s}
            </button>
          ))}
        </div>
      )}

      <form
        className="conv-form"
        onSubmit={(e) => {
          e.preventDefault();
          perguntar(texto);
        }}
      >
        <textarea
          aria-label="Sua pergunta"
          value={texto}
          maxLength={LIMITE_PERGUNTA}
          rows={2}
          placeholder="Ex.: por que restaurantes subiu tanto?"
          onChange={(e) => setTexto(e.target.value)}
          onKeyDown={(e) => {
            // Enter envia; Shift+Enter quebra linha; composição de acento (IME) não envia
            if (e.key === 'Enter' && !e.shiftKey && !e.nativeEvent.isComposing) {
              e.preventDefault();
              perguntar(texto);
            }
          }}
          disabled={respondendo}
        />
        <div className="conv-form-rodape">
          <span className="conv-contador" aria-hidden="true">
            {texto.length}/{LIMITE_PERGUNTA}
          </span>
          {respondendo ? (
            <button type="button" className="link" onClick={() => controle.current?.abort()}>
              Parar
            </button>
          ) : (
            <button type="submit" disabled={!texto.trim() || ocupado}>
              Enviar
            </button>
          )}
        </div>
      </form>

      <span className={erro ? 'conv-status conv-erro' : 'conv-status'} role="status" aria-live="polite">
        {erro || status}
      </span>

      <dialog ref={dialogo} className="dialogo-apagar" aria-labelledby="limpar-titulo">
        <h2 id="limpar-titulo">Limpar a conversa deste recorte?</h2>
        <p>As perguntas e respostas somem. O diagnóstico continua.</p>
        <div className="dialogo-acoes">
          <button type="button" className="link" onClick={() => dialogo.current?.close()}>
            Cancelar
          </button>
          <button type="button" className="perigo" onClick={limpar} disabled={ocupado}>
            Limpar
          </button>
        </div>
      </dialog>
    </section>
  );
}
```
