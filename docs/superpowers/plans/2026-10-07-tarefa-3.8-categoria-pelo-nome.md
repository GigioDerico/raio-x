/**
 * O nome que a lista mostra e a chave com que ele é comparado. Puro: o teste
 * importa, a tela também. A chave tem espelho em SQL (`chaveDoNomeSql`, em
 * txn-repo.ts) — mudou aqui, muda lá.

type ComNome = { apelido: string | null; descLimpa: string | null; descOrigem: string };

// `||` e não `??`: apelido apagado vira string vazia e tem de passar adiante
export const nomeNaTela = (l: ComNome): string => l.apelido || l.descLimpa || l.descOrigem;

export const chaveDoNome = (s: string): string => s.trim().replace(/\s+/g, ' ').toLowerCase();

export function textoMudouJunto(n: number, nome: string | null, categoria: string): string | null {
  if (n <= 0) return null;
  const qtd = `${n} ${n === 1 ? 'lançamento' : 'lançamentos'}`;
  return nome ? `Também mudei ${qtd} de “${nome}” para ${categoria}.` : `Também mudei ${qtd} com os mesmos nomes para ${categoria}.`;
}
```

/**
 * Leva a categoria a todos os lançamentos com o mesmo nome na tela de algum dos
 * `ids` (pedido do dono, 3.8): trocar "Alane Ferreira" num mês troca nos outros.
 * Pagamento de fatura e excluído ficam fora. Devolve o update para ir no mesmo
 * batch do alterado — ou muda tudo, ou nada.
 */
function mesmaCategoriaPeloNome(userId: string, ids: string[], categoryId: number | null) {
  const nomes = db
    .select({ chave: chaveDoNomeSql })
    .from(txn)
    .where(and(inArray(txn.id, ids), eq(txn.userId, userId)));
  return db
    .update(txn)
    .set({ categoryId, categoryFonte: 'manual' })
    .where(
      and(
        eq(txn.userId, userId),
        isNull(txn.excluidoEm),
        ne(txn.tipo, 'pagamento'),
        notInArray(txn.id, ids),
        sql`${chaveDoNomeSql} in ${nomes}`,
      ),
    )
    .returning({ id: txn.id, descNorm: txn.descNorm });
}
```

(importar `notInArray` e `ne` de `drizzle-orm` se faltarem). Se o Drizzle não
aceitar subquery em `sql\`... in ${nomes}\``, use `inArray(chaveDoNomeSql,
nomes)` ou escreva a subquery em `sql` com `${txn.id} = any(${ids})` — e relate.
