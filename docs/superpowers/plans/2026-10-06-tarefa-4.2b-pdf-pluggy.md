```ts
import { ehUuid } from './filtro';

/**
 * Possíveis duplicados: compra do Pluggy que pode ser a mesma de um PDF já
 * importado, e o planejador não teve como decidir sozinho. Puro: rota, tela e
 * teste usam daqui.
 */
export type LinhaDup = { id: string; accountId: string; data: string; valor: number; fonte: 'pdf' | 'pluggy' | 'manual'; excluido: boolean };

const DIA_MS = 86_400_000;
const distancia = (a: string, b: string) => Math.abs(Date.parse(`${a}T00:00:00Z`) - Date.parse(`${b}T00:00:00Z`)) / DIA_MS;
const centavos = (v: number) => Math.round(Math.abs(v) * 100);

/** O mesmo passe aproximado do planejador (4.0): mesma conta, mesmo valor, até 3 dias, só PDF não excluído. */
export function candidatosDoSuspeito(suspeito: LinhaDup, outras: readonly LinhaDup[]): LinhaDup[] {
  return outras
    .filter(
      (o) =>
        o.id !== suspeito.id &&
        o.accountId === suspeito.accountId &&
        o.fonte === 'pdf' &&
        !o.excluido &&
        centavos(o.valor) === centavos(suspeito.valor) &&
        distancia(o.data, suspeito.data) <= 3,
    )
    .sort((a, b) => distancia(a.data, suspeito.data) - distancia(b.data, suspeito.data));
}

export type Decisao = { decisao: 'mesmo'; candidatoId: string } | { decisao: 'diferente' } | { decisao: 'desfazer'; candidatoId: string };

export function validarDecisao(v: unknown): { ok: true; decisao: Decisao } | { ok: false } {
  const o = v as { decisao?: unknown; candidatoId?: unknown } | null;
  if (!o || typeof o !== 'object') return { ok: false };
  if (o.decisao === 'diferente') return { ok: true, decisao: { decisao: 'diferente' } };
  if ((o.decisao === 'mesmo' || o.decisao === 'desfazer') && typeof o.candidatoId === 'string' && ehUuid(o.candidatoId)) {
    return { ok: true, decisao: { decisao: o.decisao, candidatoId: o.candidatoId } };
  }
  return { ok: false };
}
```

```ts
/**
 * A decisão sobre um suspeito, numa transação. O candidato é relido e
 * revalidado aqui — nunca vale só porque veio da tela.
 */
export async function decidirDuplicado(userId: string, id: string, d: Decisao): Promise<boolean> {
  const [s] = await db
    .select({ id: txn.id, accountId: txn.accountId, data: txn.data, valor: txn.valor, apelido: txn.apelido, suspeito: txn.suspeitoDuplicado })
    .from(txn)
    .where(and(eq(txn.id, id), eq(txn.userId, userId), eq(txn.fonte, 'pluggy')))
    .limit(1);
  if (!s) return false;

  if (d.decisao === 'diferente') {
    if (!s.suspeito) return false;
    await db.update(txn).set({ suspeitoDuplicado: false }).where(and(eq(txn.id, s.id), eq(txn.userId, userId)));
    return true;
  }

  const [c] = await db
    .select({ id: txn.id, accountId: txn.accountId, data: txn.data, valor: txn.valor, apelido: txn.apelido, categoryId: txn.categoryId, categoryFonte: txn.categoryFonte, excluido: txn.excluidoEm })
    .from(txn)
    .where(and(eq(txn.id, d.candidatoId), eq(txn.userId, userId), eq(txn.fonte, 'pdf')))
    .limit(1);
  if (!c) return false;

  if (d.decisao === 'desfazer') {
    // só desfaz o que o "é este" fez: candidato excluído e suspeito já liberado
    if (!c.excluido || s.suspeito) return false;
    await db.batch([
      db.update(txn).set({ excluidoEm: null }).where(and(eq(txn.id, c.id), eq(txn.userId, userId))),
      db.update(txn).set({ suspeitoDuplicado: true }).where(and(eq(txn.id, s.id), eq(txn.userId, userId))),
    ]);
    return true;
  }

  // "é este": o candidato precisa ser candidato de verdade
  const valido = candidatosDoSuspeito(
    { id: s.id, accountId: s.accountId, data: s.data, valor: Number(s.valor), fonte: 'pluggy', excluido: false },
    [{ id: c.id, accountId: c.accountId, data: c.data, valor: Number(c.valor), fonte: 'pdf', excluido: Boolean(c.excluido) }],
  ).length === 1;
  if (!s.suspeito || !valido) return false;
  await db.batch([
    db
      .update(txn)
      .set({
        suspeitoDuplicado: false,
        apelido: s.apelido ?? c.apelido,
        // o que você corrigiu à mão no PDF vale também para a compra do banco (regra 7)
        ...(c.categoryFonte === 'manual' ? { categoryId: c.categoryId, categoryFonte: 'manual' } : {}),
      })
      .where(and(eq(txn.id, s.id), eq(txn.userId, userId))),
    db.update(txn).set({ excluidoEm: sql`now()` }).where(and(eq(txn.id, c.id), eq(txn.userId, userId))),
  ]);
  return true;
}
```

```ts
import type { NextRequest } from 'next/server';
import { validarDecisao } from '@/lib/duplicados';
import { ehUuid } from '@/lib/filtro';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';
import { decidirDuplicado } from '@/lib/txn-repo';

export const runtime = 'nodejs';

const naoEncontrado = () => new ErroDeUso('nao_encontrado', 'Lançamento não encontrado', 404);

export async function POST(req: NextRequest, ctx: { params: Promise<{ id: string }> }) {
  const { id } = await ctx.params;
  return comUsuario(async (userId) => {
    if (!ehUuid(id)) throw naoEncontrado();
    const v = validarDecisao(await corpoJson<unknown>(req));
    if (!v.ok) throw new ErroDeUso('decisao_invalida', 'Decisão inválida');
    if (!(await decidirDuplicado(userId, id, v.decisao))) throw naoEncontrado();
    return { ok: true };
  });
}
```

```tsx
/** Faixa e revisão dos possíveis duplicados, dentro de Lançamentos. */

            {aberto ? 'Fechar' : 'Revisar'}

            type="button"
            className="link"

        </div>
      )}
      {erro && <AvisoDeFalha falha={erro} />}
      {aberto && (
        <ul className="duplicados-lista">
          {suspeitos.map((s) => (
            <li key={s.id} className="cartao duplicado">
              <p>
                <strong>{s.nome}</strong> · {dia(s.data)} · {fmt.format(s.valor)} <span className="suave">· {s.cartao} · do banco</span>
              </p>
              {s.candidatos.length > 0 ? (
                <ul>
                  {s.candidatos.map((c) => (
                    <li key={c.id}>
                      <span>
                        {c.nome} · {dia(c.data)} · {fmt.format(c.valor)} <span className="suave">· do PDF</span>
                      </span>
                      <button
                        type="button"
                        disabled={ocupado}
                        onClick={async () => {
                          if (await decidir(s.id, { decisao: 'mesmo', candidatoId: c.id })) setDesfazer({ id: s.id, candidatoId: c.id, nome: c.nome });
                        }}
                      >
                        É este
                      </button>
                    </li>
                  ))}
                </ul>
              ) : (
                <p className="suave">O lançamento do PDF parecido não existe mais.</p>
              )}
              <button type="button" className="link" disabled={ocupado} onClick={() => void decidir(s.id, { decisao: 'diferente' })}>
                São diferentes
              </button>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
```

O "desfazer" some depois de 10 s, como na exclusão: `useEffect` com `setTimeout(() => setDesfazer(null), 10000)` dependendo de `desfazer`.
