**Goal:** Ao conectar um banco pelo widget, o app confere a posse do item no Pluggy, grava instituição, conexão e contas, e traz 3 meses de lançamentos de cada cartão, categorizados por regras aprendidas e IA.

Repositórios `conexao-repo.ts` e `categorizar-repo.ts` (`server-only`). Duas rotas: `POST /api/connect/callback` e `POST /api/accounts/[id]/sync` (uma por cartão, para caber no limite da Vercel). `ConectarBanco.tsx` orquestra no cliente.

**Tech Stack:** Next.js 15, TypeScript, Drizzle sobre Neon HTTP, API do Pluggy, OpenRouter.

export function normalizarNomeBanco(nome: string): string {
  return nome
    .normalize('NFD')
    .replace(/[̀-ͯ]/g, '')
    .toLowerCase()
    .replace(/\bs\.?\s*a\.?(?=\s|$)/g, ' ')
    .replace(/\bbanco\b/g, ' ')
    .replace(/[^a-z0-9 ]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

```ts
import 'server-only';
import { and, asc, desc, eq } from 'drizzle-orm';
import { db } from '@/db';
import { account, connection, institution } from '@/db/schema';
import { casarConta, contaDoPluggy, escolherInstituicao, type ContaPluggy, type ItemPluggy, type StatusConexao } from './mapear';

export type CartaoConectado = { id: string; nome: string };

/**
 * Grava instituição, conexão e contas de um item cuja posse já foi conferida
 * no Pluggy. Lê antes, decide com as funções puras e escreve numa transação.
 */
export async function gravarConexao(
  userId: string,
  item: ItemPluggy & { connector: NonNullable<ItemPluggy['connector']> },
  contas: ContaPluggy[],
  status: StatusConexao,
): Promise<{ conexaoId: string; cartoes: CartaoConectado[] }> {
  const insts = await db.select({ id: institution.id, nome: institution.nome, pluggyConnectorId: institution.pluggyConnectorId }).from(institution);
  const escolha = escolherInstituicao(insts, item.connector);
  let institutionId: number;
  if ('usar' in escolha) {
    institutionId = escolha.usar;
    // casou pelo nome: o conector passa a ser conhecido, sem apagar cor ou logo que já existam
    await db
      .update(institution)
      .set({ pluggyConnectorId: item.connector.id })
      .where(and(eq(institution.id, institutionId)));
  } else {
    const [nova] = await db
      .insert(institution)
      .values(escolha.criar)
      // outro usuário pode ter criado a mesma no meio-tempo: instituição é catálogo compartilhado
      .onConflictDoUpdate({ target: institution.slug, set: { nome: escolha.criar.nome } })
      .returning({ id: institution.id });
    institutionId = nova!.id;
  }

  const ultimoErro = status === 'erro' ? (item.error?.message ?? null) : null;
  const [con] = await db
    .insert(connection)
    .values({ userId, institutionId, pluggyItemId: item.id, status, ultimoErro })
    .onConflictDoUpdate({
      target: connection.pluggyItemId,
      set: { status, ultimoErro },
      // item de outro usuário nunca é tocado (a posse já foi conferida, isto é a segunda trava)
      setWhere: eq(connection.userId, userId),
    })
    .returning({ id: connection.id });
  if (!con) throw new Error('conexao_de_outro_usuario');

  const existentes = await db
    .select({ id: account.id, kind: account.kind, last4: account.last4, pluggyAccountId: account.pluggyAccountId })
    .from(account)
    .where(eq(account.userId, userId));

  const comandos = [];
  const cartoes: { id: string | null; pluggyAccountId: string; last4: string | null }[] = [];
  for (const bruta of contas) {
    const c = contaDoPluggy(bruta);
    const decisao = casarConta(existentes, c);
    const limite = c.limite === null ? null : c.limite.toFixed(2);
    // saldo só de conta corrente/poupança, para a fatia 6; do cartão o Pluggy manda a fatura em aberto, não saldo
    const saldo = c.kind !== 'cartao_credito' && typeof bruta.balance === 'number' ? { saldo: bruta.balance.toFixed(2), saldoEm: new Date() } : {};
    if ('usar' in decisao) {
      comandos.push(
        db
          .update(account)
          .set({ pluggyAccountId: c.pluggyAccountId, connectionId: con.id, limite, ...saldo })
          .where(and(eq(account.id, decisao.usar), eq(account.userId, userId))),
      );
      if (c.kind === 'cartao_credito') cartoes.push({ id: decisao.usar, pluggyAccountId: c.pluggyAccountId, last4: c.last4 });
    } else {
      comandos.push(
        db.insert(account).values({
          userId,
          connectionId: con.id,
          institutionId,
          kind: c.kind,
          pluggyAccountId: c.pluggyAccountId,
          last4: c.last4,
          nomeOrigem: c.nomeOrigem,
          limite,
          ...saldo,
        }),
      );
      if (c.kind === 'cartao_credito') cartoes.push({ id: null, pluggyAccountId: c.pluggyAccountId, last4: c.last4 });
    }
  }
  if (comandos.length) await db.batch(comandos as unknown as Parameters<typeof db.batch>[0]);

  // ids das contas recém-criadas e o rótulo do filtro (apelido, ou banco com ••last4)
  const gravados = await db
    .select({ id: account.id, apelido: account.apelido, last4: account.last4, banco: institution.nome, pluggyAccountId: account.pluggyAccountId })
    .from(account)
    .innerJoin(institution, eq(institution.id, account.institutionId))
    .where(and(eq(account.userId, userId), eq(account.connectionId, con.id), eq(account.kind, 'cartao_credito')));
  return {
    conexaoId: con.id,
    cartoes: gravados.map((g) => ({ id: g.id, nome: (g.apelido || g.banco) + (g.last4 ? ` ••${g.last4}` : '') })),
  };
}

export type ConexaoListada = { id: string; banco: string; logoUrl: string | null; status: StatusConexao; ultimoSyncEm: Date | null; ultimoErro: string | null };

export async function listarConexoes(userId: string): Promise<ConexaoListada[]> {
  return db
    .select({
      id: connection.id,
      banco: institution.nome,
      logoUrl: institution.logoUrl,
      status: connection.status,
      ultimoSyncEm: connection.ultimoSyncEm,
      ultimoErro: connection.ultimoErro,
    })
    .from(connection)
    .innerJoin(institution, eq(institution.id, connection.institutionId))
    .where(eq(connection.userId, userId))
    .orderBy(asc(institution.nome), desc(connection.criadoEm));
}

const naoEncontrado = () => new ErroDeUso('item_nao_encontrado', 'Conexão não encontrada no Pluggy', 404);

/**
 * O widget devolve o itemId ao navegador; quem garante que o item é desta
 * pessoa é o Pluggy (`clientUserId`, posto pelo connect token), não o corpo.
 * "Não existe" e "não é seu" respondem igual.

export async function POST(req: NextRequest) {
  return comUsuario(async (userId) => {
    const corpo = await corpoJson<unknown>(req);
    const itemId = (corpo as { itemId?: unknown } | null)?.itemId;
    if (typeof itemId !== 'string' || !itemId.trim() || itemId.length > 100) throw new ErroDeUso('item_invalido', 'itemId inválido');

    let item;
    try {
      item = await pluggy().item(itemId.trim());
    } catch (e) {
      if (e instanceof SemChavePluggy) throw new ErroDeUso('sem_chave_pluggy', 'Configure PLUGGY_CLIENT_ID e PLUGGY_CLIENT_SECRET no .env.local', 501);
      throw e;
    }
    if (!item || item.clientUserId !== userId || !item.connector) throw naoEncontrado();

    if (item.status === 'UPDATING') return { estado: 'atualizando' as const };
    if (item.status === 'LOGIN_ERROR' || item.status === 'WAITING_USER_INPUT') {
      throw new ErroDeUso('erro_login', 'O banco recusou o acesso. Conecte de novo e confira a senha.');
    }
    if (item.status !== 'UPDATED' && item.status !== 'OUTDATED') {
      throw new ErroDeUso('erro_pluggy', 'O Pluggy não conseguiu ler o banco. Tente de novo em instantes.', 502);
    }

    const contas = await pluggy().contasDoItem(item.id);
    const { conexaoId, cartoes } = await gravarConexao(userId, { ...item, connector: item.connector }, contas, item.status === 'UPDATED' ? 'ativa' : 'erro');
    return {
      estado: 'ok' as const,
      conexaoId,
      cartoes,
      aviso: item.status === 'OUTDATED' ? 'O banco não respondeu na última leitura; os dados podem estar atrasados.' : null,
    };
  });
}
```

import 'server-only';
import { and, eq, inArray, isNull } from 'drizzle-orm';
import { db } from '@/db';
import { txn } from '@/db/schema';
import { env } from '@/env';
import { lerCategorias, promptCategorizar } from '@/lib/ia';
import { registrarUso } from '@/lib/ia-uso-repo';
import { RespostaNaoJson, pedirJson } from '@/lib/openrouter';
import { idDaCategoria } from '@/lib/txn-repo';

const LOTE = 60;

/**
 * Pede à IA a categoria dos lançamentos da conta que chegaram sem regra
 * aprendida. Só `category_fonte = 'ia'` e sem categoria: o manual e o da regra
 * nunca são tocados (regra 7). Dado novo, então pode chamar o modelo (regra 9).
 */
export async function categorizarPendentes(userId: string, accountId: string, modelo?: string): Promise<{ categorizados: number; semCategoria: number }> {
  const pendentes = await db
    .select({ id: txn.id, data: txn.data, valor: txn.valor, descNorm: txn.descNorm })
    .from(txn)
    .where(and(eq(txn.userId, userId), eq(txn.accountId, accountId), isNull(txn.categoryId), eq(txn.categoryFonte, 'ia'), isNull(txn.excluidoEm)));
  if (!pendentes.length) return { categorizados: 0, semCategoria: 0 };
  if (!env.OPENROUTER_API_KEY) return { categorizados: 0, semCategoria: pendentes.length };

  let categorizados = 0;
  for (let i = 0; i < pendentes.length; i += LOTE) {
    const lote = pendentes.slice(i, i + LOTE);
    let dados: unknown = null;
    try {
      const r = await pedirJson<unknown>(promptCategorizar(lote.map((t) => ({ data: t.data, valor: Number(t.valor), desc: t.descNorm }))), { modelo });
      await registrarUso(userId, 'categorizar', r.modelo, r.uso);
      dados = r.dados;
    } catch (e) {
      // um lote ruim não derruba a carga: esses ficam sem categoria, para o dono corrigir
      if (e instanceof RespostaNaoJson && e.modelo && e.uso) await registrarUso(userId, 'categorizar', e.modelo, e.uso);
      else console.error('[categorizar]', e);
      continue;
    }
    const cats = lerCategorias(dados, lote.length);
    const porCategoria = new Map<string, string[]>();
    cats.forEach((c, j) => {
      if (c) porCategoria.set(c, [...(porCategoria.get(c) ?? []), lote[j]!.id]);
    });
    for (const [nome, ids] of porCategoria) {
      const categoryId = await idDaCategoria(nome);
      if (!categoryId) continue;
      await db
        .update(txn)
        .set({ categoryId })
        // de novo as travas: alguém pode ter categorizado à mão enquanto a IA respondia
        .where(and(inArray(txn.id, ids), eq(txn.userId, userId), isNull(txn.categoryId), eq(txn.categoryFonte, 'ia')));
      categorizados += ids.length;
    }
  }
  return { categorizados, semCategoria: pendentes.length - categorizados };
}
