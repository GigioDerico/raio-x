**Goal:** Cliente do Pluggy, mapeadores Pluggy → modelo, casamento de conta com a do PDF e deduplicação em três passos com promoção PDF → Pluggy, testados com respostas simuladas.

**Architecture:** Mapeadores e planejador de deduplicação são funções puras em `src/lib/pluggy/`. **Tech Stack:** TypeScript, Drizzle (Neon HTTP), `fetch` nativo; testes sem framework em `scripts/testes.ts`.

/**
 * Pluggy → modelo do app. Puro, sem `server-only`: o teste importa direto.
 *
 * Os tipos de entrada listam só o que é lido. O que o Pluggy entrega e o app
 * não pode guardar — nome e CPF do titular, saldo da conta corrente, dados do
 * pagador, CNPJ do estabelecimento — não aparece em nenhum tipo de saída
 * (regra 5 do CLAUDE.md).
 */
import { normalizarDescricao } from '../descricao';

export type ItemPluggy = { id: string; status: string };

export type ContaPluggy = {
  id: string;
  type: string;
  subtype?: string | null;
  number?: string | null;
  name?: string | null;
  marketingName?: string | null;
  balance?: number | null;
  owner?: string | null;
  taxNumber?: string | null;
  creditData?: { creditLimit?: number | null } | null;
};

export type TransacaoPluggy = {
  id?: string;
  description?: string | null;
  descriptionRaw?: string | null;
  amount?: number;
  date?: string;
  status?: string;
  type?: string;
  operationType?: string | null;
  creditCardMetadata?: {
    installmentNumber?: number | null;
    totalInstallments?: number | null;
    purchaseDate?: string | null;
    billId?: string | null;
  } | null;
  paymentData?: unknown;
  merchant?: unknown;
};

export type FaturaPluggy = { id: string; dueDate: string; totalAmount: number };

export type StatusConexao = 'ativa' | 'atualizando' | 'erro_login' | 'erro';
export type KindConta = 'cartao_credito' | 'conta_corrente' | 'conta_pagamento' | 'poupanca';

export type ContaMapeada = {
  pluggyAccountId: string;
  kind: KindConta;
  last4: string | null;
  nomeOrigem: string | null;
  limite: number | null;
};

export type Recebido = {
  pluggyTxnId: string;
  data: string;
  valor: number;
  descOrigem: string;
  descNorm: string;
  tipo: 'pagamento' | 'parcela' | 'escolha';
  parcela: { n: number; total: number } | null;
  billId: string | null;
  pendente: boolean;
};

export type FaturaMapeada = { pluggyBillId: string; mesRef: string; vencimento: string; total: number };

export type ContaExistente = { id: string; kind: string; last4: string | null; pluggyAccountId: string | null };

export function statusDoItem(status: string): StatusConexao {
  if (status === 'UPDATED') return 'ativa';
  if (status === 'UPDATING') return 'atualizando';
  if (status === 'LOGIN_ERROR' || status === 'WAITING_USER_INPUT') return 'erro_login';
  return 'erro';
}

export function contaDoPluggy(c: ContaPluggy): ContaMapeada {
  const kind: KindConta =
    c.type === 'CREDIT' || c.subtype === 'CREDIT_CARD'
      ? 'cartao_credito'
      : c.subtype === 'CHECKING_ACCOUNT'
        ? 'conta_corrente'
        : c.subtype === 'SAVINGS_ACCOUNT'
          ? 'poupanca'
          : 'conta_pagamento';
  const digitos = String(c.number ?? '').replace(/\D/g, '');
  return {
    pluggyAccountId: c.id,
    kind,
    last4: digitos ? digitos.slice(-4) : null,
    nomeOrigem: c.marketingName || c.name || null,
    limite: typeof c.creditData?.creditLimit === 'number' ? c.creditData.creditLimit : null,
  };
}

const diaSP = new Intl.DateTimeFormat('en-CA', {
  timeZone: 'America/Sao_Paulo',
  year: 'numeric',
  month: '2-digit',
  day: '2-digit',
});

// o Pluggy manda `date` em UTC; o mês da compra é decidido no fuso de quem gastou
function diaEmSaoPaulo(iso: string | undefined): string | null {
  if (!iso) return null;
  const d = new Date(iso);
  return Number.isNaN(d.getTime()) ? null : diaSP.format(d);
}

function diaValido(d: string): boolean {
  return /^\d{4}-\d{2}-\d{2}$/.test(d) && new Date(`${d}T00:00:00Z`).toISOString().slice(0, 10) === d;
}

export function lancamentoDoPluggy(t: TransacaoPluggy): Recebido | null {
  if (!t.id || typeof t.amount !== 'number' || !Number.isFinite(t.amount)) return null;
  const meta = t.creditCardMetadata ?? null;
  // o PDF registra a data da compra; usar a mesma é o que faz a chave natural colidir
  const data = meta?.purchaseDate ? meta.purchaseDate.slice(0, 10) : diaEmSaoPaulo(t.date);
  if (!data || !diaValido(data)) return null;
  const descOrigem = (t.descriptionRaw || t.description || '').trim();
  const parcela =
    meta?.installmentNumber && meta.totalInstallments && meta.totalInstallments > 1
      ? { n: meta.installmentNumber, total: meta.totalInstallments }
      : null;
  const pagamento = t.operationType === 'PAGAMENTO_FATURA' || t.description === 'PAGAMENTO_FATURA';
  return {
    pluggyTxnId: t.id,
    data,
    valor: Math.round(t.amount * 100) / 100,
    descOrigem,
    descNorm: normalizarDescricao(descOrigem),
    tipo: pagamento ? 'pagamento' : parcela ? 'parcela' : 'escolha',
    parcela,
    billId: meta?.billId ?? null,
    pendente: t.status === 'PENDING',
  };
}

export function faturaDaBill(b: FaturaPluggy): FaturaMapeada {
  // vencimento é data de calendário: o dia vem do texto, sem conversão de fuso
  const vencimento = b.dueDate.slice(0, 10);
  return { pluggyBillId: b.id, mesRef: vencimento.slice(0, 7), vencimento, total: b.totalAmount };
}

/**
 * A `dedupe_key` inclui o `account_id`: se o cartão do Pluggy virasse conta
 * nova, nenhuma compra colidiria com o PDF e tudo entraria em dobro.
 */
export function casarConta(existentes: readonly ContaExistente[], c: ContaMapeada): { usar: string } | { criar: true } {
  const jaLigada = existentes.find((e) => e.pluggyAccountId === c.pluggyAccountId);
  if (jaLigada) return { usar: jaLigada.id };
  if (c.kind !== 'cartao_credito' || !c.last4) return { criar: true };
  const candidatas = existentes.filter((e) => e.kind === 'cartao_credito' && e.last4 === c.last4 && !e.pluggyAccountId);
  return candidatas.length === 1 ? { usar: candidatas[0]!.id } : { criar: true };
}

/**
 * Decide o que fazer com cada lançamento que veio do Pluggy, nos três passos do
 * plano técnico (seção "Deduplicação"): id do Pluggy, chave natural, passe
 * aproximado. Puro: recebe o que já está gravado e devolve ações; quem grava é
 * sync-repo.ts. Assim a regra inteira é testável sem banco.
 */
import { chaveDedupe, comOrdinais } from '../dedupe';
import { JANELA_DIAS, LIMIAR_PARECIDO, parecido } from '../descricao';
import type { Recebido } from './mapear';

export type Existente = {
  id: string;
  data: string;
  valor: number;
  descNorm: string;
  fonte: 'pluggy' | 'pdf' | 'manual';
  pluggyTxnId: string | null;
  dedupeKey: string;
  excluido: boolean;
};

export type RecebidoComOrdinal = Recebido & { ordinal: number };

export type Acao =
  | { tipo: 'atualizar'; existenteId: string; recebido: RecebidoComOrdinal }
  | { tipo: 'promover'; existenteId: string; recebido: RecebidoComOrdinal }
  | { tipo: 'ignorar'; existenteId: string; recebido: RecebidoComOrdinal }
  | { tipo: 'inserir'; recebido: RecebidoComOrdinal; dedupeKey: string; suspeitoDuplicado: boolean };

const centavos = (v: number) => Math.round(Math.abs(v) * 100);
const diasEntre = (a: string, b: string) =>
  Math.abs(Date.parse(`${a}T00:00:00Z`) - Date.parse(`${b}T00:00:00Z`)) / 86_400_000;

export function planejarSync(accountId: string, existentes: readonly Existente[], recebidos: readonly Recebido[]): Acao[] {
  const porPluggy = new Map(existentes.filter((e) => e.pluggyTxnId).map((e) => [e.pluggyTxnId!, e]));
  const porChave = new Map(existentes.map((e) => [e.dedupeKey, e]));
  // um existente casa com um recebido só: sem isso, dois lançamentos do Pluggy
  // promoveriam a mesma linha do PDF e um deles sumiria
  const usados = new Set<string>();

  const casou = (e: Existente, r: RecebidoComOrdinal, senaoExcluido: 'atualizar' | 'promover'): Acao => {
    usados.add(e.id);
    // o que o dono apagou não volta (regra 6)
    if (e.excluido) return { tipo: 'ignorar', existenteId: e.id, recebido: r };
    return { tipo: senaoExcluido, existenteId: e.id, recebido: r };
  };

  return comOrdinais(recebidos).map((r): Acao => {
    const chave = chaveDedupe(accountId, r.data, r.valor, r.descNorm, r.ordinal);

    const mesmoId = porPluggy.get(r.pluggyTxnId);
    if (mesmoId && !usados.has(mesmoId.id)) return casou(mesmoId, r, 'atualizar');

    const mesmaChave = porChave.get(chave);
    if (mesmaChave && !usados.has(mesmaChave.id))
      return casou(mesmaChave, r, mesmaChave.fonte === 'pdf' ? 'promover' : 'atualizar');

    // a fatura em PDF traz a data da compra e o texto do extrato; a API pode
    // trazer outro dia e outro texto para a mesma compra
    const candidatos = existentes.filter(
      (e) =>
        !usados.has(e.id) &&
        e.fonte === 'pdf' &&
        !e.pluggyTxnId &&
        centavos(e.valor) === centavos(r.valor) &&
        diasEntre(e.data, r.data) <= JANELA_DIAS &&
        parecido(e.descNorm, r.descNorm) >= LIMIAR_PARECIDO,
    );
    if (candidatos.length === 1) return casou(candidatos[0]!, r, 'promover');

    return { tipo: 'inserir', recebido: r, dedupeKey: chave, suspeitoDuplicado: candidatos.length > 1 };
  });
}

/**
 * Cliente HTTP do Pluggy. Recebe credenciais, `fetch` e espera por parâmetro
 * — sem `server-only` e sem ler `env` — para o teste rodar com respostas
 * simuladas. Quem usa de verdade é cliente.ts, que é `server-only`.
 *
 * Erros nunca carregam credencial nem apiKey na mensagem.
 */
import type { ContaPluggy, FaturaPluggy, TransacaoPluggy } from './mapear';

const BASE = 'https://api.pluggy.ai';
// a apiKey vale 2 h; renovar antes evita uma chamada morrer no meio
const VALIDADE_MS = 110 * 60 * 1000;

export class SemChavePluggy extends Error {
  constructor() {
    super('PLUGGY_CLIENT_ID e PLUGGY_CLIENT_SECRET não configurados');
    this.name = 'SemChavePluggy';
  }
}

type Opcoes = {
  clientId?: string;
  clientSecret?: string;
  fetch?: typeof fetch;
  esperar?: (ms: number) => Promise<void>;
  agora?: () => number;
};

export function criarClientePluggy(o: Opcoes) {
  const f = o.fetch ?? fetch;
  const esperar = o.esperar ?? ((ms: number) => new Promise<void>((ok) => setTimeout(ok, ms)));
  const agora = o.agora ?? Date.now;
  let chave: { valor: string; ate: number } | null = null;

  // instabilidade do Pluggy ou do banco não é assunto do dono: primeira tentativa
  // e mais três, com espera de 1 s, 2 s e 4 s
  async function comTentativas(pedir: () => Promise<Response>): Promise<Response> {
    let ultimo: unknown;
    for (let i = 0; i < 4; i++) {
      if (i > 0) await esperar(1000 * 2 ** (i - 1));
      try {
        const r = await pedir();
        if (r.status < 500) return r;
        ultimo = new Error(`Pluggy respondeu ${r.status}`);
      } catch (e) {
        ultimo = e;
      }
    }
    throw ultimo instanceof Error ? ultimo : new Error('falha ao falar com o Pluggy');
  }

  async function autenticar(): Promise<string> {
    if (!o.clientId || !o.clientSecret) throw new SemChavePluggy();
    if (chave && chave.ate > agora()) return chave.valor;
    const r = await comTentativas(() =>
      f(`${BASE}/auth`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ clientId: o.clientId, clientSecret: o.clientSecret }),
      }),
    );
    if (!r.ok) throw new Error(`Pluggy /auth respondeu ${r.status}`);
    const { apiKey } = (await r.json()) as { apiKey?: string };
    if (!apiKey) throw new Error('Pluggy /auth não devolveu apiKey');
    chave = { valor: apiKey, ate: agora() + VALIDADE_MS };
    return apiKey;
  }

  async function get<T>(caminho: string): Promise<T> {
    const pedir = (k: string) => comTentativas(() => f(`${BASE}${caminho}`, { headers: { 'X-API-KEY': k } }));
    let r = await pedir(await autenticar());
    if (r.status === 401) {
      // a apiKey guardada pode ter sido revogada antes do prazo: uma renovação só
      chave = null;
      r = await pedir(await autenticar());
    }
    if (!r.ok) throw new Error(`Pluggy GET ${caminho.split('?')[0]} respondeu ${r.status}`);
    return (await r.json()) as T;
  }

  return {
    async contasDoItem(itemId: string): Promise<ContaPluggy[]> {
      const r = await get<{ results?: ContaPluggy[] }>(`/accounts?itemId=${encodeURIComponent(itemId)}`);
      return r.results ?? [];
    },
    async transacoes(accountId: string, de: string, ate: string): Promise<TransacaoPluggy[]> {
      const todas: TransacaoPluggy[] = [];
      for (let page = 1; ; page++) {
        const q = new URLSearchParams({ accountId, from: de, to: ate, page: String(page), pageSize: '500' });
        const r = await get<{ results?: TransacaoPluggy[]; totalPages?: number }>(`/transactions?${q}`);
        todas.push(...(r.results ?? []));
        if (page >= (r.totalPages ?? 1)) break;
      }
      return todas;
    },
    async faturas(accountId: string): Promise<FaturaPluggy[]> {
      const r = await get<{ results?: FaturaPluggy[] }>(`/bills?accountId=${encodeURIComponent(accountId)}`);
      return r.results ?? [];
    },
  };
}

export type ClientePluggy = ReturnType<typeof criarClientePluggy>;
```

      const f = r.billId ? fatura.get(r.billId) : undefined;
      // sem fatura conhecida não há mês de referência para o plano da parcela
      const installmentId =
        r.parcela && r.valor > 0 && f ? await planoDeParcela(accountId, r.descNorm, f.mesRef, r.parcela, r.valor) : null;
      const comuns = {
        valor: r.valor.toFixed(2),
        tipo: r.tipo,
        parcelaN: r.parcela?.n ?? null,
        pendente: r.pendente,
        ...(installmentId ? { installmentId } : {}),
        ...(f ? { statementId: f.statementId } : {}),
      };

      if (a.tipo === 'atualizar') {
        resultado.atualizados++;
        comandos.push(db.update(txn).set(comuns).where(and(eq(txn.id, a.existenteId), eq(txn.userId, userId))));
      } else if (a.tipo === 'promover') {
        resultado.promovidos++;
        // apelido, categoria, data, descrição e dedupe_key ficam: são do dono ou
        // são o que faz o PDF daquele mês continuar colidindo
        comandos.push(
          db
            .update(txn)
            .set({ ...comuns, pluggyTxnId: r.pluggyTxnId, fonte: 'pluggy' })
            .where(and(eq(txn.id, a.existenteId), eq(txn.userId, userId))),
        );
      } else {
        if (a.suspeitoDuplicado) resultado.suspeitos++;
        else resultado.novos++;
        const regra = regras.get(chaveAprendizado(r.descNorm));
        comandos.push(
          db.insert(txn).values({
            userId,
            accountId,
            statementId: f?.statementId ?? null,
            data: r.data,
            valor: r.valor.toFixed(2),
            descOrigem: r.descOrigem,
            descNorm: r.descNorm,
            apelido: regra?.apelido ?? null,
            categoryId: regra?.categoryId ?? null,
            categoryFonte: regra ? 'regra' : 'ia',
            tipo: r.tipo,
            installmentId,
            parcelaN: r.parcela?.n ?? null,
            fonte: 'pluggy',
            pluggyTxnId: r.pluggyTxnId,
            dedupeKey: a.dedupeKey,
            pendente: r.pendente,
            suspeitoDuplicado: a.suspeitoDuplicado,
          }),
        );
      }
    }

    if (comandos.length) await db.batch(comandos as unknown as Parameters<typeof db.batch>[0]);
  }

  await db.insert(syncRun).values({
    connectionId,
    terminadoEm: new Date(),
    resultado: 'ok',
    novos: resultado.novos,
    atualizados: resultado.atualizados,
    duplicados: resultado.suspeitos,
    detalhe: { accountId, promovidos: resultado.promovidos, ignorados: resultado.ignorados, descartados },
  });
  return resultado;
}
```

Notas:
- Se o TypeScript recusar o cast de `comandos` para o `batch`, tipar `comandos` como `BatchItem<'pg'>[]` (`import type { BatchItem } from 'drizzle-orm/batch'`) e passar `comandos as [BatchItem<'pg'>, ...BatchItem<'pg'>[]]`. Só tipagem; nunca mudar o que é gravado.
- Faturas, parcelas e `sync_run` ficam fora do batch: são idempotentes (upsert) e o lançamento é o que precisa ser tudo ou nada.
