```ts
  /** o que o cartão pode consumir guardando a meta; null sem renda */

  compromissos: number;
  disponivel: number | null;
  poupanca: number | null;

export function planoSaudavel(cfg: ConfigRenda, pj: Projecao | null): PlanoSaudavel | null {
  if (!pj) return null;
  const { renda, metaPoupanca, foraCartao } = cfg;
  const teto = renda > 0 ? Math.max(0, renda * (1 - metaPoupanca / 100) - foraCartao) : null;
  const compromissos = pj.parcelas + pj.fixos;
  return {
    teto,
    compromissos,
    disponivel: teto === null ? null : teto - compromissos,
    poupanca: renda > 0 ? (renda * metaPoupanca) / 100 : null,
  };
}

export const ESSENCIAIS = new Set(['Mercado & Padaria', 'Saúde & Farmácia', 'Transporte', 'Casa & Contas', 'Educação']);

export type CatVariavel = { cat: string; media: number };

export function variavelPorCategoria(ctx: Contexto): { cats: CatVariavel[]; meses: number } {
  const ms0 = mesesDisponiveis(ctx);
  const alvo = mesAtual(ctx);
  const ms = (alvo ? ms0.filter((m) => m <= alvo) : ms0).slice(-3);
  if (!ms.length) return { cats: [], meses: 0 };
  const fixas = new Set(recorrentesFixas(ctx).map((r) => normDesc(r.desc)));
  const acc = new Map<string, number>();
  for (const m of ms)
    for (const l of lancsDoMes(ctx, m)) {
      if (classeLanc(ctx, l, fixas) !== 'escolha') continue;
      const c = normCat(l.cat);
      acc.set(c, (acc.get(c) || 0) + (+l.val || 0));
    }
  return {
    meses: ms.length,
    cats: [...acc.entries()]
      .map(([cat, v]) => ({ cat, media: v / ms.length }))
      .filter((x) => x.media > 0)
      .sort((a, b) => b.media - a.media),
  };
}

/* metas mensais: o teto com poupança é o alvo, a renda é o limite intransponível */
export function metasDoPlano(ctx: Contexto, prefs: Prefs): Metas | null {
  const pj = projecao(ctx);
  if (!pj) return null;
  const pl = planoSaudavel(ctx, pj);
  if (!pl || pl.teto === null) return null;
  const teto = pl.teto;
  const pi = parcelasInfo(ctx);
  const vc = variavelPorCategoria(ctx);
  const variavelHoje = vc.cats.reduce((s, c) => s + c.media, 0) || pj.variavel;
  const ramp = (RITMOS[prefs.ritmo] || RITMOS.equilibrado!).ramp;
  const rendaLiquida = Math.max(0, ctx.renda - ctx.foraCartao); // o que pode ir para o cartão sem poupar nada
  const meses = [0, 1, 2].map((i) => {
    const m = addMonths(pj.prox, i);
    const compromissos = (pi.porMes.find((x) => x[0] === m)?.[1].tot || 0) + pj.fixos;
    const disp = teto - compromissos; // sobra guardando a meta
    const dispMax = rendaLiquida - compromissos; // sobra sem guardar nada
    const precisaCortar = Math.max(0, variavelHoje - disp);
    const rampa = Math.max(0, variavelHoje - precisaCortar * ramp[i]!);
    const alvo = Math.max(0, Math.min(rampa, Math.max(0, dispMax), i === 2 ? Math.max(disp, 0) : Infinity));
    return { mes: m, compromissos, disp, dispMax, alvo, poupanca: rendaLiquida - compromissos - alvo, estoura: dispMax < 0 };
  });
  return {
    meses,
    variavelHoje,
    teto: pl.teto,
    metaPoupancaValor: (ctx.renda * ctx.metaPoupanca) / 100,
    cats: vc.cats,
    mesesBase: vc.meses,
    precisa: meses.some((m) => m.alvo < variavelHoje - 1),
    impossivel: meses.some((m) => m.estoura),
  };
}

export function pisoCat(prefs: Prefs, c: { cat: string; atual: number }): number {
  if (prefs.intocaveis.includes(c.cat)) return c.atual; // não mexe, ponto
  if (prefs.cortar.includes(c.cat)) return 0.25 * c.atual; // pode cortar fundo — mas não até quase zero
  return (ESSENCIAIS.has(c.cat) ? 0.85 : 0.45) * c.atual;
}

export type CatPlano = { cat: string; atual: number; acao: string; alvos: number[]; travada?: boolean };

/* fecha o mês no alvo sem passar dos limites de cada categoria.
   Devolve o que faltou cortar (0 quando fechou). */
export function fecharMes(prefs: Prefs, cats: CatPlano[], i: number, alvo: number, cap: number): number {
  const base = (c: CatPlano) => (i === 0 ? c.atual : c.alvos[i - 1]!);
  const teto = (c: CatPlano) => base(c);
  const piso = (c: CatPlano) =>
    prefs.intocaveis.includes(c.cat) ? c.atual : Math.max(pisoCat(prefs, c), base(c) * (1 - cap));
  cats.forEach((c) => {
    c.alvos[i] = Math.min(teto(c), Math.max(piso(c), c.alvos[i]!));
  });
  const soma = () => cats.reduce((s, c) => s + c.alvos[i]!, 0);
  let dif = alvo - soma();
  for (let passo = 0; passo < 10 && Math.abs(dif) > 1; passo++) {
    const pesos = cats.map((c) => (dif > 0 ? Math.max(0, teto(c) - c.alvos[i]!) : Math.max(0, c.alvos[i]! - piso(c))));
    const tot = pesos.reduce((a, b) => a + b, 0);
    if (tot <= 0) break;
    cats.forEach((c, k) => {
      c.alvos[i] = Math.min(teto(c), Math.max(piso(c), c.alvos[i]! + (dif * pesos[k]!) / tot));
    });
    dif = alvo - soma();
  }
  cats.forEach((c) => {
    c.alvos[i] = Math.round(c.alvos[i]!);
  });
  return dif;
}

/**
 * Aplica as travas sobre o que a IA devolveu (ou sobre nada: `bruto` nulo dá
 * o plano calculado). Resposta de formato inesperado pode lançar, como no
 * legado — quem chama trata como "a IA não respondeu direito".
 */
export function ajustarPlano(prefs: Prefs, bruto: unknown, mt: Metas): ResultadoPlano | null {
  const b = bruto as BrutoIA | null | undefined;
  const atual = new Map(mt.cats.map((c) => [c.cat, c.media]));
  const cats: CatPlano[] = (b?.categorias || [])
    .filter((c) => (CATS as readonly string[]).includes(c.cat as string) && atual.has(c.cat as string))
    .map((c) => ({
      cat: c.cat as string,
      atual: atual.get(c.cat as string) || 0,
      acao: String(c.acao || '').slice(0, 110),
      alvos: [0, 1, 2].map((i) => Math.max(0, +((Array.isArray(c.alvos) ? c.alvos[i] : 0) as number) || 0)),
    }));
  for (const [cat, media] of atual)
    if (!cats.some((c) => c.cat === cat) && media > 0) cats.push({ cat, atual: media, acao: '', alvos: [media, media, media] });
  if (!cats.length) return null;

  /* o que a pessoa mandou não mexer, não se mexe — a IA não tem voto aqui */
  cats.forEach((c) => {
    if (prefs.intocaveis.includes(c.cat)) {
      c.alvos = [c.atual, c.atual, c.atual];
      c.travada = true;
    }
  });

  const avisos: { mes: string; falta: number }[] = [];
  [0, 1, 2].forEach((i) => {
    let resto = fecharMes(prefs, cats, i, mt.meses[i]!.alvo, quedaMax(prefs));
    if (resto < -1) resto = fecharMes(prefs, cats, i, mt.meses[i]!.alvo, 1); // não coube devagar: vai até o piso
    if (resto < -1) avisos.push({ mes: mt.meses[i]!.mes, falta: -resto });
  });

  cats.sort((a, c) => c.atual - c.alvos[2]! - (a.atual - a.alvos[2]!));
  return {
    cats,
    fixos: (b?.fixos || []).slice(0, 3).map((f) => ({
      desc: String(f.desc || '').slice(0, 40),
      acao: String(f.acao || '').slice(0, 110),
      economia: +(f.economia as number) || 0,
    })),
    nota: String(b?.nota || '').slice(0, 240),
    meses: mt.meses.map((m) => ({ mes: m.mes, alvo: Math.round(m.alvo), poupanca: Math.round(m.poupanca), disp: Math.round(m.disp) })),
    avisos,
    metaPoupancaValor: Math.round(mt.metaPoupancaValor || 0),
    variavelHoje: Math.round(mt.variavelHoje),
    teto: Math.round(mt.teto),
    prefs: { ...prefs },
    criadoEm: new Date().toISOString(),
  };
}

/* quanto foi gasto de verdade num mês do plano, por categoria */
export function realDoMes(ctx: Contexto, m: string, cat: string): number | null {
  if (!mesesDisponiveis(ctx).includes(m)) return null;
  const fixas = new Set(recorrentesFixas(ctx).map((r) => normDesc(r.desc)));
  let t = 0;
  for (const l of lancsDoMes(ctx, m)) if (classeLanc(ctx, l, fixas) === 'escolha' && normCat(l.cat) === cat) t += +l.val || 0;
  return t;
}
```

```ts
 * Validação das respostas do questionário do plano. Devolve resultado em vez

import type { Validado } from './ajustes';
import { CATS } from './calculos/base';
import { ASSINA, COMIDAS, OBJETIVOS, RITMOS, type Prefs } from './calculos/plano';

const erro = (codigo: string, mensagem: string) => ({ ok: false as const, codigo, mensagem });

// Object.hasOwn: "toString" e "constructor" existem em todo objeto e não são opção
const escolha = (v: unknown, opcoes: Record<string, unknown>) =>
  typeof v === 'string' && Object.hasOwn(opcoes, v) ? v : null;

function categorias(v: unknown): string[] | null {
  if (!Array.isArray(v) || !v.every((x) => typeof x === 'string' && (CATS as readonly string[]).includes(x))) return null;
  return [...new Set(v as string[])];
}

export function validarPrefs(corpo: unknown): Validado<Prefs> {
  const c = corpo && typeof corpo === 'object' ? (corpo as Record<string, unknown>) : {};
  const p = c.prefs && typeof c.prefs === 'object' && !Array.isArray(c.prefs) ? (c.prefs as Record<string, unknown>) : null;
  if (!p) return erro('prefs_ausente', 'responda o questionário');

  const objetivo = escolha(p.objetivo, OBJETIVOS);
  if (!objetivo) return erro('objetivo_invalido', 'objetivo fora da lista');
  const ritmo = escolha(p.ritmo, RITMOS);
  if (!ritmo) return erro('ritmo_invalido', 'ritmo fora da lista');
  const comida = escolha(p.comida, COMIDAS);
  if (!comida) return erro('comida_invalida', 'resposta sobre comida fora da lista');
  const assinaturas = escolha(p.assinaturas, ASSINA);
  if (!assinaturas) return erro('assinaturas_invalidas', 'resposta sobre assinaturas fora da lista');

  const intocaveis = categorias(p.intocaveis);
  const cortarTudo = categorias(p.cortar);
  if (!intocaveis || !cortarTudo) return erro('categoria_invalida', 'categoria fora da lista');
  const cortar = cortarTudo.filter((x) => !intocaveis.includes(x));

  if (typeof p.evento !== 'string') return erro('evento_invalido', 'evento precisa ser texto');
  const evento = p.evento.trim();
  if (evento.length > 120) return erro('evento_longo', 'evento até 120 caracteres');

  return { ok: true, mudanca: { objetivo, ritmo, intocaveis, cortar, comida, assinaturas, evento } };
```

```ts
  const { ctx } = await carregarContexto(userId, { mes: null, cartoes: [], obra });
  return ctx;
}

export async function planoAtivo(userId: string) {
  const [linha] = await db
    .select({ id: plan.id, prefs: plan.prefs, resultado: plan.resultado })
    .from(plan)
    .where(and(eq(plan.userId, userId), eq(plan.ativo, true)))
    .orderBy(desc(plan.criadoEm))
    .limit(1);
  // plano gravado antes da migração 0004 não tem resultado: é como não ter plano
  if (!linha?.resultado) return null;
  return { id: linha.id, prefs: linha.prefs as Prefs, resultado: linha.resultado as ResultadoPlano };
}

type RegistroIA = { modelo: string; uso: Uso; prompt: string; conteudo: unknown };

/**
 * Tudo ou nada (`db.batch` é uma transação no driver HTTP do Neon): desativa
 * o plano anterior, grava o novo e as metas, e o uso da IA quando houver.
 */
export async function gravarPlano(userId: string, r: ResultadoPlano, cfg: Pick<Contexto, 'renda' | 'metaPoupanca'>, ia?: RegistroIA) {
  const nomes = r.cats.map((c) => c.cat);
  const ids = await db.select({ id: category.id, nome: category.nome }).from(category).where(inArray(category.nome, nomes));
  const idDe = new Map(ids.map((x) => [x.nome, x.id]));
  const planId = randomUUID();
  // ajustarPlano nunca devolve plano sem categoria, e as categorias vêm de CATS,
  // semeadas em 0001_categorias.sql: a lista de metas nunca é vazia
  const metas = r.cats.flatMap((c) => {
    const categoryId = idDe.get(c.cat);
    if (categoryId === undefined) return [];
    return r.meses.map((m, i) => ({ planId, mesRef: `${m.mes}-01`, categoryId, alvo: c.alvos[i]!.toFixed(2) }));
  });

  const desativar = db.update(plan).set({ ativo: false }).where(and(eq(plan.userId, userId), eq(plan.ativo, true)));
  const inserir = db.insert(plan).values({
    id: planId,
    userId,
    ritmo: r.prefs.ritmo,
    objetivo: r.prefs.objetivo,
    rendaBase: cfg.renda.toFixed(2),
    metaPoupanca: cfg.metaPoupanca.toFixed(2),
    prefs: r.prefs,
    resultado: r,
  });
  const inserirMetas = db.insert(planTarget).values(metas);

  if (!ia) {
    await db.batch([desativar, inserir, inserirMetas]);
    return planId;
  }
  const inputHash = createHash('sha256').update(ia.prompt).digest('hex');
  const custoUsd = ia.uso.custoUsd === null ? null : ia.uso.custoUsd.toFixed(6);
  // refinar duas vezes com o mesmo prompt bate na chave única (user, kind, escopo, input_hash)
  const registrar = db
    .insert(aiOutput)
    .values({ userId, kind: 'plano', escopo: 'plano', modelo: ia.modelo, inputHash, conteudo: ia.conteudo, tokensIn: ia.uso.tokensIn, tokensOut: ia.uso.tokensOut, custoUsd })
    .onConflictDoUpdate({
      target: [aiOutput.userId, aiOutput.kind, aiOutput.escopo, aiOutput.inputHash],
      set: { modelo: ia.modelo, conteudo: ia.conteudo, tokensIn: ia.uso.tokensIn, tokensOut: ia.uso.tokensOut, custoUsd, criadoEm: sql`now()` },
    });
  await db.batch([desativar, inserir, inserirMetas, registrar]);
  return planId;
}
```

```ts
import type { NextRequest } from 'next/server';
import { recorrentesFixas } from '@/lib/calculos/composicao';
import { ajustarPlano, metasDoPlano, type ResultadoPlano } from '@/lib/calculos/plano';
import { lerFiltro } from '@/lib/filtro';
import { promptPlano } from '@/lib/ia';
import { RespostaNaoJson, SemChaveIA, pedirJson } from '@/lib/openrouter';
import { validarPrefs } from '@/lib/plano-prefs';
import { contextoDoPlano, gravarPlano, planoAtivo } from '@/lib/plano-repo';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';

export const runtime = 'nodejs';

const iaInvalida = () => new ErroDeUso('ia_invalida', 'A IA não respondeu direito; o plano calculado segue valendo', 502);
const semRenda = () => new ErroDeUso('sem_renda', 'Informe a renda em Ajustes');

export async function POST(req: NextRequest) {
  // só "separar obra" vale aqui; cartões e mês do filtro não entram no plano
  const { obra } = lerFiltro(req.nextUrl.searchParams);

  return comUsuario(async (userId) => {
    const corpo = await corpoJson<unknown>(req);
    const c = corpo && typeof corpo === 'object' ? (corpo as Record<string, unknown>) : {};
    const refinar = c.refinar === true;
    if (refinar === 'prefs' in c) throw new ErroDeUso('pedido_invalido', 'mande as preferências ou peça para refinar');

    const ctx = await contextoDoPlano(userId, obra);

    if (!refinar) {
      const v = validarPrefs(c);
      if (!v.ok) throw new ErroDeUso(v.codigo, v.mensagem);
      const mt = metasDoPlano(ctx, v.mudanca);
      if (!mt) throw semRenda();
      const r = ajustarPlano(v.mudanca, null, mt);
      if (!r) throw new ErroDeUso('sem_variavel', 'Não há gasto variável nos últimos meses para planejar');
      await gravarPlano(userId, r, ctx);
      return { ok: true };
    }

    const ativo = await planoAtivo(userId);
    if (!ativo) throw new ErroDeUso('sem_plano', 'Monte o plano primeiro', 404);
    const mt = metasDoPlano(ctx, ativo.prefs);
    if (!mt) throw semRenda();
    const prompt = promptPlano(mt, ativo.prefs, recorrentesFixas(ctx));

    let resposta: Awaited<ReturnType<typeof pedirJson<unknown>>>;
    try {
      resposta = await pedirJson<unknown>(prompt);
    } catch (e) {
      if (e instanceof SemChaveIA) throw new ErroDeUso('sem_chave', 'Configure OPENROUTER_API_KEY');
      if (e instanceof RespostaNaoJson) throw iaInvalida();
      throw e;
    }

    // a resposta do modelo é dado não confiável: formato inesperado é "não respondeu direito", não 503
    let r: ResultadoPlano | null;
    try {
      r = ajustarPlano(ativo.prefs, resposta.dados, mt);
    } catch {
      r = null;
    }
    if (!r) throw iaInvalida();
    await gravarPlano(userId, r, ctx, { modelo: resposta.modelo, uso: resposta.uso, prompt, conteudo: resposta.dados });
    return { ok: true };
  });
}
```
