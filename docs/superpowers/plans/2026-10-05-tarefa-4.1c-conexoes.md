```ts
/**
 * Estado de uma conexão na tela: o texto e a ação que o guia ("Conexões") pede
 * para cada caso. Puro: o teste importa, a página também.
 */
export type EstadoBanco = 'ativa' | 'atualizando' | 'erro_login' | 'consentimento_expirado' | 'desconectada' | 'erro';

export type EntradaEstado = {
  banco: string;
  status: EstadoBanco;
  saude: string | null;
  consentExpiraEm: Date | null;
  ultimoSyncEm: Date | null;
  agora: Date;
};

export type EstadoTela = {
  chave: 'ativa' | 'atualizando' | 'erro_login' | 'consentimento' | 'instavel' | 'erro';
  texto: string;
  acao: 'atualizar' | 'reconectar' | 'renovar' | null;
};

export function haQuanto(quando: Date, agora: Date): string {
  const min = Math.floor((agora.getTime() - quando.getTime()) / 60_000);
  if (min < 1) return 'agora há pouco';
  if (min < 60) return `há ${min} ${min === 1 ? 'minuto' : 'minutos'}`;
  const h = Math.floor(min / 60);
  if (h < 24) return `há ${h} ${h === 1 ? 'hora' : 'horas'}`;
  const d = Math.floor(h / 24);
  return `há ${d} ${d === 1 ? 'dia' : 'dias'}`;
}

export function estadoDaConexao(e: EntradaEstado): EstadoTela {
  // banco fora do ar é deles: mandar "reconectar" faria a pessoa procurar defeito onde não há
  if (e.saude === 'UNSTABLE' || e.saude === 'OFFLINE') {
    return { chave: 'instavel', texto: `O ${e.banco} está instável agora — é deles, não seu.`, acao: null };
  }
  if (e.status === 'consentimento_expirado' || (e.consentExpiraEm && e.consentExpiraEm.getTime() <= e.agora.getTime())) {
    return { chave: 'consentimento', texto: 'A autorização expirou.', acao: 'renovar' };
  }
  if (e.status === 'erro_login') return { chave: 'erro_login', texto: 'O banco pediu sua senha de novo.', acao: 'reconectar' };
  if (e.status === 'atualizando') return { chave: 'atualizando', texto: 'Atualizando…', acao: null };
  if (e.status === 'erro') return { chave: 'erro', texto: 'A última atualização falhou.', acao: 'atualizar' };
  return {
    chave: 'ativa',
    texto: e.ultimoSyncEm ? `Atualizado ${haQuanto(e.ultimoSyncEm, e.agora)}` : 'Ainda sem lançamentos',
    acao: 'atualizar',
  };
}

/** Conexões que ficaram sem conta (os cartões foram religados a um item novo). */
export function conexoesOrfas(conexoes: { id: string; status: string }[], comConta: string[], atual: string): string[] {
  const usadas = new Set(comConta);
  return conexoes.filter((c) => c.id !== atual && c.status !== 'desconectada' && !usadas.has(c.id)).map((c) => c.id);
}
```

```ts
/**
 * "Atualizar agora": do dia 1 do mês do último sync, menos um mês, até hoje —
 * pega o pendente que virou postado e a fatura que fechou sem repetir 3 meses.

export function janelaAtualizar(ultimoSyncEm: Date | null, agora: Date): { de: string; ate: string } {
  if (!ultimoSyncEm) return janelaInicial(agora);
  const ultimo = diaSP.format(ultimoSyncEm);
  const ano = Number(ultimo.slice(0, 4));
  const mes = Number(ultimo.slice(5, 7)) - 1;
  const anoDe = mes < 1 ? ano - 1 : ano;
  const mesDe = mes < 1 ? mes + 12 : mes;
  return { de: `${anoDe}-${String(mesDe).padStart(2, '0')}-01`, ate: diaSP.format(agora) };
}
```

```ts
/** O item do Pluggy de uma conexão do usuário — `itemId` nunca vem do cliente. */
export async function itemDaConexao(userId: string, conexaoId: string) {
  const [c] = await db
    .select({ id: connection.id, pluggyItemId: connection.pluggyItemId, ultimoSyncEm: connection.ultimoSyncEm, status: connection.status })
    .from(connection)
    .where(and(eq(connection.id, conexaoId), eq(connection.userId, userId), ne(connection.status, 'desconectada')))
    .limit(1);
  return c ?? null;
}

/** Grava o que o Pluggy disse do item: estado, saúde do conector, prazo do consentimento e erro. */
export async function registrarItem(userId: string, conexaoId: string, item: ItemPluggy): Promise<EstadoConexao> {
  const status: EstadoConexao =
    item.status === 'WAITING_USER_ACTION' ? 'erro_login' : item.status === 'OUTDATED' ? 'erro' : statusDoItem(item.status);
  await db
    .update(connection)
    .set({
      status,
      saudeConector: item.connector?.health?.status ?? null,
      consentExpiraEm: item.consentExpiresAt ? new Date(item.consentExpiresAt) : null,
      ultimoErro: status === 'ativa' || status === 'atualizando' ? null : (item.error?.message ?? null),
    })
    .where(and(eq(connection.id, conexaoId), eq(connection.userId, userId)));
  return status;
}

/**
 * Desconecta as conexões que ficaram sem conta. A revogação no Pluggy é melhor
 * esforço (falha vai ao log); a marcação local acontece de qualquer jeito, senão
 * a sobra continuaria na tela.
 */
export async function limparOrfas(userId: string, atual: string) {
  const conexoes = await db.select({ id: connection.id, status: connection.status, itemId: connection.pluggyItemId }).from(connection).where(eq(connection.userId, userId));
  const comConta = await db
    .selectDistinct({ id: account.connectionId })
    .from(account)
    .where(and(eq(account.userId, userId), isNotNull(account.connectionId)));
  const orfas = conexoesOrfas(conexoes, comConta.map((c) => c.id!).filter(Boolean), atual);
  for (const id of orfas) {
    const itemId = conexoes.find((c) => c.id === id)!.itemId;
    try {
      await pluggy().apagarItem(itemId);
    } catch (e) {
      console.error('[orfas]', e);
    }
  }
  if (orfas.length) {
    await db.update(connection).set({ status: 'desconectada' }).where(and(inArray(connection.id, orfas), eq(connection.userId, userId)));
  }
}

/** Revoga no Pluggy e para de sincronizar; os lançamentos ficam (guia, "Conexões"). */
export async function desconectar(userId: string, conexaoId: string): Promise<boolean> {
  const c = await itemDaConexao(userId, conexaoId);
  if (!c) return false;
  await pluggy().apagarItem(c.pluggyItemId);
  await db.batch([
    db.update(connection).set({ status: 'desconectada' }).where(and(eq(connection.id, c.id), eq(connection.userId, userId))),
    // o pluggy_account_id fica: um "Conectar banco" futuro religa o mesmo cartão
    db.update(account).set({ connectionId: null }).where(and(eq(account.connectionId, c.id), eq(account.userId, userId))),
  ]);
  return true;
}

  return gravados.map((g) => ({ id: g.id, nome: (g.apelido || g.banco) + (g.last4 ? ` ••${g.last4}` : '') }));
}
```

```ts
export async function POST(req: NextRequest) {
  return comUsuario(async (userId) => {
    // corpo opcional: sem ele, conexão nova; com `conexaoId`, o widget abre no item existente (reconectar/renovar)

  });
```

```ts
import type { NextRequest } from 'next/server';
import { ehUuid } from '@/lib/filtro';
import { desconectar } from '@/lib/pluggy/conexao-repo';
import { SemChavePluggy } from '@/lib/pluggy/http';
import { ErroDeUso, comUsuario } from '@/lib/rota';

export const runtime = 'nodejs';

const naoEncontrada = () => new ErroDeUso('conexao_nao_encontrada', 'Conexão não encontrada', 404);

export async function DELETE(_req: NextRequest, ctx: { params: Promise<{ id: string }> }) {
  const { id } = await ctx.params;
  return comUsuario(async (userId) => {
    if (!ehUuid(id)) throw naoEncontrada();
    try {
      if (!(await desconectar(userId, id))) throw naoEncontrada();
```

```ts
import type { NextRequest } from 'next/server';
import { ehUuid } from '@/lib/filtro';
import { pluggy } from '@/lib/pluggy/cliente';
import { cartoesDaConexao, itemDaConexao, limparOrfas, registrarItem } from '@/lib/pluggy/conexao-repo';
import { SemChavePluggy } from '@/lib/pluggy/http';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';

export const runtime = 'nodejs';
export const maxDuration = 60;

const naoEncontrada = () => new ErroDeUso('conexao_nao_encontrada', 'Conexão não encontrada', 404);

/**
 * "Atualizar agora" em duas etapas para caber no limite da Vercel: `iniciar`
 * pede ao Pluggy para buscar no banco; `conferir` (a tela chama a cada 3 s)
 * diz quando o item ficou pronto e devolve os cartões para trazer.
 */
export async function POST(req: NextRequest, ctx: { params: Promise<{ id: string }> }) {
  const { id } = await ctx.params;
  return comUsuario(async (userId) => {
    if (!ehUuid(id)) throw naoEncontrada();
    const corpo = await corpoJson<unknown>(req);
    const etapa = (corpo as { etapa?: unknown } | null)?.etapa === 'conferir' ? 'conferir' : 'iniciar';
    const c = await itemDaConexao(userId, id);
    if (!c) throw naoEncontrada();

    let item;
    try {
      item = etapa === 'iniciar' ? await pluggy().atualizarItem(c.pluggyItemId) : await pluggy().item(c.pluggyItemId);
    } catch (e) {
      if (e instanceof SemChavePluggy) throw new ErroDeUso('sem_chave_pluggy', 'Configure PLUGGY_CLIENT_ID e PLUGGY_CLIENT_SECRET no .env.local', 501);
      throw e;
    }
    if (!item) throw naoEncontrada();
    const status = await registrarItem(userId, c.id, item);

    if (status === 'erro_login') throw new ErroDeUso('erro_login', 'O banco pediu sua senha de novo. Use "Reconectar".');
    if (status === 'atualizando' || etapa === 'iniciar') return { estado: 'atualizando' as const };
    if (status !== 'ativa' && item.status !== 'OUTDATED') {
      throw new ErroDeUso('erro_pluggy', 'O Pluggy não conseguiu ler o banco. Tente de novo em instantes.', 502);
    }
    await limparOrfas(userId, c.id);
    return {
      estado: 'ok' as const,
      cartoes: await cartoesDaConexao(userId, c.id),
      aviso: item.status === 'OUTDATED' ? 'O banco não respondeu na última leitura; os dados podem estar atrasados.' : null,
    };
  });
}
```

Nota: `iniciar` sempre responde `atualizando` — mesmo se o PATCH voltar `UPDATED` na hora, a tela confere em seguida e segue.

```ts
import { buscar, falhaDe, textoDoErro } from '@/lib/falha';

export type Cartao = { id: string; nome: string };
export type Resumo = { nome: string; texto: string; falhou: boolean };

const plural = (n: number, um: string, varios: string) => `${n} ${n === 1 ? um : varios}`;
export const espera = (ms: number) => new Promise((ok) => setTimeout(ok, ms));

/** Traz os cartões um a um (uma requisição cada, pelo limite da Vercel); um que falha não para os outros. */
export async function trazerCartoes(
  cartoes: Cartao[],
  janela: 'inicial' | 'atualizar',
  aoProgresso: (aviso: string, resumos: Resumo[]) => void,
): Promise<Resumo[]> {
  const feitos: Resumo[] = [];
  for (let i = 0; i < cartoes.length; i++) {
    const c = cartoes[i]!;
    aoProgresso(`Trazendo ${c.nome} — ${i + 1} de ${cartoes.length}…`, [...feitos]);
    try {
      const r = await buscar(`/api/accounts/${c.id}/sync${janela === 'atualizar' ? '?janela=atualizar' : ''}`, { method: 'POST' });
      if (!r.ok) throw await falhaDe(r);
      const s = (await r.json()) as { novos: number; atualizados: number; promovidos: number; suspeitos: number; ignorados: number; categorizados: number; semCategoria: number };
      const partes = [
        plural(s.novos, 'novo', 'novos'),
        s.ignorados + s.atualizados + s.promovidos > 0 ? `${s.ignorados + s.atualizados + s.promovidos} já estavam` : null,
        s.categorizados > 0 ? plural(s.categorizados, 'categorizado', 'categorizados') : null,
        s.suspeitos > 0 ? plural(s.suspeitos, 'suspeito de duplicado', 'suspeitos de duplicado') : null,
        s.semCategoria > 0 ? `${s.semCategoria} sem categoria` : null,
      ].filter(Boolean);
      feitos.push({ nome: c.nome, texto: partes.join(', '), falhou: false });
    } catch (e) {
      feitos.push({ nome: c.nome, texto: textoDoErro(e, navigator.onLine).texto, falhou: true });
    }
    aoProgresso(`Trazendo ${c.nome} — ${i + 1} de ${cartoes.length}…`, [...feitos]);
  }
  return feitos;
}

/** A frase do fim, contando falhas — "Pronto." só quando tudo veio. */
export function fraseDoFim(cartoes: number, resumos: Resumo[], aviso: string | null): string {
  const falhas = resumos.filter((r) => r.falhou).length;
  const fim = !cartoes
    ? 'Conectado, mas este banco não tem cartão de crédito — contas correntes entram na fatia 6.'
    : falhas === cartoes
      ? 'Nenhum cartão foi trazido; veja os erros abaixo.'
      : falhas > 0
        ? `Pronto, com falhas em ${plural(falhas, 'cartão', 'cartões')}.`
        : 'Pronto.';
  return aviso && falhas < cartoes ? `${fim} ${aviso}` : fim;
}
```

```tsx
'use client';

import { useRef, useState, useTransition } from 'react';
import { useRouter } from 'next/navigation';
import AvisoDeFalha from '@/components/casca/AvisoDeFalha';
import ConectarBanco from '@/components/ajustes/ConectarBanco';
import { ErroLocal, buscar, falhaDe, textoDoErro, type Falha } from '@/lib/falha';
import { espera, fraseDoFim, trazerCartoes, type Cartao, type Resumo } from '@/lib/pluggy/fluxo-cliente';

type Props = { conexaoId: string; banco: string; acao: 'atualizar' | 'reconectar' | 'renovar' | null };

export default function AcaoConexao({ conexaoId, banco, acao }: Props) {
  const router = useRouter();
  const [, startTransition] = useTransition();
  const [ocupado, setOcupado] = useState(false);
  const [aviso, setAviso] = useState('');
  const [resumos, setResumos] = useState<Resumo[]>([]);
  const [erro, setErro] = useState<Falha | null>(null);
  const dialogo = useRef<HTMLDialogElement>(null);
  const [erroDialogo, setErroDialogo] = useState('');

  async function atualizar() {
    setOcupado(true);
    setErro(null);
    setResumos([]);
    try {
      const url = `/api/connections/${conexaoId}/sync`;
      const pedir = async (etapa: 'iniciar' | 'conferir') => {
        const r = await buscar(url, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ etapa }) });
        if (!r.ok) throw await falhaDe(r);
        return (await r.json()) as { estado: 'ok' | 'atualizando'; cartoes?: Cartao[]; aviso?: string | null };
      };
      setAviso('Pedindo ao banco os dados novos…');
      await pedir('iniciar');
      let d: Awaited<ReturnType<typeof pedir>> | null = null;
      for (let t = 0; t < 40; t++) {
        await espera(3000);
        d = await pedir('conferir');
        if (d.estado === 'ok') break;
        setAviso('O banco ainda está mandando os dados…');
      }
      if (d?.estado !== 'ok') throw new ErroLocal('O banco demorou demais. Tente de novo daqui a pouco.');
      const cartoes = d.cartoes ?? [];
      const feitos = await trazerCartoes(cartoes, 'atualizar', (a, r) => {
        setAviso(a);
        setResumos(r);
      });
      setResumos(feitos);
      setAviso(fraseDoFim(cartoes.length, feitos, d.aviso ?? null));
    } catch (e) {
      setAviso('');
      setErro(textoDoErro(e, navigator.onLine));
    } finally {
      setOcupado(false);
      startTransition(() => router.refresh());
    }
  }

  async function confirmarDesconectar() {
    setErroDialogo('');
    try {
      const r = await buscar(`/api/connections/${conexaoId}`, { method: 'DELETE' });
      if (!r.ok) throw await falhaDe(r);
      dialogo.current?.close();
      startTransition(() => router.refresh());
    } catch (e) {
      setErroDialogo(textoDoErro(e, navigator.onLine).texto);
    }
  }

  return (
    <div className="conexao-acoes">
      {acao === 'atualizar' && (
        <button type="button" onClick={() => void atualizar()} disabled={ocupado}>
          {ocupado ? 'Atualizando…' : 'Atualizar agora'}
        </button>
      )}
      {(acao === 'reconectar' || acao === 'renovar') && <ConectarBanco conexaoId={conexaoId} rotulo={acao === 'renovar' ? 'Renovar' : 'Reconectar'} />}
      <button type="button" className="link" onClick={() => { setErroDialogo(''); dialogo.current?.showModal(); }} disabled={ocupado}>
        Desconectar
      </button>
      {erro && <AvisoDeFalha falha={erro} />}
      <div role="status" aria-live="polite">
        <span className="suave">{aviso}</span>
        {resumos.length > 0 && (
          <ul className="conectar-resumo">
            {resumos.map((r, i) => (
              <li key={`${i}-${r.nome}`} className={r.falhou ? 'falhou' : undefined}>
                <strong>{r.nome}:</strong> {r.texto}
              </li>
            ))}
          </ul>
        )}
      </div>
      <dialog ref={dialogo} className="dialogo-apagar" aria-labelledby={`desc-${conexaoId}`}>
        <h2 id={`desc-${conexaoId}`}>Desconectar {banco}?</h2>
        <p>Os lançamentos já trazidos ficam. O app para de buscar novos, e a autorização no Pluggy é revogada.</p>
        {erroDialogo && <p className="erro" role="alert">{erroDialogo}</p>}
        <div className="dialogo-acoes">
          <button type="button" className="link" onClick={() => dialogo.current?.close()}>Cancelar</button>
          <button type="button" className="perigo" onClick={() => void confirmarDesconectar()}>Desconectar</button>
        </div>
      </dialog>
    </div>
  );
}
```
