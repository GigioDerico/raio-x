- [ ] **Step 3: `src/lib/ia-uso-repo.ts`**

```ts
import 'server-only';
import { and, eq, gte, sql } from 'drizzle-orm';
import { db } from '@/db';
import { aiUso } from '@/db/schema';
import type { Uso } from './openrouter';

/**
 * Contabilidade, não dado do usuário: se a gravação falhar, a importação e o
 * plano seguem — perder uma linha de custo é melhor que perder a importação.
 */
export async function registrarUso(userId: string, kind: 'extracao' | 'plano', modelo: string, uso: Uso) {
  try {
    await db.insert(aiUso).values({
      userId,
      kind,
      modelo,
      tokensIn: uso.tokensIn,
      tokensOut: uso.tokensOut,
      custoUsd: uso.custoUsd === null ? null : uso.custoUsd.toFixed(6),
    });
  } catch (e) {
    console.error('[ai_uso]', e);
  }
}

export type GastoDoMes = { gasto: number; chamadas: number; semCusto: number };

export async function gastoDoMes(userId: string): Promise<GastoDoMes> {
  // o mês é o do dono, em São Paulo — não o do servidor em UTC
  const inicio = sql`(date_trunc('month', now() at time zone 'America/Sao_Paulo') at time zone 'America/Sao_Paulo')`;
  const [linha] = await db
    .select({
      gasto: sql<string>`coalesce(sum(${aiUso.custoUsd}), 0)`,
      chamadas: sql<number>`count(*)::int`,
      semCusto: sql<number>`count(*) filter (where ${aiUso.custoUsd} is null)::int`,
    })
    .from(aiUso)
    .where(and(eq(aiUso.userId, userId), gte(aiUso.criadoEm, inicio)));
  return { gasto: Number(linha?.gasto ?? 0), chamadas: linha?.chamadas ?? 0, semCusto: linha?.semCusto ?? 0 };
}
```

- [ ] **Step 4: `src/lib/ia-gasto.ts`**

```ts
/** Gasto de IA do mês contra o teto. Puro: a consulta fica em ia-uso-repo.ts. */
import type { GastoDoMes } from './ia-uso-repo';

export type ResumoGasto = GastoDoMes & { teto: number; pct: number | null; acima: boolean };

/** `teto` 0 = sem teto: nada de barra nem de aviso. */
export function resumoGasto(g: GastoDoMes, teto: number): ResumoGasto {
  if (!teto) return { ...g, teto: 0, pct: null, acima: false };
  return { ...g, teto, pct: Math.min(100, (g.gasto / teto) * 100), acima: g.gasto > teto };
}
```

```ts
/**
 * CSV de lançamentos no formato do app antigo (`;`, BOM, vírgula decimal):
 * abre direto no Excel em português. Puro, para o teste.
 */
export type LinhaCsv = {
  mes: string;
  banco: string;
  cartao: string | null;
  data: string;
  desc: string;
  apelido: string | null;
  categoria: string | null;
  parcelaN: number | null;
  totalParcelas: number | null;
  tipo: string;
  valor: number;
  cartaoApelido: string | null;
  fonte: string;
};

const COLUNAS = [
  'mes_fatura', 'banco', 'cartao', 'data', 'estabelecimento_na_fatura', 'meu_nome',
  'categoria', 'parcela', 'tipo', 'valor', 'cartao_apelido', 'fonte',
];

// texto vindo de fatura é de terceiros: "=HYPERLINK(...)" viraria fórmula no Excel
const guarda = (s: string) => (/^[=+\-@\t\r]/.test(s) ? `'${s}` : s);
const celula = (s: string) => `"${s.replace(/"/g, '""')}"`;
const texto = (s: string | null) => celula(guarda(s ?? ''));

export function montarCsv(linhas: readonly LinhaCsv[]): string {
  const corpo = linhas.map((l) =>
    [
      texto(l.mes),
      texto(l.banco),
      texto(l.cartao),
      texto(l.data),
      texto(l.desc),
      texto(l.apelido || l.desc),
      texto(l.categoria),
      texto(l.parcelaN && l.totalParcelas ? `${l.parcelaN}/${l.totalParcelas}` : ''),
      texto(l.tipo),
      // número não passa pela guarda: o sinal de menos de um estorno é o valor
      celula(l.valor.toFixed(2).replace('.', ',')),
      texto(l.cartaoApelido),
      texto(l.fonte),
    ].join(';'),
  );
  return '﻿' + [COLUNAS.map(celula).join(';'), ...corpo].join('\r\n');
}
```

import 'server-only';
import { and, asc, eq, isNull } from 'drizzle-orm';
import { db } from '@/db';
import { account, category, installmentPlan, institution, statement, txn } from '@/db/schema';
import type { LinhaCsv } from './csv';
import { mesDaTxn } from './txn-repo';

/** Todos os cartões e meses, sem excluídos nem suspeitos — o filtro da tela não vale aqui. */
export async function linhasParaCsv(userId: string): Promise<LinhaCsv[]> {
  const linhas = await db
    .select({
      mes: mesDaTxn,
      banco: institution.nome,
      cartao: account.last4,
      data: txn.data,
      desc: txn.descOrigem,
      apelido: txn.apelido,
      categoria: category.nome,
      parcelaN: txn.parcelaN,
      totalParcelas: installmentPlan.totalParcelas,
      tipo: txn.tipo,
      valor: txn.valor,
      cartaoApelido: account.apelido,
      fonte: txn.fonte,
    })
    .from(txn)
    .innerJoin(account, and(eq(account.id, txn.accountId), eq(account.userId, userId)))
    .innerJoin(institution, eq(institution.id, account.institutionId))
    .leftJoin(statement, and(eq(statement.id, txn.statementId), eq(statement.accountId, txn.accountId)))
    .leftJoin(category, eq(category.id, txn.categoryId))
    .leftJoin(installmentPlan, and(eq(installmentPlan.id, txn.installmentId), eq(installmentPlan.accountId, txn.accountId)))
    .where(and(eq(txn.userId, userId), isNull(txn.excluidoEm), eq(txn.suspeitoDuplicado, false)))
    .orderBy(asc(mesDaTxn), asc(institution.nome), asc(txn.data), asc(txn.id));
  return linhas.map((l) => ({ ...l, valor: Number(l.valor) }));
}

 * o histórico de sync. Usuário, senha e sessão ficam — o dono cai no app vazio.

export async function apagarTudo(userId: string): Promise<{ conexoesApagadas: number }> {
  const conexoes = await db.select({ id: connection.id }).from(connection).where(eq(connection.userId, userId));
  await db.batch([
    db.delete(account).where(eq(account.userId, userId)),
    db.delete(txn).where(eq(txn.userId, userId)),
    db.delete(learnedLabel).where(eq(learnedLabel.userId, userId)),

import { apagarTudo } from '@/lib/dados-repo';

import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';

export const runtime = 'nodejs';

export async function POST(req: NextRequest) {

export default function ApagarTudo() {
  const router = useRouter();
  const dialogo = useRef<HTMLDialogElement>(null);
  const [texto, setTexto] = useState('');

    setApagando(true);
    setErro('');
    try {
      const r = await fetch('/api/dados/apagar', {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ confirmacao: texto }),
      });
      const d = await r.json().catch(() => ({}));
      if (!r.ok) {
        setErro(r.status === 400 ? d?.detalhe || 'confirmação inválida' : 'não consegui apagar, tente de novo');

        return;
      }
      router.push('/');
      router.refresh();

      setErro('não consegui apagar, tente de novo');
    } finally {
      setApagando(false);
    }
  }

  return (
    <>
      <button
        type="button"
        className="link perigo-texto"
        onClick={() => {
          setTexto('');
          setErro('');
          dialogo.current?.showModal();
        }}
      >
        Apagar tudo…
      </button>
