**Goal:** Entregar `/parcelas` com a projeção da próxima fatura (pelo padrão e o que caberia na renda) em cima e todas as compras parceladas em aberto embaixo.

**Architecture:** `projecao` e `planoSaudavel` são portadas linha a linha do legado, com paridade no harness `vm` existente. A página server-side carrega o contexto pelo repositório existente — que passa a trazer renda, meta e gasto fora do cartão — e renderiza três componentes servidor. Nenhuma migração, rota de API ou dependência nova.

**Tech Stack:** Next.js 15 App Router, React 19, TypeScript, Drizzle; testes sem framework em `scripts/testes.ts`; paridade em `scripts/paridade.ts`.

```ts
export type Projecao = {
  prox: string;
  parcelas: number;
  fixos: number;
  variavel: number;
  total: number;
  /** total já importado do mês `prox`; null quando ele ainda não chegou */
  real: number | null;
  meses: number;
};

export function projecao(ctx: Contexto): Projecao | null {
  const ms = mesesDisponiveis(ctx);
  const m = mesAtual(ctx) || (ms.length ? ms[ms.length - 1]! : null);
  if (!m) return null;
  const prox = addMonths(m, 1);
  const pi = parcelasInfo(ctx);
  const parcelas = pi.porMes.find((x) => x[0] === prox)?.[1].tot || 0;
  const fixasArr = recorrentesFixas(ctx);
  const fixas = new Set(fixasArr.map((r) => normDesc(r.desc)));
  const fixos = fixasArr.reduce((s, r) => s + r.medio, 0);
  const base = ms.filter((x) => x <= m).slice(-3);
  const vals = base.map((x) => composicaoMes(ctx, x, fixas).escolha);
  const variavel = vals.length ? vals.reduce((a, b) => a + b, 0) / vals.length : 0;
  const real = ms.includes(prox) ? porMes(ctx).find((x) => x[0] === prox)?.[1] || 0 : null;
  return { prox, parcelas, fixos, variavel, total: parcelas + fixos + variavel, real, meses: base.length };
}

/**
 * Porta do "cenário saudável" do app antigo (docs/legado/app-antigo.html,
 * `planoSaudavel`). O resto do plano de 3 meses entra aqui na fatia 3.5.
 * Puro, sem `server-only`: o teste importa direto.

import type { Contexto } from './base';
import type { Projecao } from './parcelas';

export type ConfigRenda = Pick<Contexto, 'renda' | 'metaPoupanca' | 'foraCartao'>;

export type PlanoSaudavel = {
  /** o que o cartão pode consumir guardando a meta; null sem renda */
  teto: number | null;
  compromissos: number;
  disponivel: number | null;
  poupanca: number | null;
};

export function planoSaudavel(cfg: ConfigRenda, pj: Projecao | null): PlanoSaudavel | null {
  if (!pj) return null;
  const { renda, metaPoupanca, foraCartao } = cfg;
  const teto = renda > 0 ? Math.max(0, renda * (1 - metaPoupanca / 100) - foraCartao) : null;
  const compromissos = pj.parcelas + pj.fixos;
  return {
    teto,
    compromissos,
    disponivel: teto === null ? null : teto - compromissos,
    poupanca: renda > 0 ? (renda * metaPoupanca) / 100 : null,
  };
}
```

/**
 * Números da tela Previsto, as mesmas contas de `renderProjecao` e
 * `renderPrevisto` do app antigo (docs/legado/app-antigo.html, :1380-1399 e
 * :1482-1544), separadas do HTML para serem testadas. As frases ficam no
 * componente; aqui só o que decide qual frase e qual cor.
 */
import type { Contexto } from './base';
import { projecao, type Projecao } from './parcelas';
import { planoSaudavel, type ConfigRenda } from './plano';

export type Veredito = 'compromissos-acima' | 'cabe' | 'cortar';

export type ColunaRenda = {
  teto: number;
  compromissos: number;
  disponivel: number;
  poupanca: number;
  /** teto - total: folga quando o padrão cabe */
  sobra: number;
  /** o que sobra para as escolhas do mês, nunca negativo */
  paraEscolhas: number;
  corte: number;
  pctCorte: number;
  porSemana: number | null;
  porDia: number | null;
  larguras: { contratado: number; paraEscolhas: number; vazio: number };
  estado: 'ok' | 'alerta' | 'ruim';
  veredito: Veredito;
};

export type Previsto = Projecao & {
  dif: number | null;
  pctRenda: number | null;
  padrao: { parcelas: number; fixos: number; variavel: number; vazio: number };
  renda: ColunaRenda | null;
};

export function montarPrevisto(pj: Projecao | null, cfg: ConfigRenda): Previsto | null {
  if (!pj || !pj.total) return null;
  const pl = planoSaudavel(cfg, pj)!;
  // a mesma escala nas duas barras: é o que deixa ver o tamanho do ajuste
  const escala = Math.max(pj.total, pl.teto || 0, 1);
  const larg = (v: number) => (v / escala) * 100;

  let renda: ColunaRenda | null = null;
  if (pl.teto !== null) {
    const disp = pl.disponivel!;
    const paraEscolhas = Math.max(0, disp);
    const corte = pj.variavel - paraEscolhas;
    const estado = pj.total <= pl.teto ? 'ok' : pj.total <= pl.teto * 1.1 ? 'alerta' : 'ruim';
    renda = {
      teto: pl.teto,
      compromissos: pl.compromissos,
      disponivel: disp,
      poupanca: pl.poupanca!,
      sobra: pl.teto - pj.total,
      paraEscolhas,
      corte,
      pctCorte: pj.variavel ? corte / pj.variavel : 0,
      porSemana: disp > 0 ? disp / 4.3 : null,
      porDia: disp > 0 ? disp / 30 : null,
      larguras: {
        contratado: larg(Math.min(pl.compromissos, pl.teto)),
        paraEscolhas: larg(paraEscolhas),
        vazio: Math.max(0, larg(escala - pl.teto)),
      },
      estado: disp < 0 ? 'ruim' : estado,
      veredito: disp < 0 ? 'compromissos-acima' : pj.total <= pl.teto ? 'cabe' : 'cortar',
    };
  }

  return {
    ...pj,
    dif: pj.real !== null ? pj.real / pj.total - 1 : null,
    pctRenda: cfg.renda > 0 ? pj.total / cfg.renda : null,
    padrao: {
      parcelas: larg(pj.parcelas),
      fixos: larg(pj.fixos),
      variavel: larg(pj.variavel),
      vazio: Math.max(0, larg(escala - pj.total)),
    },
    renda,
  };
}

export function previsto(ctx: Contexto): Previsto | null {
  return montarPrevisto(projecao(ctx), ctx);
}
