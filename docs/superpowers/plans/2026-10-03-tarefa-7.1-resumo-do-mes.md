# Tarefa 7.1 — Resumo do mês por IA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Um cartão no painel com 2 ou 3 frases da IA sobre o recorte da tela, gerado só por botão, guardado em `ai_output` e marcado "desatualizado" quando os dados mudam.

**Architecture:** `dadosResumo` (porta do legado), `chaveRecorte` e `lerResumo` são puros; `promptResumo` fica em `ia.ts`. A rota `POST /api/resumo` monta o prompt, devolve o guardado quando o hash do prompt não mudou e só então chama a IA. O painel calcula o mesmo hash (só conta) para decidir entre atual, desatualizado e nenhum.

**Tech Stack:** Next.js 15 App Router, React 19, TypeScript, Drizzle (Neon HTTP), OpenRouter via `pedirJson`; testes em `scripts/testes.ts`.

**Spec:** `docs/superpowers/specs/2026-10-03-resumo-do-mes-design.md`. Legado: `docs/legado/app-antigo.html:1128-1244`. Acompanhamento: `docs/acompanhamento/7.1-resumo-do-mes.md`.

## Global Constraints

- Regra 9: nenhuma chamada de modelo por navegação. O painel só **calcula** o prompt e o hash; a IA só é chamada pela rota, no clique.
- Regra 4: o prompt passa por `ia.ts` e não leva last4, id, saldo, limite nem nome de titular.
- `user_id` só da sessão; `ai_output` lido e gravado com `user_id` no `where`.
- Resposta da IA é dado não confiável: só `texto` string, aparado, até 600 caracteres.
- Sem migração. Não rodar `npm run build` no diretório principal nem `npm run db:migrate`.

/**
 * Dados do resumo do mês: porta de `dadosResumo` do app antigo
 * (docs/legado/app-antigo.html:1132-1148) sobre o Contexto já filtrado por
 * cartão e obra. Puro, sem `server-only`: o teste importa direto.
 */
import { totalFatura, type Contexto } from './base';
import { cartoesInfo, lancsDoMes, mesAnterior, mesAtual, porCategoria, porMes, type LancDoMes } from './meses';
import type { Filtro } from '../filtro';

export type DadosResumo = {
  m: string | null;
  prevM: string | null;
  linhas: LancDoMes[];
  total: number;
  prevTot: number;
  cats: [string, number][];
  prevCats: Map<string, number> | null;
  porCartao: { nome: string; total: number }[];
};

export function dadosResumo(ctx: Contexto): DadosResumo {
  const m = mesAtual(ctx);
  const linhas = lancsDoMes(ctx, m);
  const total = linhas.reduce((s, l) => s + (+l.val || 0), 0);
  const cats = [...porCategoria(ctx, m).entries()].filter(([, v]) => v > 0).sort((a, b) => b[1] - a[1]);
  const prevM = m ? mesAnterior(ctx, m) : null;
  const prevCats = prevM ? porCategoria(ctx, prevM) : null;
  const prevTot = prevM ? porMes(ctx).find((x) => x[0] === prevM)?.[1] || 0 : 0;
  // o legado juntava "••1234" ao nome do cartão; o prompt não leva os quatro
  // dígitos (regra 4), então fica só o apelido ou o banco
  const porCartao = cartoesInfo(ctx)
    .filter((c) => c.faturas.some((f) => !m || f.mesRef === m))
    .map((c) => ({
      nome: c.nome,
      total: c.faturas.filter((f) => !m || f.mesRef === m).reduce((t, f) => t + totalFatura(f), 0),
    }))
    .filter((c) => c.total)
    .sort((a, b) => b.total - a.total);
  return { m, prevM, linhas, total, prevTot, cats, prevCats, porCartao };
}

/** O resumo vale para o recorte da tela: mês resolvido, cartões e obra. */
export function chaveRecorte(filtro: Pick<Filtro, 'cartoes' | 'obra'>, mes: string | null): string {
  const cartoes = filtro.cartoes.length ? [...filtro.cartoes].sort().join(',') : 'todos';
  return `${mes ?? 'todos'}|${cartoes}|${filtro.obra ? 1 : 0}`;
}

/** A resposta da IA é dado não confiável: só um texto, aparado e com limite. */
export function lerResumo(dados: unknown): string | null {
  if (!dados || typeof dados !== 'object') return null;
  const texto = (dados as { texto?: unknown }).texto;
  if (typeof texto !== 'string') return null;
  const t = texto.trim();
  return t ? t.slice(0, 600) : null;
}

/**
 * Prompt do resumo do mês, o texto do app antigo (app-antigo.html:1150-1180).
 * Vai: totais, categorias, composição, parcelas, fixos e os 6 maiores
 * lançamentos pelo nome exibido. Não vai: os quatro dígitos do cartão (o
 * legado mandava), id, saldo, limite. A resposta é pedida em JSON para usar
 * `pedirJson`, com o modelo escolhido e o registro de custo.
 */
export function promptResumo(ctx: Contexto, d: DadosResumo): string {
  const L: string[] = [];
  L.push('Mês da fatura: ' + (d.m ? rotuloMes(d.m) : 'todos os meses carregados'));
  L.push('Cartões: ' + (d.porCartao.map((c) => `${c.nome} ${moeda(c.total)}`).join(', ') || 'nenhum'));
  L.push(
    'Total: ' +
      moeda(d.total) +
      (d.prevTot && d.prevM ? ` · mês anterior (${rotuloMes(d.prevM)}): ${moeda(d.prevTot)} (${pct(d.total / d.prevTot - 1)})` : ''),
  );
  L.push(
    'Categorias: ' +
      d.cats
        .slice(0, 7)
        .map(([c, v]) => {
          const p = d.prevCats ? d.prevCats.get(c) || 0 : 0;
          return `${c} ${moeda(v)}` + (p > 0 ? ` (${pct(v / p - 1)})` : ' (novo)');
        })
        .join('; '),
  );
  const enc = encargosDoMes(ctx, d.m);
  if (enc > 0) L.push('Encargos e juros no mês: ' + moeda(enc));
  const comp = composicaoMes(ctx, d.m);
  L.push(`Já contratado antes do mês (fixos+parcelas+encargos): ${moeda(comp.comprometido)} · decidido no mês: ${moeda(comp.escolha)}`);
  const pj = projecao(ctx);
  if (pj && !pj.real) L.push('Próximo mês já começa em: ' + moeda(pj.total));
  const pi = parcelasInfo(ctx);
  if (pi.total > 0) L.push(`Parcelas a vencer: ${moeda(pi.total)} em ${pi.porMes.length} meses`);
  const fixas = recorrentesFixas(ctx);
  if (fixas.length) L.push('Fixos por mês: ' + moeda(fixas.reduce((s, r) => s + r.medio, 0)));
  const top = d.linhas
    .filter((l) => l.val > 0)
    .sort((a, b) => b.val - a.val)
    .slice(0, 6);
  L.push('Maiores lançamentos: ' + top.map((l) => `${nomeExibido(ctx, l)} ${moeda(l.val)}`).join(', '));
  return `Você resume faturas de cartão de crédito para o dono do cartão, em português do Brasil.

Escreva 2 ou 3 frases curtas contando o que aconteceu neste mês: o tamanho da fatura, o que puxou o gasto e a mudança mais relevante em relação ao mês anterior. Termine apontando o ponto que merece atenção.
Use os números que importam, sem repetir todos. Sem listas, sem títulos, sem saudação, sem conselhos genéricos. Máximo 55 palavras.

DADOS:
${L.join('\n')}

Responda SOMENTE com JSON: {"texto": "<o resumo>"}`;
}

```ts
import 'server-only';
import { createHash } from 'node:crypto';
import { and, desc, eq, sql } from 'drizzle-orm';
import { db } from '@/db';
import { aiOutput } from '@/db/schema';
import { lerResumo } from './calculos/resumo';
import type { Uso } from './openrouter';

// o hash do próprio prompt é a assinatura: muda com qualquer dado que a IA veria
export const hashDoPrompt = (prompt: string) => createHash('sha256').update(prompt).digest('hex');

export async function resumoGuardado(userId: string, escopo: string) {
  const [linha] = await db
    .select({ conteudo: aiOutput.conteudo, inputHash: aiOutput.inputHash })
    .from(aiOutput)
    .where(and(eq(aiOutput.userId, userId), eq(aiOutput.kind, 'resumo'), eq(aiOutput.escopo, escopo)))
    .orderBy(desc(aiOutput.criadoEm))
    .limit(1);
  if (!linha) return null;
  const texto = lerResumo(linha.conteudo);
  return texto ? { texto, inputHash: linha.inputHash } : null;
}

export async function guardarResumo(userId: string, escopo: string, inputHash: string, texto: string, modelo: string, uso: Uso) {
  const custoUsd = uso.custoUsd === null ? null : uso.custoUsd.toFixed(6);
  await db
    .insert(aiOutput)
    .values({ userId, kind: 'resumo', escopo, modelo, inputHash, conteudo: { texto }, tokensIn: uso.tokensIn, tokensOut: uso.tokensOut, custoUsd })
    // "↻" com o mesmo prompt bate na chave única (user, kind, escopo, input_hash)
    .onConflictDoUpdate({
      target: [aiOutput.userId, aiOutput.kind, aiOutput.escopo, aiOutput.inputHash],
      set: { conteudo: { texto }, modelo, tokensIn: uso.tokensIn, tokensOut: uso.tokensOut, custoUsd, criadoEm: sql`now()` },
    });
}
```

```ts
import type { NextRequest } from 'next/server';
import { chaveRecorte, dadosResumo, lerResumo } from '@/lib/calculos/resumo';
import { lerFiltro } from '@/lib/filtro';
import { promptResumo } from '@/lib/ia';
import { registrarUso } from '@/lib/ia-uso-repo';
import { RespostaNaoJson, SemChaveIA, pedirJson } from '@/lib/openrouter';
import { carregarContexto } from '@/lib/painel-repo';
import { guardarResumo, hashDoPrompt, resumoGuardado } from '@/lib/resumo-repo';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';
import { configuracoes } from '@/lib/sessao-servidor';

export const runtime = 'nodejs';
// espera o modelo, e pedirJson pode tentar o de reserva
export const maxDuration = 60;

const iaInvalida = () => new ErroDeUso('ia_invalida', 'A IA não respondeu direito; tente de novo', 502);

export async function POST(req: NextRequest) {
  const filtro = lerFiltro(req.nextUrl.searchParams);

  return comUsuario(async (userId) => {
    const corpo = await corpoJson<unknown>(req);
    const forcar = !!corpo && typeof corpo === 'object' && (corpo as { forcar?: unknown }).forcar === true;

    const { ctx } = await carregarContexto(userId, filtro);
    const d = dadosResumo(ctx);
    if (!d.linhas.length) throw new ErroDeUso('sem_lancamentos', 'Não há lançamentos neste recorte');

    const prompt = promptResumo(ctx, d);
    const hash = hashDoPrompt(prompt);
    const escopo = chaveRecorte(filtro, d.m);

    if (!forcar) {
      // dado igual, resumo igual: não paga de novo pelo mesmo texto
      const guardado = await resumoGuardado(userId, escopo);
      if (guardado && guardado.inputHash === hash) return { texto: guardado.texto };
    }

    const cfg = await configuracoes(userId);
    let resposta: Awaited<ReturnType<typeof pedirJson<unknown>>>;
    try {
      resposta = await pedirJson<unknown>(prompt, { modelo: cfg?.modeloIa });
    } catch (e) {
      if (e instanceof SemChaveIA) throw new ErroDeUso('sem_chave', 'Configure OPENROUTER_API_KEY');
      if (e instanceof RespostaNaoJson) {
        if (e.modelo && e.uso) await registrarUso(userId, 'resumo', e.modelo, e.uso);
        throw iaInvalida();
      }
      throw e;
    }
    await registrarUso(userId, 'resumo', resposta.modelo, resposta.uso);

    const texto = lerResumo(resposta.dados);
    if (!texto) throw iaInvalida();
    await guardarResumo(userId, escopo, hash, texto, resposta.modelo, resposta.uso);
    return { texto };
  });
}
```

```tsx
'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';

type Estado = 'atual' | 'desatualizado' | 'nenhum';
type Props = { titulo: string; texto: string | null; estado: Estado; query: string };

export default function ResumoMes({ titulo, texto, estado, query }: Props) {
  const router = useRouter();
  const [pedindo, setPedindo] = useState(false);
  const [erro, setErro] = useState('');

  async function pedir(forcar: boolean) {
    setPedindo(true);
    setErro('');
    try {
      const r = await fetch(`/api/resumo?${query}`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ forcar }),
      });
      const d = await r.json().catch(() => ({}));
      if (!r.ok) {
        // 400 e 502 trazem mensagem escrita para o usuário ("configure a chave", "não respondeu direito")
        setErro([400, 502].includes(r.status) ? d?.detalhe || 'não deu certo' : 'não consegui falar com a IA, tente de novo');
        return;
      }
      router.refresh();
    } catch {
      setErro('não consegui falar com a IA, tente de novo');
    } finally {
      setPedindo(false);
    }
  }

  return (
    <section className="cartao resumo-mes" aria-labelledby="resumo-titulo">
      <div className="resumo-topo">
        <span id="resumo-titulo" className="eyebrow">{titulo}</span>
        {estado === 'atual' && (
          <button type="button" className="link resumo-refazer" onClick={() => pedir(true)} disabled={pedindo} aria-label="Gerar o resumo de novo" title="Gerar o resumo de novo">
            ↻
          </button>
        )}
      </div>
      {pedindo ? (
        <p className="resumo-texto suave">Resumindo o mês…</p>
      ) : texto ? (
        <p className="resumo-texto">{texto}</p>
      ) : null}
      {estado === 'desatualizado' && !pedindo && <p className="resumo-nota">Os dados mudaram desde este resumo.</p>}
      {estado !== 'atual' && (
        <button type="button" className="resumo-pedir" onClick={() => pedir(false)} disabled={pedindo}>
          {estado === 'nenhum' ? 'Resumir o mês' : 'Refazer'}
        </button>
      )}
      <span className="resumo-erro" role="status" aria-live="polite">{erro}</span>
    </section>
  );
}
```
