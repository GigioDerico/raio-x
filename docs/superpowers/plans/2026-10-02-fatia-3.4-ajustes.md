```ts
/**
 * Validação e formatação dos campos de Ajustes. Devolve resultado em vez de
 * lançar `ErroDeUso`: ele mora em rota.ts, que é `server-only`, e este módulo
 * precisa ser importável pelos testes.
 */
import { parseValor } from './descricao';
import { rotuloMes } from './filtro';

export type Validado<T> = { ok: true; mudanca: T } | { ok: false; codigo: string; mensagem: string };
export type MudancaConfig = { renda?: number | null; metaPoupanca?: number; foraCartao?: number };
export type MudancaConta = { apelido?: string | null; diaFechamento?: number | null; diaVencimento?: number | null };
export type TipoCampo = 'dinheiro' | 'percentual' | 'dia' | 'texto';

export type CartaoAjuste = {
  id: string;
  banco: string;
  last4: string | null;
  apelido: string | null;
  nomeOrigem: string | null;
  diaFechamento: number | null;
  diaVencimento: number | null;
  faturas: number;
  /** 'YYYY-MM-DD' de statement.mes_ref; null sem fatura */
  primeiroMes: string | null;
  ultimoMes: string | null;
};

const erro = (codigo: string, mensagem: string) => ({ ok: false as const, codigo, mensagem });
const vazio = (v: unknown) => v === null || (typeof v === 'string' && v.trim() === '');
const objeto = (c: unknown): Record<string, unknown> => (c && typeof c === 'object' ? (c as Record<string, unknown>) : {});

function lerNumero(v: unknown): number {
  if (typeof v === 'number') return Number.isFinite(v) ? Math.round(v * 100) / 100 : NaN;
  const t = String(v).trim().replace(/[R$\s]/g, '');
  // `parseValor` usa parseFloat, que leria "12abc" como 12
  if (!/^-?[\d.,]+$/.test(t)) return NaN;
  // "8.000" é oito mil para quem digita em reais; parseValor leria 8
  const n = parseValor(/^\d{1,3}(\.\d{3})+$/.test(t) ? t.replace(/\./g, '') : t);
  return Math.round(n * 100) / 100;
}

export function validarConfig(corpo: unknown): Validado<MudancaConfig> {
  const c = objeto(corpo);
  const m: MudancaConfig = {};
  if ('renda' in c) {
    if (vazio(c.renda)) m.renda = null;
    else {
      const n = lerNumero(c.renda);
      if (Number.isNaN(n)) return erro('valor_invalido', 'renda não é um número');
      if (n < 0) return erro('renda_negativa', 'renda não pode ser negativa');
      m.renda = n;
    }
  }
  if ('metaPoupanca' in c) {
    const n = lerNumero(c.metaPoupanca);
    if (Number.isNaN(n)) return erro('valor_invalido', 'meta não é um número');
    if (n < 0 || n > 80) return erro('meta_fora', 'meta entre 0 e 80%');
    m.metaPoupanca = n;
  }
  if ('foraCartao' in c) {
    // a coluna é `not null default 0`: apagar o campo é dizer "nada fora do cartão"
    if (vazio(c.foraCartao)) m.foraCartao = 0;
    else {
      const n = lerNumero(c.foraCartao);
      if (Number.isNaN(n)) return erro('valor_invalido', 'gasto fora do cartão não é um número');
      if (n < 0) return erro('fora_negativo', 'gasto fora do cartão não pode ser negativo');
      m.foraCartao = n;
    }
  }
  if (Object.keys(m).length === 0) return erro('sem_mudanca', 'nenhum campo para salvar');
  return { ok: true, mudanca: m };
}

export function validarConta(corpo: unknown): Validado<MudancaConta> {
  const c = objeto(corpo);
  const m: MudancaConta = {};
  if ('apelido' in c) {
    if (c.apelido === null) m.apelido = null;
    else if (typeof c.apelido !== 'string') return erro('apelido_invalido', 'apelido precisa ser texto');
    else {
      const t = c.apelido.trim();
      if (t.length > 40) return erro('apelido_longo', 'apelido até 40 caracteres');
      m.apelido = t || null;
    }
  }
  for (const campo of ['diaFechamento', 'diaVencimento'] as const) {
    if (!(campo in c)) continue;
    const v = c[campo];
    if (vazio(v)) {
      m[campo] = null;
      continue;
    }
    const n = typeof v === 'number' ? v : /^\s*\d{1,2}\s*$/.test(String(v)) ? Number(v) : NaN;
    if (!Number.isInteger(n) || n < 1 || n > 31) return erro('dia_invalido', 'dia entre 1 e 31');
    m[campo] = n;
  }
  if (Object.keys(m).length === 0) return erro('sem_mudanca', 'nenhum campo para salvar');
  return { ok: true, mudanca: m };
}

/** Texto que o campo mostra. Dinheiro zerado fica vazio: o placeholder "0,00" já diz isso. */
export function formatarCampo(tipo: TipoCampo, v: unknown): string {
  if (v === null || v === undefined) return '';
  if (tipo === 'dinheiro') {
    const n = Number(v);
    return n ? n.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) : '';
  }
  if (tipo === 'percentual') return Number(v).toLocaleString('pt-BR', { maximumFractionDigits: 2 });
  return String(v);
}

export function linhaDoCartao(c: CartaoAjuste): string {
  if (!c.faturas || !c.primeiroMes || !c.ultimoMes) return 'nenhuma fatura importada';
  const de = rotuloMes(c.primeiroMes.slice(0, 7));
  const ate = rotuloMes(c.ultimoMes.slice(0, 7));
  const periodo = de === ate ? de : `${de} a ${ate}`;
  return `na fatura: ${c.nomeOrigem || c.banco} · ${c.faturas} ${c.faturas === 1 ? 'fatura' : 'faturas'} · ${periodo}`;
}
```

```ts
import 'server-only';
import { and, asc, eq, sql } from 'drizzle-orm';
import { db } from '@/db';
import { account, institution, statement, userSetting } from '@/db/schema';
import { configDoUsuario } from './calculos/adaptador';
import type { CartaoAjuste, MudancaConfig, MudancaConta } from './ajustes';

const colunasConfig = {
  renda: userSetting.rendaDeclarada,
  meta: userSetting.metaPoupanca,
  fora: userSetting.gastoForaCartao,
};

export async function lerConfig(userId: string) {
  const [linha] = await db.select(colunasConfig).from(userSetting).where(eq(userSetting.userId, userId)).limit(1);
  return configDoUsuario(linha);
}

export async function salvarConfig(userId: string, m: MudancaConfig) {
  const campos: Partial<typeof userSetting.$inferInsert> = {};
  if (m.renda !== undefined) campos.rendaDeclarada = m.renda === null ? null : m.renda.toFixed(2);
  if (m.metaPoupanca !== undefined) campos.metaPoupanca = m.metaPoupanca.toFixed(2);
  if (m.foraCartao !== undefined) campos.gastoForaCartao = m.foraCartao.toFixed(2);
  const [linha] = await db.update(userSetting).set(campos).where(eq(userSetting.userId, userId)).returning(colunasConfig);
  // a linha nasce no primeiro login (obterOuCriarUsuario); faltar é defeito, não erro do usuário
  if (!linha) throw new Error('user_setting ausente');
  return configDoUsuario(linha);
}

/**
 * Posse e tipo no `where`: id de outro dono, de conta corrente ou inexistente
 * dão o mesmo `null`, e nenhum dos três conta qual foi o caso.
 */
export async function salvarConta(userId: string, id: string, m: MudancaConta) {
  const [linha] = await db
    .update(account)
    .set(m)
    .where(and(eq(account.id, id), eq(account.userId, userId), eq(account.kind, 'cartao_credito')))
    .returning({
      id: account.id,
      apelido: account.apelido,
      diaFechamento: account.diaFechamento,
      diaVencimento: account.diaVencimento,
    });
  return linha ?? null;
}

export async function listarCartoes(userId: string): Promise<CartaoAjuste[]> {
  return db
    .select({
      id: account.id,
      banco: institution.nome,
      last4: account.last4,
      apelido: account.apelido,
      nomeOrigem: account.nomeOrigem,
      diaFechamento: account.diaFechamento,
      diaVencimento: account.diaVencimento,
      faturas: sql<number>`count(${statement.id})::int`,
      primeiroMes: sql<string | null>`min(${statement.mesRef})::text`,
      ultimoMes: sql<string | null>`max(${statement.mesRef})::text`,
    })
    .from(account)
    .innerJoin(institution, eq(institution.id, account.institutionId))
    .leftJoin(statement, eq(statement.accountId, account.id))
    .where(and(eq(account.userId, userId), eq(account.kind, 'cartao_credito')))
    .groupBy(account.id, institution.nome)
    .orderBy(asc(institution.nome), asc(account.last4));
}
```

`salvarConta` recebe `MudancaConta` direto no `.set(m)`: as chaves (`apelido`, `diaFechamento`, `diaVencimento`) são exatamente as propriedades do Drizzle, e `validarConta` só devolve essas três. Se o TypeScript reclamar do tipo, montar `const campos: Partial<typeof account.$inferInsert> = { ...m }` antes.

```ts
export async function PATCH(req: NextRequest) {
  return comUsuario(async (userId) => {
    // dentro do comUsuario: JSON inválido tem de virar 400, não 500
    const v = validarConfig(await corpoJson<unknown>(req));
    if (!v.ok) throw new ErroDeUso(v.codigo, v.mensagem);
    return { ok: true, config: await salvarConfig(userId, v.mudanca) };
  });
}
```

```ts
import type { NextRequest } from 'next/server';
import { validarConta } from '@/lib/ajustes';
import { salvarConta } from '@/lib/ajustes-repo';
import { ehUuid } from '@/lib/filtro';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';

export const runtime = 'nodejs';

export async function PATCH(req: NextRequest, ctx: { params: Promise<{ id: string }> }) {
  const { id } = await ctx.params;
  return comUsuario(async (userId) => {
    const v = validarConta(await corpoJson<unknown>(req));
    if (!v.ok) throw new ErroDeUso(v.codigo, v.mensagem);
    // id que não é uuid faria o Postgres estourar (503); tem de ser o mesmo 404 de um id alheio
    const conta = ehUuid(id) ? await salvarConta(userId, id, v.mudanca) : null;
    if (!conta) throw new ErroDeUso('nao_encontrado', 'cartão não encontrado', 404);
    return { ok: true, conta };
  });
}
```

```tsx
'use client';

import { useId, useRef, useState } from 'react';
import { useRouter } from 'next/navigation';
import { formatarCampo, type TipoCampo } from '@/lib/ajustes';

type Props = {
  url: string;
  campo: string;
  rotulo: string;
  tipo: TipoCampo;
  valor: string;
  placeholder?: string;
  /** recarrega os dados do servidor depois de salvar (o apelido aparece no filtro e no hero) */
  atualizarTela?: boolean;
};

type Estado = { k: 'ocioso' | 'salvando' | 'salvo' | 'erro'; msg?: string };

export default function CampoAjuste({ url, campo, rotulo, tipo, valor, placeholder, atualizarTela }: Props) {
  const id = useId();
  const router = useRouter();
  const [texto, setTexto] = useState(valor);
  const gravado = useRef(valor);
  const [estado, setEstado] = useState<Estado>({ k: 'ocioso' });

  async function salvar() {
    if (texto === gravado.current) return;
    setEstado({ k: 'salvando' });
    try {
      const r = await fetch(url, {
        method: 'PATCH',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ [campo]: texto }),
      });
      const d = await r.json().catch(() => ({}));
      if (!r.ok) {
        // 400/404 trazem mensagem escrita para o usuário; o resto é falha nossa
        const msg = r.status === 400 || r.status === 404 ? d?.detalhe || 'valor inválido' : 'não consegui salvar, tente de novo';
        setEstado({ k: 'erro', msg });
        return;
      }
      const novo = formatarCampo(tipo, (d.config ?? d.conta)?.[campo]);
      gravado.current = novo;
      setTexto(novo);
      setEstado({ k: 'salvo' });
      setTimeout(() => setEstado((e) => (e.k === 'salvo' ? { k: 'ocioso' } : e)), 2000);
      if (atualizarTela) router.refresh();
    } catch {
      setEstado({ k: 'erro', msg: 'não consegui salvar, tente de novo' });
    }
  }

  return (
    <div className="campo-ajuste">
      <label htmlFor={id}>{rotulo}</label>
      <input
        id={id}
        type="text"
        inputMode={tipo === 'texto' ? 'text' : tipo === 'dia' ? 'numeric' : 'decimal'}
        value={texto}
        placeholder={placeholder}
        maxLength={tipo === 'texto' ? 40 : 20}
        onChange={(e) => setTexto(e.target.value)}
        onBlur={salvar}
        onKeyDown={(e) => {
          if (e.key === 'Enter') e.currentTarget.blur();
        }}
        aria-invalid={estado.k === 'erro'}
        aria-describedby={`${id}-msg`}
      />
      <span id={`${id}-msg`} className={`campo-msg ${estado.k}`} aria-live="polite">
        {estado.k === 'salvando' ? 'salvando…' : estado.k === 'salvo' ? 'salvo' : estado.k === 'erro' ? estado.msg : ''}
      </span>
    </div>
  );
}
```
