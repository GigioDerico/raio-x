# Fatia 3.0 — arrumação, correções de dados e casca — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deixar docs coerentes, corrigir como a importação grava pagamento e parcela, fazer o mês ser o da fatura em toda parte, e montar a casca do app (menu lateral/inferior + filtro global na URL) com Lançamentos dentro dela.

**Architecture:** Regras novas nascem como funções puras em `src/lib/` (sem `server-only`) e são testadas em `scripts/testes.ts`; o repositório (`txn-repo.ts`) só as usa. A casca é um route group `src/app/(app)/` com `layout.tsx` servidor que busca as opções do filtro e um componente cliente `Casca` que desenha menu, filtro e o conteúdo. A URL é a fonte da verdade do filtro; o middleware completa a URL a partir do cookie `rx_filtro` quando ela chega sem filtro.

**Tech Stack:** Next.js 15.5 (App Router, React 19), Drizzle 0.45 + Neon, TypeScript estrito, testes sem framework (`tsx scripts/testes.ts`).

**Spec:** `docs/superpowers/specs/2026-09-30-fatia-3-design.md` (seções 3, 5 e 9).

/**
 * Regras da importação que não dependem de banco.
 *
 * `txn.tipo` guarda só a ORIGEM do lançamento (parcela, encargo, pagamento ou
 * escolha). Se é "fixo" ou "escolha" no painel é decidido na hora por
 * `classeLanc`, porque depende de quais cobranças se repetiram nos últimos
 * meses — e isso muda a cada fatura importada.
 */
import { somarMeses, type Parcela } from './descricao';
import type { TipoLanc } from './extracao';

export type TipoTxn = 'parcela' | 'encargo' | 'pagamento' | 'escolha';

export function tipoDoTxn(tipo: TipoLanc, parcela: Parcela | null): TipoTxn {
  if (tipo === 'pagamento') return 'pagamento';
  if (tipo === 'encargo') return 'encargo';
  if (parcela) return 'parcela';
  return 'escolha';
}

export type PlanoParcela = { totalParcelas: number; valorParcela: string; primeiraRef: string };

/**
 * A parcela n de uma fatura do mês M foi comprada no mês M − (n − 1). É o que
 * permite reconhecer a mesma compra parcelada em faturas diferentes.
 */
export function planoDaParcela(mesRef: string, parcela: Parcela, valor: number): PlanoParcela {
  return {
    totalParcelas: parcela.total,
    valorParcela: Math.abs(valor).toFixed(2),
    primeiraRef: `${somarMeses(mesRef, -(parcela.n - 1))}-01`,
  };
}
```

 * consertou o que a fatia 2 gravou errado), mas nunca toca no que é seu:
 * apelido, categoria, e o que você excluiu continua excluído.

export async function importarDoPdf(
  userId: string,
  accountId: string,
  statementId: string,
  mesRef: string,
  lancs: readonly LancLimpo[],
): Promise<ResultadoImport> {
  const regras = await regrasAprendidas(userId);

  const preparados = comOrdinais(
    lancs
      .filter((l) => l.d)
      .map((l) => ({
        data: l.d,
        valor: l.tipo === 'estorno' || l.tipo === 'pagamento' ? -Math.abs(l.val) : l.val,
        descNorm: normalizarDescricao(l.desc),
        original: l,

  let atualizados = 0;
  let duplicados = 0;

  for (const p of preparados) {
    const regra = regras.get(chaveAprendizado(p.descNorm));
    const categoryId = regra?.categoryId ?? (await idDaCategoria(p.original.cat));
    const parcela = lerParcela(p.original.parc);
    const tipo = tipoDoTxn(p.original.tipo, parcela);
    const installmentId =
      parcela && tipo === 'parcela'
        ? await planoDeParcela(accountId, p.descNorm, mesRef, parcela, p.valor)
        : null;

    const [linha] = await db
      .insert(txn)
      .values({
        userId,
        accountId,
        statementId,
        data: p.data,
        valor: p.valor.toFixed(2),
        descOrigem: p.original.desc,
        descNorm: p.descNorm,
        apelido: regra?.apelido ?? null,
        categoryId,
        categoryFonte: regra ? 'regra' : 'ia',
        tipo,
        installmentId,
        parcelaN: parcela?.n ?? null,
        fonte: 'pdf',
        dedupeKey: chaveDedupe(accountId, p.data, p.valor, p.descNorm, p.ordinal),

      .returning({ novo: sql<boolean>`(xmax = 0)` });

/**
 * Filtro global: mês, cartões e "separar obra".
 *
 * A URL é a fonte da verdade — é o que faz voltar e favoritar funcionarem e o
 * server component ler o recorte sem estado no cliente. O cookie `rx_filtro`
 * só lembra o último recorte para quando a URL chega sem nenhum.
 *
 * Puro, sem `server-only`: o middleware, as páginas e o componente do filtro
 * usam as mesmas regras.
 */

export type Filtro = {
  /** 'YYYY-MM', 'TODOS', ou null = mês mais recente com dado */
  mes: string | null;
  /** ids de conta; vazio = todos */
  cartoes: string[];
  obra: boolean;
};

export const COOKIE_FILTRO = 'rx_filtro';
const CHAVES = ['mes', 'cartoes', 'obra'] as const;
const MES = /^\d{4}-(0[1-9]|1[0-2])$/;
const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
const MAX_CARTOES = 20;

export function lerFiltro(q: URLSearchParams): Filtro {
  const m = q.get('mes');
  const mes = m === 'TODOS' || (m && MES.test(m)) ? m : null;
  const cartoes = (q.get('cartoes') ?? '')
    .split(',')
    .filter((c) => UUID.test(c))
    .slice(0, MAX_CARTOES);
  return { mes, cartoes, obra: q.get('obra') === '1' };
}

export function paraQuery(f: Filtro): string {
  const q = new URLSearchParams();
  if (f.mes) q.set('mes', f.mes);
  if (f.cartoes.length) q.set('cartoes', f.cartoes.join(','));
  if (f.obra) q.set('obra', '1');
  return q.toString();
}

/** O mês em foco, como `mesAtual()` do legado. null = todos os meses. */
export function resolverMes(mes: string | null, disponiveis: readonly string[]): string | null {
  if (mes === 'TODOS') return null;
  if (mes && disponiveis.includes(mes)) return mes;
  return disponiveis[0] ?? null;
}

/**
 * Se a URL chegou sem filtro e há um recorte lembrado, devolve a nova `search`
 * para redirecionar; senão null.
 */
export function completarComCookie(search: string, cookie: string | undefined): string | null {
  const atual = new URLSearchParams(search);
  if (CHAVES.some((k) => atual.has(k)) || !cookie) return null;
  const lembrado = paraQuery(lerFiltro(new URLSearchParams(cookie)));
  if (!lembrado) return null;
  for (const [k, v] of new URLSearchParams(lembrado)) atual.set(k, v);
  return `?${atual.toString()}`;
}

const MESES = ['jan', 'fev', 'mar', 'abr', 'mai', 'jun', 'jul', 'ago', 'set', 'out', 'nov', 'dez'];

export function rotuloMes(m: string): string {
  return `${MESES[Number(m.slice(5, 7)) - 1]}/${m.slice(2, 4)}`;
}
```

export async function listarTxn(userId: string, f: FiltroTxn = {}) {
  const cond = [eq(txn.userId, userId), isNull(txn.excluidoEm)];

  if (f.mes) cond.push(sql`${mesDaTxn} = ${f.mes}`);
  // os ids vêm do cliente, mas o `user_id` acima já garante que só os seus casam

      data: txn.data,
      mes: mesDaTxn,
      valor: txn.valor,
      descOrigem: txn.descOrigem,

'use client';

import { usePathname, useRouter, useSearchParams } from 'next/navigation';
import { COOKIE_FILTRO, lerFiltro, paraQuery, resolverMes, rotuloMes, type Filtro } from '@/lib/filtro';
import type { OpcoesFiltro } from '@/lib/txn-repo';

type Props = OpcoesFiltro & { iniciarTransicao: (fn: () => void) => void };

export default function FiltroGlobal({ meses, cartoes, iniciarTransicao }: Props) {
  const router = useRouter();
  const caminho = usePathname();
  const params = useSearchParams();
  const f = lerFiltro(params);
  const mesEmFoco = f.mes === 'TODOS' ? 'TODOS' : (resolverMes(f.mes, meses) ?? '');

  function aplicar(novo: Filtro) {
    const filtro = paraQuery(novo);
    // o resto da URL (busca, categoria) é da tela, não do filtro: fica como está
    const q = new URLSearchParams(params);
    for (const k of ['mes', 'cartoes', 'obra']) q.delete(k);
    for (const [k, v] of new URLSearchParams(filtro)) q.set(k, v);
    document.cookie = `${COOKIE_FILTRO}=${encodeURIComponent(filtro)}; path=/; max-age=31536000; samesite=lax`;
    const s = q.toString();
    iniciarTransicao(() => router.replace(s ? `${caminho}?${s}` : caminho, { scroll: false }));
  }

  function alternarCartao(id: string) {
    const tem = f.cartoes.includes(id);
    aplicar({ ...f, cartoes: tem ? f.cartoes.filter((c) => c !== id) : [...f.cartoes, id] });
  }

  const rotuloCartoes =
    f.cartoes.length === 0
      ? 'Todos os cartões'
      : f.cartoes.length === 1
        ? (cartoes.find((c) => c.id === f.cartoes[0])?.nome ?? '1 cartão')
        : `${f.cartoes.length} cartões`;

  return (
    <div className="filtro-global" role="group" aria-label="Recorte">
      <select
        aria-label="Mês"
        value={mesEmFoco}
        onChange={(e) => aplicar({ ...f, mes: e.target.value })}
        disabled={meses.length === 0}
      >
        {meses.map((m) => (
          <option key={m} value={m}>
            {rotuloMes(m)}
          </option>
        ))}
        <option value="TODOS">Todos os meses</option>
      </select>

      {cartoes.length > 1 && (
        <details className="filtro-cartoes">
          <summary>{rotuloCartoes}</summary>
          <div className="filtro-cartoes-lista">
            {cartoes.map((c) => (
              <label key={c.id}>
                <input type="checkbox" checked={f.cartoes.includes(c.id)} onChange={() => alternarCartao(c.id)} />
                {c.nome}
              </label>
            ))}
            {f.cartoes.length > 0 && (
              <button type="button" className="link" onClick={() => aplicar({ ...f, cartoes: [] })}>
                Todos
              </button>
            )}
          </div>
        </details>
      )}

      <label className="filtro-obra">
        <input type="checkbox" checked={f.obra} onChange={(e) => aplicar({ ...f, obra: e.target.checked })} />
        separar obra
      </label>
    </div>
  );
}
