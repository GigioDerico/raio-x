# Fatia 3.1 — painel do mês — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** O painel (`/`) mostra os números do mês — Quanto foi, Quanto era escolha, Para onde foi — calculados pelas funções do app antigo portadas para TypeScript, com teste de paridade contra o próprio legado, e o interruptor "separar obra" passa a valer.

**Architecture:** As funções de `docs/legado/calculos-do-app-antigo.js` são portadas linha a linha para `src/lib/calculos/` (puras, sem banco, sem `server-only`), recebendo um `Contexto` no lugar do estado global. Um adaptador puro monta o formato do legado (`faturas[]` com `lancamentos[]`) a partir das linhas do banco; `src/lib/painel-repo.ts` faz a consulta. O teste de paridade roda o arquivo legado num `vm` do Node e compara a saída das duas versões sobre as mesmas faturas.

**Tech Stack:** Next.js 15.5 (App Router, React 19), Drizzle 0.45 + Neon, TypeScript estrito, testes sem framework (`tsx scripts/testes.ts`), `node:vm`.

**Spec:** `docs/superpowers/specs/2026-09-30-fatia-3-design.md` (seções 4, 5 "Painel", 9).

**Acompanhamento:** `docs/acompanhamento/3.1-painel.md` (criado na Task 0). Ao fim de cada task, marcar o item no checklist e registrar em **Feito**, no mesmo commit.

`scripts/paridade.ts`:

```ts
/**
 * Paridade: roda o arquivo do app antigo num `vm` do Node, com o estado global
 * simulado, e compara cada função com a porta em src/lib/calculos sobre as
 * mesmas faturas. Se a porta divergir, o teste quebra — é a prova de que a
 * lógica mudou de casa sem mudar de regra.
 *
 * A comparação é por JSON depois de normalizar Map e Set: objetos criados no
 * `vm` têm outro `Object.prototype`, e `deepStrictEqual` os daria por
 * diferentes mesmo iguais. Por isso também `instanceof Map` não serve aqui.
 */
import { readFileSync } from 'node:fs';
import { join } from 'node:path';
import vm from 'node:vm';
import * as base from '../src/lib/calculos/base';
import type { Contexto, Fatura } from '../src/lib/calculos/base';
import { APELIDOS, NOMES_CARTAO, faturasDeTeste } from './fixture-legado';

type Fn = (...a: unknown[]) => unknown;
type Legado = Record<string, Fn>;
type Checar = (nome: string, ok: boolean) => void;

const EXPOSTAS = [
  'normCat', 'normDesc', 'tokensDesc', 'parecido', 'nomeExibido', 'addMonths',
  'chaveCartao', 'nomeCartao', 'gastos', 'totalFatura',
  'porMes', 'mesesDisponiveis', 'mesAtual', 'mesAnterior', 'lancsDoMes',
  'porCategoria', 'categoriasPorMes',
  'recorrentes', 'recorrentesFixas', 'classeLanc', 'composicaoMes',
];

const CODIGO = readFileSync(join(process.cwd(), 'docs/legado/calculos-do-app-antigo.js'), 'utf8');

export function legado(faturas: Fatura[], mesFiltro: string | null): Legado {
  const sandbox: Record<string, unknown> = {
    faturas: JSON.parse(JSON.stringify(faturas)),
    apelidos: { ...APELIDOS },
    nomesCartao: { ...NOMES_CARTAO },
    mesFiltro,
    cartaoFiltro: new Set(),
    structuredClone,
    Intl,
  };
  vm.createContext(sandbox);
  // as funções do arquivo são `const`/`function` de topo: um objeto no fim do
  // mesmo script é o jeito de tirá-las do escopo léxico do vm
  return vm.runInContext(`${CODIGO}\n;({${EXPOSTAS.join(',')}})`, sandbox) as Legado;
}

export function contexto(mesFiltro: string | null, faturas = faturasDeTeste()): Contexto {
  return { faturas, apelidos: { ...APELIDOS }, mesFiltro, nomesCartao: { ...NOMES_CARTAO } };
}

function normalizar(x: unknown): unknown {
  const tipo = Object.prototype.toString.call(x);
  if (tipo === '[object Map]') return [...(x as Map<unknown, unknown>)].map(([k, v]) => [k, normalizar(v)]);
  if (tipo === '[object Set]') return [...(x as Set<unknown>)].map(normalizar);
  if (Array.isArray(x)) return x.map(normalizar);
  if (x && typeof x === 'object') {
    return Object.fromEntries(Object.entries(x).map(([k, v]) => [k, normalizar(v)]));
  }
  return x;
}

export const igual = (a: unknown, b: unknown) => JSON.stringify(normalizar(a)) === JSON.stringify(normalizar(b));

export function paridadeBase(checar: Checar) {
  const L = legado(faturasDeTeste(), null);
  const ctx = contexto(null);

  const descs = [
    'ASSAI ATACADISTA 4471', 'Compra PÃO DE AÇÚCAR 03/10', 'PIX ADRIANA MARTINS 8821',
    'TRANSFERENCIA PARA ADRIANA MARTINS', 'Adriana — faxina', '', 'IOF', 'NETFLIX.COM',
  ];
  checar('paridade normDesc', descs.every((d) => L.normDesc!(d) === base.normDesc(d)));
  checar('paridade tokensDesc', descs.every((d) => igual(L.tokensDesc!(d), base.tokensDesc(d))));
  checar(
    'paridade parecido',
    descs.every((a) => descs.every((b) => L.parecido!(a, b) === base.parecido(a, b))),
  );
  const cats = ['Mercado', 'Mercado & Padaria', 'Viagem Espacial', '', undefined, null, 'Outros'];
  checar('paridade normCat', cats.every((c) => L.normCat!(c) === base.normCat(c)));
  checar(
    'paridade addMonths',
    [['2026-11', 3], ['2026-01', -1], ['2026-06', 0], ['2025-12', 13]].every(
      ([m, k]) => L.addMonths!(m, k) === base.addMonths(m as string, k as number),
    ),
  );

  const fats = faturasDeTeste();
  checar('paridade chaveCartao', fats.every((f) => L.chaveCartao!(f) === base.chaveCartao(f)));
  checar(
    'paridade chaveCartao sem dígitos',
    L.chaveCartao!({ banco: 'Itaú Personnalité', cartao: '' }) ===
      base.chaveCartao({ banco: 'Itaú Personnalité', cartao: '' }),
  );
  checar('paridade nomeCartao', fats.every((f) => L.nomeCartao!(f) === base.nomeCartao(ctx, f)));
  checar('paridade gastos', fats.every((f) => igual(L.gastos!(f), base.gastos(f))));
  checar('paridade totalFatura', fats.every((f) => L.totalFatura!(f) === base.totalFatura(f)));
  checar(
    'paridade nomeExibido',
    fats.every((f) => f.lancamentos.every((x) => L.nomeExibido!(x) === base.nomeExibido(ctx, x))),
  );
}
```

```ts
/**
 * Porta de docs/legado/calculos-do-app-antigo.js — constantes, texto e seleção
 * de faturas. A lógica é a do app antigo, linha a linha: mudar a regra e o
 * lugar ao mesmo tempo tornaria impossível saber qual dos dois quebrou. O teste
 * de paridade (scripts/paridade.ts) prova que as duas dão o mesmo resultado.
 *
 * O estado global de lá (`faturas`, `apelidos`, `mesFiltro`, `nomesCartao`)
 * vira o `Contexto`. O filtro de cartão (`fats()`) é aplicado antes, por quem
 * monta o contexto: aqui `ctx.faturas` já é o recorte.
 *
 * `normDesc` e `parecido` daqui NÃO são os de src/lib/descricao.ts: estes são os
 * do app antigo, dos quais dependem `recorrentes` e as cobranças fixas; os de lá
 * servem à deduplicação e ao aprendizado. Unificar fica para depois.
 *
 * Puro, sem `server-only`: o teste importa direto.
 */

export type TipoLegado = 'compra' | 'encargo' | 'estorno' | 'pagamento';

export type Lanc = {
  desc: string;
  apelido: string | null;
  val: number;
  cat: string;
  parc: string | null; // "3/10"
  tipo: TipoLegado;
};

export type Fatura = { id: string; mesRef: string; banco: string; cartao: string; lancamentos: Lanc[] };

export type Contexto = {
  faturas: Fatura[];
  apelidos: Record<string, string>;
  /** null = mês mais recente; 'TODOS'; ou 'YYYY-MM' */
  mesFiltro: string | null;
  nomesCartao: Record<string, string>;
};

export const CATS = [
  'Mercado & Padaria', 'Restaurantes & Delivery', 'Transporte', 'Assinaturas & Apps', 'Casa & Contas',
  'Construção / Obras', 'Saúde & Farmácia', 'Compras & Vestuário', 'Lazer & Viagem', 'Educação',
  'Serviços & Profissional', 'Encargos do cartão', 'Outros',
] as const;

// categorias renomeadas ao longo do tempo
const RENOMES: Record<string, string> = { Mercado: 'Mercado & Padaria' };

export function normCat(c: string | null | undefined): string {
  const n = (c != null && RENOMES[c]) || c;
  return n != null && (CATS as readonly string[]).includes(n) ? n : 'Outros';
}

export function addMonths(m: string, k: number): string {
  const y = +m.slice(0, 4);
  const mo = +m.slice(5, 7) - 1 + k;
  const d = new Date(Date.UTC(y, mo, 1));
  return d.getUTCFullYear() + '-' + String(d.getUTCMonth() + 1).padStart(2, '0');
}

// palavras que não identificam ninguém — ignoradas ao comparar nomes
const GENERICOS = new Set([
  'TRANSFERENCIA', 'TRANSFERENCIAS', 'TRANSF', 'PARA', 'PIX', 'TED', 'DOC', 'ENVIADO', 'RECEBIDO',
  'PAGAMENTO', 'PAGTO', 'COMPRA', 'CARTAO', 'DEBITO', 'CREDITO', 'LTDA', 'EIRELI', 'MEI', 'BRASIL', 'COM', 'APP',
  'SERVICOS', 'SERVICO', 'LOJA', 'REF', 'PARC', 'MENSALIDADE',
]);

export function tokensDesc(s: string | null | undefined): string[] {
  return String(s || '')
    .toUpperCase()
    .normalize('NFD')
    .replace(/[̀-ͯ]/g, '')
    .replace(/[^A-Z ]/g, ' ')
    .split(/\s+/)
    .filter((t) => t.length > 2 && !GENERICOS.has(t));
}

/** Mesmo estabelecimento com escrita diferente? */
export function parecido(a: string, b: string): boolean {
  const A = new Set(tokensDesc(a));
  const B = new Set(tokensDesc(b));
  if (!A.size || !B.size) return false;
  if (A.size === 1 || B.size === 1) return [...A].join(' ') === [...B].join(' ');
  let inter = 0;
  A.forEach((t) => {
    if (B.has(t)) inter++;
  });
  return inter >= 1 && inter / Math.min(A.size, B.size) >= 0.6 && inter / Math.max(A.size, B.size) >= 0.5;
}

export function normDesc(s: string | null | undefined): string {
  return String(s || '')
    .toUpperCase()
    .normalize('NFD')
    .replace(/[̀-ͯ]/g, '')
    .replace(/\d+\/\d+/g, ' ')
    .replace(/[^A-Z ]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim()
    .split(' ')
    .slice(0, 3)
    .join(' ');
}

/** O nome na tela: apelido do lançamento > apelido de um parecido > nome da fatura. */
export function nomeExibido(ctx: Contexto, l: Lanc): string {
  if (l.apelido) return l.apelido;
  const direto = ctx.apelidos[normDesc(l.desc)];
  if (direto) return direto;
  for (const key in ctx.apelidos) if (parecido(key, l.desc)) return ctx.apelidos[key]!;
  return l.desc;
}

/** Os 4 dígitos mandam; sem eles, o nome do banco normalizado. */
export function chaveCartao(f: { cartao?: string | null; banco?: string | null }): string {
  const dig = String(f.cartao || '').replace(/\D/g, '').slice(-4);
  if (dig.length === 4) return 'n' + dig;
  return (
    'b' +
    String(f.banco || '')
      .toUpperCase()
      .normalize('NFD')
      .replace(/[̀-ͯ]/g, '')
      .replace(/[^A-Z0-9]/g, '')
  );
}

export function nomeCartao(ctx: Contexto, f: Fatura): string {
  return (ctx.nomesCartao[chaveCartao(f)] || f.banco || 'Cartão') + (f.cartao ? ' ••' + f.cartao : '');
}

export const gastos = (f: Fatura) => f.lancamentos.filter((l) => l.tipo !== 'pagamento');

export function totalFatura(f: Fatura): number {
  return gastos(f).reduce((s, l) => s + (+l.val || 0), 0);
}
```

```ts
 * o `mesRef` — o mês da fatura, não o da compra.

import { CATS, gastos, normCat, totalFatura, type Contexto, type Lanc } from './base';

export function porMes(ctx: Contexto): [string, number][] {
  const m = new Map<string, number>();
  for (const f of ctx.faturas) m.set(f.mesRef, (m.get(f.mesRef) || 0) + totalFatura(f));

/** O mês em foco no painel — null significa "todos os meses". */
export function mesAtual(ctx: Contexto): string | null {
  const ms = mesesDisponiveis(ctx);
  if (ctx.mesFiltro === 'TODOS') return null;
  if (ctx.mesFiltro && ms.includes(ctx.mesFiltro)) return ctx.mesFiltro;
  return ms.length ? ms[ms.length - 1]! : null;
}

export function mesAnterior(ctx: Contexto, m: string | null): string | null {
  const ms = mesesDisponiveis(ctx);
  const i = m === null ? -1 : ms.indexOf(m);
  return i > 0 ? ms[i - 1]! : null;
}

export type LancDoMes = Lanc & { banco: string; cartao: string };

export function lancsDoMes(ctx: Contexto, m: string | null): LancDoMes[] {
  const out: LancDoMes[] = [];
  for (const f of ctx.faturas) {
    if (!m || f.mesRef === m) for (const l of gastos(f)) out.push({ ...l, banco: f.banco, cartao: f.cartao });
  }
  return out;
}

export function porCategoria(ctx: Contexto, m: string | null): Map<string, number> {
  const map = new Map<string, number>(CATS.map((c) => [c, 0]));
  for (const l of lancsDoMes(ctx, m)) {
    const c = normCat(l.cat);
    map.set(c, (map.get(c) || 0) + (+l.val || 0));
  }
  return map;
}

/** Categorias de todos os meses de uma vez (evita recalcular a cada consulta). */
export function categoriasPorMes(ctx: Contexto): Map<string, Map<string, number>> {
  const out = new Map<string, Map<string, number>>();
  for (const m of mesesDisponiveis(ctx)) out.set(m, porCategoria(ctx, m));
  return out;
}
```

```ts
/**
 * Porta de "Recorrentes" e "Compromisso × escolha" do app antigo. "Fixo" não é
 * um campo do lançamento: é a cobrança que se repetiu nos últimos meses com
 * valor quase igual — por isso a classe é calculada aqui, e não gravada.
 */
import { gastos, nomeExibido, normCat, normDesc, type Contexto, type Lanc } from './base';
import { lancsDoMes, mesAtual, mesesDisponiveis } from './meses';

export type Recorrente = { desc: string; cat: string; meses: number; medio: number; fixa: boolean; vals: number[] };

export function recorrentes(ctx: Contexto): Recorrente[] {
  const alvo = mesAtual(ctx);
  const todos = mesesDisponiveis(ctx);
  const ms = (alvo ? todos.filter((m) => m <= alvo) : todos).slice(-3);
  const g = new Map<string, { desc: string; cat: string; meses: Map<string, number> }>();
  for (const f of ctx.faturas) {
    if (!ms.includes(f.mesRef)) continue;
    for (const l of gastos(f)) {
      if (l.parc || l.tipo === 'estorno' || (+l.val || 0) <= 0) continue;
      const nome = nomeExibido(ctx, l);
      const k = normDesc(nome);
      if (!k) continue;
      if (!g.has(k)) g.set(k, { desc: nome, cat: normCat(l.cat), meses: new Map() });
      const r = g.get(k)!;
      r.meses.set(f.mesRef, (r.meses.get(f.mesRef) || 0) + (+l.val || 0)); // soma do MÊS, não da compra
      if (nome.length < r.desc.length) r.desc = nome;
    }
  }
  const out: Recorrente[] = [];
  for (const r of g.values()) {
    if (r.meses.size < 2) continue;
    const vals = [...r.meses.values()];
    const medio = vals.reduce((a, b) => a + b, 0) / vals.length; // média POR MÊS
    const variacao = medio ? Math.max(...vals.map((v) => Math.abs(v - medio))) / medio : 0;
    out.push({ desc: r.desc, cat: r.cat, meses: r.meses.size, medio, fixa: variacao <= 0.08, vals });
  }
  return out.sort((a, b) => b.medio - a.medio);
}

export const recorrentesFixas = (ctx: Contexto) => recorrentes(ctx).filter((r) => r.fixa);

export type Classe = 'fixo' | 'parcela' | 'encargo' | 'escolha';

/** Cada lançamento é compromisso (fixo, parcela, encargo) ou escolha do mês. */
export function classeLanc(ctx: Contexto, l: Lanc, fixas: Set<string>): Classe {
  if (l.tipo === 'encargo' || normCat(l.cat) === 'Encargos do cartão') return 'encargo';
  if (l.parc) return 'parcela';
  if (fixas.has(normDesc(nomeExibido(ctx, l)))) return 'fixo';
  return 'escolha';
}

export type Composicao = {
  fixo: number;
  parcela: number;
  encargo: number;
  escolha: number;
  tot: number;
  comprometido: number;
};

export function composicaoMes(ctx: Contexto, m: string | null, fixas?: Set<string>): Composicao {
  const set = fixas || new Set(recorrentesFixas(ctx).map((r) => normDesc(r.desc)));
  const b = { fixo: 0, parcela: 0, encargo: 0, escolha: 0 };
  for (const l of lancsDoMes(ctx, m)) b[classeLanc(ctx, l, set)] += +l.val || 0;
  const tot = b.fixo + b.parcela + b.encargo + b.escolha;
  return { ...b, tot, comprometido: b.fixo + b.parcela + b.encargo };
}
```

```ts
/**
 * Do banco ao formato do app antigo. As funções portadas continuam lendo
 * `faturas[].lancamentos[]`, e é aqui — não nelas — que o banco se encaixa.
 * Puro: a consulta fica em src/lib/painel-repo.ts.

import { chaveCartao, normCat, type Fatura, type Lanc, type TipoLegado } from './base';

/** Uma linha da consulta do painel (txn ⋈ statement ⋈ account ⋈ category ⋈ plano). */
export type LinhaPainel = {
  statementId: string | null;
  accountId: string;
  mes: string; // 'YYYY-MM', o mês da fatura (mesDaTxn)
  banco: string;
  last4: string | null;
  apelidoConta: string | null;
  desc: string;
  apelido: string | null;
  valor: string; // numeric chega como texto
  categoria: string | null;
  tipo: 'fixo' | 'parcela' | 'encargo' | 'escolha' | 'pagamento';
  parcelaN: number | null;
  totalParcelas: number | null;
};

/**
 * `txn.tipo` guarda a origem; o app antigo distinguia compra, encargo, estorno e
 * pagamento. Estorno não tem tipo próprio no banco: é o valor negativo que o
 * denuncia.
 */
function tipoLegado(l: LinhaPainel, val: number): TipoLegado {
  if (l.tipo === 'pagamento') return 'pagamento';
  if (l.tipo === 'encargo') return 'encargo';
  if (val < 0) return 'estorno';
  return 'compra';
}

export function montarFaturas(linhas: readonly LinhaPainel[]): {
  faturas: Fatura[];
  nomesCartao: Record<string, string>;
} {
  const porId = new Map<string, Fatura>();
  const nomesCartao: Record<string, string> = {};

  for (const l of linhas) {
    // sem fatura (conta corrente, fatia 6): uma fatura por conta e mês
    const id = l.statementId ?? `${l.accountId}|${l.mes}`;
    let f = porId.get(id);
    if (!f) {
      f = { id, mesRef: l.mes, banco: l.banco, cartao: l.last4 ?? '', lancamentos: [] };
      porId.set(id, f);
    }
    if (l.apelidoConta) nomesCartao[chaveCartao(f)] = l.apelidoConta;

    const val = Number(l.valor);
    const lanc: Lanc = {
      desc: l.desc,
      apelido: l.apelido,
      val,
      cat: l.categoria ?? 'Outros',
      parc: l.parcelaN && l.totalParcelas ? `${l.parcelaN}/${l.totalParcelas}` : null,
      tipo: tipoLegado(l, val),
    };
    f.lancamentos.push(lanc);
  }

  return { faturas: [...porId.values()], nomesCartao };
}

const OBRA = 'Construção / Obras';

/**
 * "Separar obra": a obra sai dos totais, das médias e das anomalias, e volta
 * como um número à parte por mês. Feito antes das contas para que nenhuma
 * função portada precise saber que a obra existe.
 */
export function separarObra(faturas: readonly Fatura[]): {
  faturas: Fatura[];
  obraPorMes: Record<string, number>;
} {
  const obraPorMes: Record<string, number> = {};
  const semObra = faturas.map((f) => {
    const resto: Lanc[] = [];
    for (const l of f.lancamentos) {
      if (normCat(l.cat) === OBRA && l.tipo !== 'pagamento') {
        obraPorMes[f.mesRef] = (obraPorMes[f.mesRef] ?? 0) + (+l.val || 0);
      } else {
        resto.push(l);
      }
    }
    return { ...f, lancamentos: resto };
  });
  return { faturas: semObra, obraPorMes };
}
```

```ts
  const total = mes ? (pm.find((x) => x[0] === mes)?.[1] ?? 0) : pm.reduce((s, x) => s + x[1], 0);

  const ant = mes ? mesAnterior(ctx, mes) : null;
  const anterior = ant ? { mes: ant, total: pm.find((x) => x[0] === ant)?.[1] ?? 0 } : null;
  const variacao = anterior && anterior.total > 0 ? total / anterior.total - 1 : null;

  const obraBruta = mes
    ? (obraPorMes[mes] ?? 0)
    : Object.values(obraPorMes).reduce((s, v) => s + v, 0);

  const categorias = [...porCategoria(ctx, mes)]
    .filter(([, v]) => v > 0)
    .sort((a, b) => b[1] - a[1])
    .map(([cat, valor]) => ({ cat, valor }));

  return {
    mes,
    meses,
    total,
    anterior,
    variacao,
    obra: separar && obraBruta > 0 ? obraBruta : null,
    composicao: composicaoMes(ctx, mes),
    categorias,
    vazio: lancsDoMes(ctx, mes).length === 0,
  };
}
```

```ts
import 'server-only';
import { and, asc, eq, inArray, isNull, isNotNull } from 'drizzle-orm';
import { db } from '@/db';
import { account, category, installmentPlan, institution, learnedLabel, statement, txn } from '@/db/schema';
import { montarFaturas, separarObra, type LinhaPainel } from './calculos/adaptador';
import type { Contexto } from './calculos/base';
import type { Filtro } from './filtro';
import { mesDaTxn } from './txn-repo';

/**
 * Monta o contexto das contas do painel: uma consulta pelos lançamentos do
 * recorte de cartões, outra pelos apelidos aprendidos. Todas as faturas do
 * recorte vêm, sem janela de meses — são dezenas, e escolher em SQL quais meses
 * entram seria reescrever `mesAtual` e `recorrentes`.
 *
 * Posse no `where`: `txn.user_id` e `learned_label.user_id` da sessão; os ids
 * de cartão do filtro só estreitam o que já é seu.
 */
export async function carregarContexto(
  userId: string,
  f: Filtro,
): Promise<{ ctx: Contexto; obraPorMes: Record<string, number> }> {
  const cond = [eq(txn.userId, userId), isNull(txn.excluidoEm)];
  if (f.cartoes.length) cond.push(inArray(txn.accountId, f.cartoes));

  const [linhas, rotulos] = await Promise.all([
    db
      .select({
        statementId: txn.statementId,
        accountId: txn.accountId,
        mes: mesDaTxn,
        banco: institution.nome,
        last4: account.last4,
        apelidoConta: account.apelido,
        desc: txn.descOrigem,
        apelido: txn.apelido,
        valor: txn.valor,
        categoria: category.nome,
        tipo: txn.tipo,
        parcelaN: txn.parcelaN,
        totalParcelas: installmentPlan.totalParcelas,
      })
      .from(txn)
      .leftJoin(statement, eq(statement.id, txn.statementId))
      .innerJoin(account, eq(account.id, txn.accountId))
      .innerJoin(institution, eq(institution.id, account.institutionId))
      .leftJoin(category, eq(category.id, txn.categoryId))
      .leftJoin(installmentPlan, eq(installmentPlan.id, txn.installmentId))
      .where(and(...cond))
      // a ordem importa: o app antigo somava na ordem em que os lançamentos
      // estavam na fatura, e `recorrentes` guarda o nome mais curto visto
      .orderBy(asc(txn.data), asc(txn.criadoEm), asc(txn.id)),
    db
      .select({ descNorm: learnedLabel.descNorm, apelido: learnedLabel.apelido })
      .from(learnedLabel)
      .where(and(eq(learnedLabel.userId, userId), isNotNull(learnedLabel.apelido))),
  ]);

  const { faturas, nomesCartao } = montarFaturas(linhas as LinhaPainel[]);
  const { faturas: recorte, obraPorMes } = f.obra ? separarObra(faturas) : { faturas, obraPorMes: {} };
  const apelidos: Record<string, string> = {};
  for (const r of rotulos) if (r.apelido) apelidos[r.descNorm] = r.apelido;

  return { ctx: { faturas: recorte, apelidos, mesFiltro: f.mes, nomesCartao }, obraPorMes };
}
```

Nota: a chave de `learned_label` é `chaveAprendizado` (3 palavras da descrição limpa), não o `normDesc` do legado. `nomeExibido` tenta a chave exata e depois `parecido` sobre todas as chaves — o segundo passo cobre a diferença. Registre isso em **Decisões**.

Se o typecheck reclamar do `as LinhaPainel[]` (ex.: `tipo` como enum mais largo), ajuste o tipo `LinhaPainel['tipo']` para o `typeof txnKind.enumValues[number]` e remova o cast — sem cast é melhor.

```tsx
import Link from 'next/link';
import Dica from '@/components/Dica';
import { resumoDoPainel } from '@/lib/calculos/painel';
import { deParametros, lerFiltro, paraQuery, rotuloMes } from '@/lib/filtro';
import { moeda0, pct } from '@/lib/formato';
import { carregarContexto } from '@/lib/painel-repo';
import { exigirUsuario } from '@/lib/sessao-servidor';
import { opcoesDoFiltro } from '@/lib/txn-repo';

export const dynamic = 'force-dynamic';

type Params = Promise<Record<string, string | string[] | undefined>>;

const PARTES = [
  { chave: 'fixo', rotulo: 'fixos' },
  { chave: 'parcela', rotulo: 'parcelas' },
  { chave: 'encargo', rotulo: 'encargos' },
  { chave: 'escolha', rotulo: 'escolha' },
] as const;

export default async function Painel({ searchParams }: { searchParams: Params }) {
  const userId = await exigirUsuario();
  const f = lerFiltro(deParametros(await searchParams));
  const { meses } = await opcoesDoFiltro(userId);

  if (meses.length === 0) {
    return (
      <section className="cartao vazio">
        <h2>Importe uma fatura para começar</h2>
        <p className="suave">O painel aparece assim que houver um mês aqui dentro.</p>
        <Link href="/lancamentos?importar=1" className="botao">
          Importar fatura em PDF
        </Link>
      </section>
    );
  }

  const { ctx, obraPorMes } = await carregarContexto(userId, f);
  const r = resumoDoPainel(ctx, obraPorMes, f.obra);
  const recorte = r.mes ? rotuloMes(r.mes) : 'todos os meses';

  if (r.vazio) {
    return (
      <section className="cartao vazio">
        <h2>
          Nada em {recorte}
          {f.cartoes.length ? ' nestes cartões' : ''}
        </h2>
        <p className="suave">Troque o mês no filtro acima{f.cartoes.length ? ' ou volte a todos os cartões' : ''}.</p>
        {f.cartoes.length > 0 && (
          <Link href={`/?${paraQuery({ ...f, cartoes: [] })}`} className="botao">
            Ver todos os cartões
          </Link>
        )}
      </section>
    );
  }

  // a categoria abre a lista já filtrada, mantendo o recorte
  const filtroQ = paraQuery(f);
  const verCategoria = (cat: string) => {
    const q = new URLSearchParams(filtroQ);
    q.set('cat', cat);
    return `/lancamentos?${q}`;
  };

  const c = r.composicao;
  // estorno pode deixar uma parte negativa; a barra só desenha o que é positivo
  const partes = PARTES.map((p) => ({ ...p, valor: Math.max(0, c[p.chave]) }));
  const somaPartes = partes.reduce((s, p) => s + p.valor, 0);
  const maiorCat = r.categorias[0]?.valor ?? 0;

  return (
    <div className="painel">
      <h1>Painel · {recorte}</h1>

      <section className="cartao painel-bloco">
        <span className="eyebrow">Quanto foi</span>
        <p className="painel-total">{moeda0(r.total)}</p>
        <p className="suave">
          {r.mes === null
            ? `Soma de ${r.meses.length} ${r.meses.length === 1 ? 'mês' : 'meses'}.`
            : r.anterior && r.variacao !== null
              ? `${pct(r.variacao)} contra ${rotuloMes(r.anterior.mes)} (${moeda0(r.anterior.total)})`
              : 'Primeiro mês com dado — ainda sem comparação.'}
        </p>
      </section>

      {r.obra !== null && (
        <p className="painel-obra">
          + {moeda0(r.obra)} de Construção / Obras {r.mes ? 'neste mês' : 'no período'}, fora dos números abaixo.
        </p>
      )}

      <section className="cartao painel-bloco">
        <div className="painel-cabeca">
          <span className="eyebrow">Quanto era escolha</span>
          <Dica rotulo="O que é compromisso e escolha?">
            Compromisso é o que já estava decidido antes do mês: cobranças fixas (as que se repetem com o mesmo
            valor), parcelas e encargos do cartão. Escolha é o resto — onde dá para mexer.
          </Dica>
        </div>
        <p>
          <strong>{moeda0(c.comprometido)}</strong> já estava decidido · <strong>{moeda0(c.escolha)}</strong> foi
          escolha
        </p>
        <div
          className="barra-composicao"
          role="img"
          aria-label={partes.map((p) => `${p.rotulo} ${moeda0(p.valor)}`).join(', ')}
        >
          {somaPartes > 0 &&
            partes.map((p) =>
              p.valor > 0 ? (
                <span key={p.chave} className={`parte-${p.chave}`} style={{ width: `${(p.valor / somaPartes) * 100}%` }} />
              ) : null,
            )}
        </div>
        <ul className="legenda">
          {partes.map((p) => (
            <li key={p.chave}>
              <span className={`marca-cor parte-${p.chave}`} aria-hidden="true" />
              {p.rotulo} {moeda0(p.valor)}
            </li>
          ))}
        </ul>
      </section>

      <section className="cartao painel-bloco">
        <span className="eyebrow">Para onde foi</span>
        <ul className="categorias">
          {r.categorias.map((x) => (
            <li key={x.cat}>
              <Link href={verCategoria(x.cat)}>
                <span className="cat-nome">{x.cat}</span>
                <span className="cat-barra" aria-hidden="true">
                  <span style={{ width: `${maiorCat ? (x.valor / maiorCat) * 100 : 0}%` }} />
                </span>
                <span className="valor">{moeda0(x.valor)}</span>
              </Link>
            </li>
          ))}
        </ul>
      </section>
    </div>
  );
}
```
