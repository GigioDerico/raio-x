```ts
/**
 * Porta de `mdLeve` (docs/legado/app-antigo.html:2829-2844), mas devolvendo
 * blocos em vez de HTML: o texto da IA é entrada não confiável e só vira
 * elemento React, que escapa tudo (plano técnico, segurança item 6).
 */
export type Parte = { texto: string; forte?: true; dinheiro?: true };
export type Bloco =
  | { tipo: 'titulo'; partes: Parte[] }
  | { tipo: 'paragrafo'; partes: Parte[] }
  | { tipo: 'lista'; itens: Parte[][] };

const DINHEIRO = /(R\$\s?[\d.,]+)/g;

function partesDe(s: string): Parte[] {
  const out: Parte[] = [];
  for (const pedaco of s.split(/(\*\*[^*]+\*\*)/g)) {
    if (!pedaco) continue;
    const forte = /^\*\*[^*]+\*\*$/.test(pedaco);
    const texto = forte ? pedaco.slice(2, -2) : pedaco;
    for (const q of texto.split(DINHEIRO)) {
      if (!q) continue;
      const p: Parte = { texto: q };
      if (forte) p.forte = true;
      if (/^R\$\s?[\d.,]+$/.test(q)) p.dinheiro = true;
      out.push(p);
    }
  }
  return out;
}

export function markdownLeve(t: string): Bloco[] {
  const blocos: Bloco[] = [];
  let lista: Parte[][] = [];
  const fechar = () => {
    if (lista.length) blocos.push({ tipo: 'lista', itens: lista });
    lista = [];
  };
  for (const bruta of t.split('\n')) {
    const l = bruta.trim();
    if (/^#{2,4}\s/.test(l)) {
      fechar();
      blocos.push({ tipo: 'titulo', partes: partesDe(l.replace(/^#+\s*/, '')) });
    } else if (/^([-*•]|\d+\.)\s/.test(l)) {
      lista.push(partesDe(l.replace(/^([-*•]|\d+\.)\s*/, '')));
    } else if (!l) {
      fechar();
    } else {
      fechar();
      blocos.push({ tipo: 'paragrafo', partes: partesDe(l) });
    }
  }
  fechar();
  return blocos;
}
```

```ts
/**
 * A rota do diagnóstico responde uma linha JSON por evento. Puro: a rota
 * escreve com `linhaNdjson`, o cliente lê com `lerNdjson`, e o teste usa os dois.
 */
export type EventoDiagnostico = { t: string } | { fim: true } | { erro: string };
export type EventoLido = { t?: unknown; fim?: unknown; erro?: unknown };

export const linhaNdjson = (e: EventoDiagnostico) => JSON.stringify(e) + '\n';

/** Devolve os eventos das linhas completas; o pedaço sem `\n` fica para o próximo. */
export function lerNdjson(buffer: string): { eventos: EventoLido[]; resto: string } {
  const linhas = buffer.split('\n');
  const resto = linhas.pop() ?? '';
  const eventos: EventoLido[] = [];
  for (const linha of linhas) {
    const s = linha.trim();
    if (!s) continue;
    try {
      const v: unknown = JSON.parse(s);
      if (v && typeof v === 'object' && !Array.isArray(v)) eventos.push(v as EventoLido);
    } catch {
      // uma linha estragada não derruba o texto que já chegou
    }
  }
  return { eventos, resto };
}
```

```ts
/**
 * Uma linha do stream SSE do OpenRouter. Fora do `openrouter.ts` porque ele é
 * `server-only` e o teste precisa importar isto.
 */
export type UsoSse = { tokensIn: number; tokensOut: number; custoUsd: number | null };
export type LinhaSse = { texto: string; uso: UsoSse | null; fim: boolean; erro: string | null };

type Pedaco = {
  choices?: Array<{ delta?: { content?: unknown } }>;
  usage?: { prompt_tokens?: number; completion_tokens?: number; cost?: unknown };
  error?: { message?: unknown };
};

export function lerLinhaSse(linha: string): LinhaSse | null {
  const l = linha.trim();
  // ": OPENROUTER PROCESSING" é comentário de keep-alive; linha vazia separa eventos
  if (!l.startsWith('data:')) return null;
  const dado = l.slice(5).trim();
  if (dado === '[DONE]') return { texto: '', uso: null, fim: true, erro: null };
  let p: Pedaco;
  try {
    p = JSON.parse(dado) as Pedaco;
  } catch {
    return null;
  }
  if (p.error) return { texto: '', uso: null, fim: false, erro: String(p.error.message ?? 'falha') };
  const c = p.choices?.[0]?.delta?.content;
  const u = p.usage;
  return {
    texto: typeof c === 'string' ? c : '',
    uso: u
      ? { tokensIn: u.prompt_tokens ?? 0, tokensOut: u.completion_tokens ?? 0, custoUsd: typeof u.cost === 'number' ? u.cost : null }
      : null,
    fim: false,
    erro: null,
  };
}
```

```ts
/** A resposta da IA é dado não confiável: só um texto, aparado e com limite. */
export function lerTexto(dados: unknown, max: number): string | null {
  if (!dados || typeof dados !== 'object') return null;
  const texto = (dados as { texto?: unknown }).texto;
  if (typeof texto !== 'string') return null;
  const t = texto.trim();
  return t ? t.slice(0, max) : null;
}

export const lerResumo = (dados: unknown) => lerTexto(dados, 600);
```

```ts
export function dadosDiagnostico(ctx: Contexto, filtrado: boolean, plano: PlanoParaIA | null): string {
  const ms = porMes(ctx);
  const m = mesAtual(ctx);
  const prevM = ms.length > 1 ? ms[ms.length - 2]![0] : null;
  const cur = porCategoria(ctx, m);
  const prev = prevM ? porCategoria(ctx, prevM) : null;
  const tot = [...cur.values()].reduce((a, b) => a + b, 0);
  const L: string[] = [];
  L.push('Recorte analisado: ' + (m ? 'fatura de ' + rotuloMes(m) : 'todos os meses carregados'));
  L.push('Renda mensal informada: ' + (ctx.renda > 0 ? moeda(ctx.renda) : 'não informada'));
  L.push('Total por mês de fatura: ' + ms.map(([k, v]) => rotuloMes(k) + ' ' + moeda(v)).join(' | '));
  L.push(
    'Cartões considerados: ' +
      cartoesInfo(ctx).map((c) => c.nome).join(', ') +

  );
  L.push('');
  L.push(

  );
  [...cur.entries()]
    .filter(([, v]) => v > 0)
    .sort((a, b) => b[1] - a[1])
    .forEach(([c, v]) => {
      const p = prev ? prev.get(c) || 0 : 0;
      L.push('- ' + c + ': ' + moeda(v) + ', ' + (tot ? ((v / tot) * 100).toFixed(0) : 0) + '%' + (p > 0 ? ', ' + pct(v / p - 1) : prev ? ', novo' : ''));
    });
  L.push('');
  L.push('Encargos/juros no mês: ' + moeda(encargosDoMes(ctx, m)));

      ' | escolha do mês ' + moeda(comp.escolha) + ' (já contratado: ' + moeda(comp.comprometido) + ')',

  if (pj)
    L.push(
      'Previsão para ' + rotuloMes(pj.prox) + ': ' + moeda(pj.total) + ' (parcelas ' + moeda(pj.parcelas) + ' + fixos ' + moeda(pj.fixos) +

  return `Você é um analista de finanças pessoais falando português do Brasil, direto e sem rodeios.
Abaixo estão os números reais das faturas de cartão de crédito de uma pessoa.

Escreva um diagnóstico curto usando os valores reais (não invente nada que não esteja nos dados).
Estruture exatamente nestas seções, com "###" antes de cada título:
### O diagnóstico — 2 ou 3 frases sobre o que os números mostram.
### Onde está o problema — as 3 categorias ou hábitos que mais pesam, com valor e variação.
### O que cortar primeiro — 3 cortes concretos, cada um com a economia estimada por mês em reais.
```

```ts
import { lerTexto } from './calculos/resumo';

export type KindSaida = 'resumo' | 'diagnostico';
// o mesmo limite que a rota aplicou ao gravar: um jsonb mexido à mão não passa disso
const LIMITE: Record<KindSaida, number> = { resumo: 600, diagnostico: 6000 };

/**
 * Prefere a linha do mesmo prompt (o "↻" pode ter deixado outra mais nova com
 * dado diferente, e voltar ao dado antigo não deve pagar de novo). Sem ela, a
 * mais recente do recorte, para a tela dizer que os dados mudaram.
 */
export async function saidaGuardada(userId: string, kind: KindSaida, escopo: string, hash: string) {
  const [linha] = await db
    .select({ conteudo: aiOutput.conteudo, inputHash: aiOutput.inputHash })
    .from(aiOutput)
    .where(and(eq(aiOutput.userId, userId), eq(aiOutput.kind, kind), eq(aiOutput.escopo, escopo)))
    .orderBy(desc(sql`${aiOutput.inputHash} = ${hash}`), desc(aiOutput.criadoEm))
    .limit(1);
  if (!linha) return null;
  const texto = lerTexto(linha.conteudo, LIMITE[kind]);
  return texto ? { texto, atual: linha.inputHash === hash } : null;
}

export async function guardarSaida(userId: string, kind: KindSaida, escopo: string, inputHash: string, texto: string, modelo: string, uso: Uso) {
  const custoUsd = uso.custoUsd === null ? null : uso.custoUsd.toFixed(6);
  await db
    .insert(aiOutput)
    .values({ userId, kind, escopo, modelo, inputHash, conteudo: { texto }, tokensIn: uso.tokensIn, tokensOut: uso.tokensOut, custoUsd })
    // "↻" com o mesmo prompt bate na chave única (user, kind, escopo, input_hash)
    .onConflictDoUpdate({
      target: [aiOutput.userId, aiOutput.kind, aiOutput.escopo, aiOutput.inputHash],
      set: { conteudo: { texto }, modelo, tokensIn: uso.tokensIn, tokensOut: uso.tokensOut, custoUsd, criadoEm: sql`now()` },
    });
}

export const resumoGuardado = (userId: string, escopo: string, hash: string) => saidaGuardada(userId, 'resumo', escopo, hash);
export const guardarResumo = (userId: string, escopo: string, inputHash: string, texto: string, modelo: string, uso: Uso) =>
  guardarSaida(userId, 'resumo', escopo, inputHash, texto, modelo, uso);
```

```ts
export type EventoStream = { tipo: 'texto'; t: string } | { tipo: 'fim'; modelo: string; uso: Uso };

async function abrirStream(modelo: string, prompt: string, sinal?: AbortSignal) {
  const r = await fetch(URL_BASE, {
    method: 'POST',
    signal: sinal,
    headers: {
      authorization: `Bearer ${env.OPENROUTER_API_KEY}`,
      'content-type': 'application/json',
      'x-title': 'Raio-X',
    },
    body: JSON.stringify({
      model: modelo,
      messages: [{ role: 'user', content: prompt }],
      stream: true,
      // texto corrido: um pouco de variação lê melhor que o zero do JSON
      temperature: 0.3,
      usage: { include: true },
    }),
  });
  if (!r.ok || !r.body) {
    const corpo = await r.text().catch(() => '');
    throw new Error(`OpenRouter ${r.status}: ${corpo.slice(0, 200)}`);
  }
  return r.body;
}

async function* linhasDe(corpo: ReadableStream<Uint8Array>) {
  const leitor = corpo.getReader();
  const dec = new TextDecoder();
  let buf = '';
  try {
    for (;;) {
      const { value, done } = await leitor.read();
      if (done) break;
      buf += dec.decode(value, { stream: true });
      const linhas = buf.split('\n');
      buf = linhas.pop() ?? '';
      yield* linhas;
    }
    buf += dec.decode();
    if (buf) yield buf;
  } finally {
    leitor.releaseLock();
  }
}

/**
 * Texto em stream. O reserva só entra se o principal falhar antes da primeira
 * palavra — depois disso, trocar de modelo emendaria duas respostas diferentes.
 * O uso vem no último pedaço (`usage.include`).
 */
export async function* pedirTextoEmStream(
  prompt: string,
  opcoes: { modelo?: string; sinal?: AbortSignal } = {},
): AsyncGenerator<EventoStream> {
  if (!env.OPENROUTER_API_KEY) throw new SemChaveIA();

  const tentativas = [opcoes.modelo ?? env.OPENROUTER_MODEL, env.OPENROUTER_MODEL_FALLBACK].filter(
    (m, i, a): m is string => Boolean(m) && a.indexOf(m) === i,
  );

  let ultimoErro: unknown;
  for (const modelo of tentativas) {
    let comecou = false;
    try {
      const corpo = await abrirStream(modelo, prompt, opcoes.sinal);
      let uso: Uso = { tokensIn: 0, tokensOut: 0, custoUsd: null };
      for await (const linha of linhasDe(corpo)) {
        const p = lerLinhaSse(linha);
        if (!p) continue;
        if (p.erro) throw new Error(`OpenRouter: ${p.erro}`);
        if (p.uso) uso = p.uso;
        if (p.texto) {
          comecou = true;
          yield { tipo: 'texto', t: p.texto };
        }
        if (p.fim) break;
      }
      if (!comecou) throw new Error('OpenRouter: resposta vazia');
      yield { tipo: 'fim', modelo, uso };
      return;
    } catch (e) {
      if (comecou || opcoes.sinal?.aborted) throw e;
      ultimoErro = e;
    }
  }
  throw ultimoErro instanceof Error ? ultimoErro : new Error('falha ao chamar o modelo');
}
```

```ts
import type { NextRequest } from 'next/server';
import { env } from '@/env';
import { chaveRecorte, lerTexto } from '@/lib/calculos/resumo';
import { lancsDoMes, mesAtual } from '@/lib/calculos/meses';
import { lerFiltro } from '@/lib/filtro';
import { dadosDiagnostico, promptDiagnostico } from '@/lib/ia';
import { registrarUso } from '@/lib/ia-uso-repo';
import { linhaNdjson, type EventoDiagnostico } from '@/lib/ndjson';
import { pedirTextoEmStream } from '@/lib/openrouter';
import { carregarContexto } from '@/lib/painel-repo';
import { planoAtivo } from '@/lib/plano-repo';
import { guardarSaida, hashDoPrompt, saidaGuardada } from '@/lib/resumo-repo';
import { ErroDeUso, comUsuario, corpoJson } from '@/lib/rota';
import { configuracoes } from '@/lib/sessao-servidor';

export const runtime = 'nodejs';
// o stream de umas 400 palavras cabe folgado; o reserva só entra antes da primeira
export const maxDuration = 60;

type Preparo = { userId: string; prompt: string; hash: string; escopo: string; modelo?: string };

export async function POST(req: NextRequest) {
  const filtro = lerFiltro(req.nextUrl.searchParams);
  // objeto, não `let`: o TS não acompanha atribuição feita dentro do callback
  const p: { v?: Preparo } = {};

  // tudo que pode falhar com mensagem para o usuário acontece antes de abrir o stream
  const antes = await comUsuario(async (userId) => {
    const corpo = await corpoJson<unknown>(req);
    const forcar = !!corpo && typeof corpo === 'object' && (corpo as { forcar?: unknown }).forcar === true;

    const { ctx } = await carregarContexto(userId, filtro);
    const m = mesAtual(ctx);
    if (!lancsDoMes(ctx, m).length) throw new ErroDeUso('sem_lancamentos', 'Não há lançamentos neste recorte');

    const plano = await planoAtivo(userId);
    const prompt = promptDiagnostico(dadosDiagnostico(ctx, filtro.cartoes.length > 0, plano?.resultado ?? null));
    const hash = hashDoPrompt(prompt);
    const escopo = chaveRecorte(filtro, m);

    if (!forcar) {
      // mesmos números, mesmo diagnóstico: não paga de novo pelo mesmo texto
      const guardado = await saidaGuardada(userId, 'diagnostico', escopo, hash);
      if (guardado?.atual) return { texto: guardado.texto };
    }
    if (!env.OPENROUTER_API_KEY) throw new ErroDeUso('sem_chave', 'Configure OPENROUTER_API_KEY');

    const cfg = await configuracoes(userId);
    p.v = { userId, prompt, hash, escopo, modelo: cfg?.modeloIa };
    return null;
  });
  if (!p.v) return antes;

  const { userId, prompt, hash, escopo, modelo } = p.v;
  const enc = new TextEncoder();
  const corpo = new ReadableStream<Uint8Array>({
    async start(controle) {
      const enviar = (e: EventoDiagnostico) => controle.enqueue(enc.encode(linhaNdjson(e)));
      let texto = '';
      try {
        // req.signal: o "Parar" do cliente corta também a chamada ao modelo
        for await (const ev of pedirTextoEmStream(prompt, { modelo, sinal: req.signal })) {
          if (ev.tipo === 'texto') {
            texto += ev.t;
            enviar({ t: ev.t });
          } else {
            await registrarUso(userId, 'diagnostico', ev.modelo, ev.uso);
            const final = lerTexto({ texto }, 6000);
            if (final) await guardarSaida(userId, 'diagnostico', escopo, hash, final, ev.modelo, ev.uso);
            enviar({ fim: true });
          }
        }
      } catch (e) {
        // interrompido pelo usuário: não há para quem responder, e nada é guardado
        if (!req.signal.aborted) {
          console.error('[diagnostico]', e);
          enviar({ erro: 'A IA não terminou o diagnóstico; tente de novo' });
        }
      } finally {
        try {
          controle.close();
        } catch {
          // já fechado pelo cancelamento do cliente
        }
      }
    },
  });

  return new Response(corpo, {
    headers: { 'content-type': 'application/x-ndjson; charset=utf-8', 'cache-control': 'no-store' },
  });
}
```

```tsx
'use client';

import { useRef, useState, useTransition } from 'react';
import { useRouter } from 'next/navigation';
import { lerNdjson } from '@/lib/ndjson';
import TextoIA from './TextoIA';

type Estado = 'atual' | 'desatualizado' | 'nenhum';
type Props = { titulo: string; texto: string | null; estado: Estado; query: string };

const FALHA = 'não consegui falar com a IA, tente de novo';

export default function Diagnostico({ titulo, texto, estado, query }: Props) {
  const router = useRouter();
  const [gerando, setGerando] = useState(false);
  const [atualizando, startTransition] = useTransition();
  const [erro, setErro] = useState('');
  const [aviso, setAviso] = useState('');
  // o texto que está chegando ou acabou de chegar; sem ele a tela piscaria o antigo até o refresh
  const [novo, setNovo] = useState('');
  const [concluido, setConcluido] = useState(false);
  const controle = useRef<AbortController | null>(null);

  const ocupado = gerando || atualizando;
  const textoVisto = novo || texto;
  const estadoVisto: Estado = concluido ? 'atual' : estado;

  async function gerar(forcar: boolean) {
    const ac = new AbortController();
    controle.current = ac;
    setGerando(true);
    setErro('');
    setAviso('Gerando o diagnóstico…');
    setNovo('');
    setConcluido(false);
    let acumulado = '';
    let fim = false;
    let falhou = false;
    try {
      const r = await fetch(`/api/diagnostico?${query}`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ forcar }),
        signal: ac.signal,
      });
      if (!r.ok) {
        const d = await r.json().catch(() => ({}));
        // 400 traz mensagem escrita para o usuário ("configure a chave", "sem lançamentos")
        setErro(r.status === 400 ? d?.detalhe || 'não deu certo' : FALHA);
        return;
      }
      if ((r.headers.get('content-type') ?? '').includes('application/json')) {
        // mesmo recorte e mesmos números: a rota devolveu o guardado sem chamar a IA
        const d = await r.json().catch(() => ({}));
        if (typeof d?.texto === 'string') setNovo(d.texto);
        fim = true;
      } else if (r.body) {
        const leitor = r.body.getReader();
        const dec = new TextDecoder();
        let buf = '';
        for (;;) {
          const { value, done } = await leitor.read();
          if (done) break;
          const lido = lerNdjson(buf + dec.decode(value, { stream: true }));
          buf = lido.resto;
          for (const e of lido.eventos) {
            if (typeof e.t === 'string') {
              acumulado += e.t;
              setNovo(acumulado);
            } else if (e.fim === true) fim = true;
            else if (typeof e.erro === 'string') {
              falhou = true;
              setErro(e.erro);
            }
          }
        }
      }
      if (fim) {
        setConcluido(true);
        setAviso('Diagnóstico pronto.');
        startTransition(() => router.refresh());
      } else if (!falhou) {
        setErro('a resposta chegou pela metade; tente de novo');
      }
    } catch {
      if (ac.signal.aborted) setAviso('Interrompido. O texto parcial não foi guardado.');
      else setErro(FALHA);
    } finally {
      setGerando(false);
      controle.current = null;
      if (!fim && !ac.signal.aborted) setAviso('');
    }
  }

  return (
    <section className="cartao diagnostico" aria-labelledby="diag-titulo" aria-busy={ocupado}>
      <div className="diag-topo">
        <h1 id="diag-titulo">{titulo}</h1>
        {estadoVisto === 'atual' && !gerando && (
          <button type="button" className="link diag-refazer" onClick={() => gerar(true)} disabled={ocupado} aria-label="Gerar o diagnóstico de novo" title="Gerar o diagnóstico de novo">
            ↻
          </button>
        )}
      </div>
      {estadoVisto === 'desatualizado' && !gerando && <p className="diag-nota">Os números mudaram desde este diagnóstico.</p>}
      {/* sem aria-live aqui: anunciar cada pedaço do stream seria ruído; quem anuncia é o status abaixo */}
      <div className="diag-corpo">
        {gerando && !novo ? (
          <p className="suave">Lendo seus números…</p>
        ) : textoVisto ? (
          <TextoIA texto={textoVisto} />
        ) : (
          <p className="suave">Nenhum diagnóstico deste recorte ainda. Ele só é gerado quando você pede.</p>
        )}
      </div>
      <div className="diag-acoes">
        {gerando ? (
          <button type="button" className="link" onClick={() => controle.current?.abort()}>
            Parar
          </button>
        ) : (
          estadoVisto !== 'atual' && (
            <button type="button" onClick={() => gerar(false)} disabled={ocupado}>
              {estadoVisto === 'nenhum' ? 'Gerar diagnóstico' : 'Refazer'}
            </button>
          )
        )}
      </div>
      <span className={erro ? 'diag-status diag-erro' : 'diag-status'} role="status" aria-live="polite">
        {erro || aviso}
      </span>
    </section>
  );
}
```
